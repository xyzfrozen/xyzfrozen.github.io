每个节点维护的是一个树形结构，而不再是单纯的 `int` 类信息，相当于把结构体替换成了一棵树

时空复杂度在 $O(n\log ^2n \sim n\log^3n)$  不等，用来**在线**查询待修**二维偏序**



[模板题](https://www.luogu.com.cn/problem/P3380)

二维偏序，区间+值域

我们从区间入手，维护一颗线段树，每个节点内部维护一颗平衡树，维护值域

对于查排名，转化为 $\lt x$ 的个数 $+1$ ，查权值，可以直接二分这个权值，转化为查排名

修改操作就是在对应的 $O(\log n)$ 个区间内都插入这个权值

时间复杂度 $O(m\log ^3n)$ ，空间复杂度 $O(n\log n)$

注意这里的排名是算上相同的数了的

```cpp
#include<bits/stdc++.h>
#include<ext/pb_ds/assoc_container.hpp>
#include<ext/pb_ds/tree_policy.hpp>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define mk make_pair
#define IT set<pi>::iterator
using namespace std;
using namespace __gnu_pbds;

const int N=5e4+10;
int n,m,inf=2e9;
int a[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

#define Fhq tree<pi,null_type,less<pi>,rb_tree_tag,tree_order_statistics_node_update>
int l[N<<2],r[N<<2];
Fhq tr[N*22];

void build(int ql,int qr,int idx)
{
    l[idx]=ql,r[idx]=qr;
    for(int i=ql;i<=qr;i++) tr[idx].insert(mk(a[i],i));
    if(ql==qr) return;
    int mid=(ql+qr)>>1;
    build(ql,mid,idx<<1),build(mid+1,qr,idx<<1|1);
}

int order(int ql,int qr,int idx,pi val)
{
    if(ql<=l[idx] && qr>=r[idx])
        return tr[idx].order_of_key(val);
    int mid=(l[idx]+r[idx])>>1,res=0;
    if(ql<=mid) res+=order(ql,qr,idx<<1,val);
    if(qr>mid) res+=order(ql,qr,idx<<1|1,val);
    return res;
}

int key(int ql,int qr,int rnk)
{
    int l=0,r=1e8,ans=0;
    while(l<=r)
    {
        int mid=(l+r)>>1;
        if(order(ql,qr,1,mk(mid,inf))<rnk) l=mid+1;
        else r=mid-1,ans=mid;
    }
    return ans;
}

void modify(int pos,int idx,int v)
{
    tr[idx].erase(mk(a[pos],pos)),tr[idx].insert(mk(v,pos));
    if(l[idx]==r[idx]) return;
    int mid=(l[idx]+r[idx])>>1;
    if(pos<=mid) modify(pos,idx<<1,v);
    else modify(pos,idx<<1|1,v);
}

int pre(int ql,int qr,int val)
{
    int rnk=order(ql,qr,1,mk(val,0))+1;
    if(rnk==1) return -2147483647;
    return key(ql,qr,rnk-1);
}

int nxt(int ql,int qr,int val)
{
    int rnk=order(ql,qr,1,mk(val,inf));
    if(rnk==qr-ql+1) return 2147483647;
    return key(ql,qr,rnk+1);
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++) a[i]=fr();
    build(1,n,1);
    while(m--)
    {
        int op=fr(),l=fr(),r=fr();
        if(op==1) fw(order(l,r,1,mk(fr(),0))+1),nl;
        else if(op==2) fw(key(l,r,fr())),nl;
        else if(op==3) modify(l,1,r),a[l]=r;
        else if(op==4) fw(pre(l,r,fr())),nl;
        else fw(nxt(l,r,fr())),nl;
    }
    return 0;
}
```



我们从值域入手，维护动态开点权值线段树，每个节点内部平衡树维护序列

修改操作就是在 $O(\log W)$ 个区间内部插入这个 $pos$

查排名，我们到所有有贡献的值域区间 $[0,x]$ 内部查询 $pos \in [ql,qr]$ 数的个数，复杂度可以分开考虑

对于不递归的情况，复杂度是 $O(\log n)$

对于递归的情况，至多 $O(\log W)$ 个，而不递归的区间必然是由递归区间产生的，所以可以推得总复杂度 $O(m\log n\log W)$

查权值，线段树二分即可，至多 $O(\log W)$ 个区间，查询 $O(\log n)$

```cpp
#include<ext/pb_ds/assoc_container.hpp>
#include<ext/pb_ds/tree_policy.hpp>
#define pt putchar(' ')

using namespace __gnu_pbds;

const int N=5e4+10;
int n,m,rt,inf=1e8+10,cnt;
int a[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

#define Fhq tree<int,null_type,less<int>,rb_tree_tag,tree_order_statistics_node_update>
int ls[N*33],rs[N*33];
Fhq tr[N*33];

void modify(int vl,int vr,int pos,int &idx,int v,int ty)
{
    if(!idx) idx=++cnt;
    if(ty) tr[idx].insert(pos);
    else tr[idx].erase(pos);
    if(vl==vr) return;
    int mid=(vl+vr)>>1;
    if(v<=mid) modify(vl,mid,pos,ls[idx],v,ty);
    else modify(mid+1,vr,pos,rs[idx],v,ty);
}

int order(int vl,int vr,int ql,int qr,int val,int idx)
{
    if(!idx || vl>=val) return 0;
    if(vr<val) return tr[idx].order_of_key(qr+1)-tr[idx].order_of_key(ql);
    int mid=(vl+vr)>>1;
    return order(vl,mid,ql,qr,val,ls[idx])+order(mid+1,vr,ql,qr,val,rs[idx]);
}

int key(int vl,int vr,int ql,int qr,int rnk,int &idx)
{   
    if(!idx) idx=++cnt;
    if(vl==vr) return vl;
    int cnt=tr[ls[idx]].order_of_key(qr+1)-tr[ls[idx]].order_of_key(ql);
    int mid=(vl+vr)>>1;
    if(cnt>=rnk) return key(vl,mid,ql,qr,rnk,ls[idx]);
    return key(mid+1,vr,ql,qr,rnk-cnt,rs[idx]);
}

int pre(int ql,int qr,int val)
{
    int rnk=order(0,inf,ql,qr,val,rt)+1;
    if(rnk==1) return -2147483647;
    return key(0,inf,ql,qr,rnk-1,rt);
}

int nxt(int ql,int qr,int val)
{
    int rnk=order(0,inf,ql,qr,val+1,rt);
    if(rnk==qr-ql+1) return 2147483647;
    return key(0,inf,ql,qr,rnk+1,rt);
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++) modify(0,inf,i,rt,a[i]=fr(),1);
    for(int i=1;i<=m;i++)
    {
        int op=fr(),l=fr(),r=fr();
        if(op==1) fw(order(0,inf,l,r,fr(),rt)+1),nl;
        else if(op==2) fw(key(0,inf,l,r,fr(),rt)),nl;
        else if(op==3) modify(0,inf,l,rt,a[l],0),modify(0,inf,l,rt,a[l]=r,1);
        else if(op==4) fw(pre(l,r,fr())),nl;
        else fw(nxt(l,r,fr())),nl;
    }
    return 0;
}
```

[P3157 [CQOI2011\] 动态逆序对 - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/P3157)

删除一个数的贡献是，左侧比它大的个数+右侧比它小的数，用树套树维护二维偏序即可

```cpp
#include<bits/stdc++.h>
#include<ext/pb_ds/assoc_container.hpp>
#include<ext/pb_ds/tree_policy.hpp>
#define ll long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;
using namespace __gnu_pbds;

const int N=1e5+10;
int n,m,cnt,rt;ll ans;
int a[N],s[N],pos[N];

#define Fhq tree<int,null_type,less<int>,rb_tree_tag,tree_order_statistics_node_update>
int ls[N*20],rs[N*20];
Fhq tr[N*20];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
void add(int i,int x){for(;i<=n;i+=i&(-i)) s[i]+=x;}
int query(int i,int res=0){for(;i;i-=i&(-i)) res+=s[i];return res;}

void modify(int vl,int vr,int pos,int &idx,int val,int ty)
{
    if(!idx) idx=++cnt;
    if(ty) tr[idx].insert(pos);
    else tr[idx].erase(pos);
    if(vl==vr) return;
    int mid=(vl+vr)>>1;
    if(val<=mid) modify(vl,mid,pos,ls[idx],val,ty);
    else modify(mid+1,vr,pos,rs[idx],val,ty);
}

int lower(int vl,int vr,int ql,int qr,int idx,int val)
{
    if(!idx || vl>=val) return 0;
    if(vr<=val) return tr[idx].size()-tr[idx].order_of_key(ql);
    int mid=(vl+vr)>>1;
    return lower(vl,mid,ql,qr,ls[idx],val)+lower(mid+1,vr,ql,qr,rs[idx],val);
}

int upper(int vl,int vr,int ql,int qr,int idx,int val)
{
    if(!idx || vr<=val) return 0;
    if(vl>=val) return tr[idx].order_of_key(qr+1);
    int mid=(vl+vr)>>1;
    return upper(vl,mid,ql,qr,ls[idx],val)+upper(mid+1,vr,ql,qr,rs[idx],val);
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++)
    {
        ans+=i-1-query(a[i]=fr()),add(a[i],1);
        modify(1,n,pos[a[i]]=i,rt,a[i],1);
    }

    //左侧比它大的个数+右侧比它小的数
    while(m--)
    {
        printf("%lld\n",ans);
        int now=fr();
        modify(1,n,pos[now],rt,now,0);
        ans-=upper(1,n,1,pos[now],rt,now)+lower(1,n,pos[now],n,rt,now);
    }
    return 0;
}
```

[P3332 [ZJOI2013\] K大数查询 - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/P3332)

典中典题，相当于给包含 $c$ 的 $O(\log W)$ 个区间插入了 $[ql,qr]$ 这个区间

插入区间，我们自然想到一个本身就是维护区间的区间线段树，相当于区间加+查询区间和

全部采用动态开点，时空复杂度 $O(m\log W\log n)$ ，需要注意的是下传 `lazy tag` 的时候一定要对空的儿子节点新建并算贡献

```cpp
void chs(int idx,int vl,int vr)
{
    if(!tag[idx]) return;
    int mid=(vl+vr)>>1ll;
    if(!ls[idx]) ls[idx]=++cnt;
    if(!rs[idx]) rs[idx]=++cnt;
    Tag(ls[idx],tag[idx],mid-vl+1);
    Tag(rs[idx],tag[idx],vr-mid);
    tag[idx]=0;
}
```

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define IT set<node>::iterator
using namespace std;

const int N=5e4+10;
int n,m,cnt,R;
int ls[N*400],rs[N*400],rt[N*400],sum[N*400],tag[N*400];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

struct Seg_Tree
{
    void Tag(int idx,int v,int len){sum[idx]+=len*v,tag[idx]+=v;}
    void chs(int idx,int vl,int vr)
    {
        if(!tag[idx]) return;
        int mid=(vl+vr)>>1ll;
        if(!ls[idx]) ls[idx]=++cnt;
        if(!rs[idx]) rs[idx]=++cnt;
        Tag(ls[idx],tag[idx],mid-vl+1);
        Tag(rs[idx],tag[idx],vr-mid);
        tag[idx]=0;
    }

    void modify(int vl,int vr,int ql,int qr,int &idx)
    {
        if(!idx) idx=++cnt;
        if(ql<=vl && qr>=vr)
            return void(Tag(idx,1,vr-vl+1));
        
        chs(idx,vl,vr);
        int mid=(vl+vr)>>1ll;
        if(ql<=mid) modify(vl,mid,ql,qr,ls[idx]);
        if(qr>mid) modify(mid+1,vr,ql,qr,rs[idx]);
        sum[idx]=sum[ls[idx]]+sum[rs[idx]];
    }

    int query(int vl,int vr,int ql,int qr,int idx)
    {
        if(!idx) return 0;
        if(ql<=vl && qr>=vr) return sum[idx];
        
        chs(idx,vl,vr);
        int mid=(vl+vr)>>1ll,res=0;
        if(ql<=mid) res+=query(vl,mid,ql,qr,ls[idx]);
        if(qr>mid) res+=query(mid+1,vr,ql,qr,rs[idx]);
        return res;
    }
}Seg;

void modify(int vl,int vr,int ql,int qr,int &idx,int val)
{
    if(!idx) idx=++cnt;
    Seg.modify(1,n,ql,qr,rt[idx]);
    if(vl==vr) return;
    int mid=(vl+vr)>>1ll;
    val<=mid?modify(vl,mid,ql,qr,ls[idx],val):modify(mid+1,vr,ql,qr,rs[idx],val);
}

int query(int vl,int vr,int ql,int qr,int &idx,int rnk)
{
    if(vl==vr) return vl;
    if(!idx) idx=++cnt;
    int cnt=Seg.query(1,n,ql,qr,rt[rs[idx]]),mid=(vl+vr)>>1ll;
    if(cnt<rnk) return query(vl,mid,ql,qr,ls[idx],rnk-cnt);
    return query(mid+1,vr,ql,qr,rs[idx],rnk);
}

signed main()
{
    n=fr(),m=fr();
    while(m--)
    {
        int op=fr(),l=fr(),r=fr(),c=fr();
        if(op&1) modify(-n,n,l,r,R,c);
        else fw(query(-n,n,l,r,R,c)),nl;
    }
    return 0;
}
```
