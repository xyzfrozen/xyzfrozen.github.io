简单的情况就是 $B_i=0$ 这就是线性 $Dp$ 

考虑到 $B_i \leq 7$ 我们可以考虑状压后面人的状态，去除后效性，即 $f_{i,S}$ 表示 $i$ 及其后面 $7$ 个人的状态

同时我们还需要知道上一个人是谁，然后枚举这一次选谁转移，还要再开一维

$f_{i,S,k}$ 表示做完了 $i-1$ 个人，$i$ 和后面 $7$ 个人的状态是 $S$，上一个人是 $i+k$

这里用了一个小技巧就是上一个人用 $i+k$ 表示，这样就优化了一维 $O(n)$ 的空间，同时考虑到 $k$ 可能为负数，还需要加上一个偏移量 $P$ 

这里 $k\in [-8,7]$ ，为什么是 $-8$ ？可能恰好 $i-8$ 由 $i-1$ 做掉了，所以是可以取到的

注意第一道菜不需要时间，即上一个人的 $i+k \leq 0$ 需要特判一下

注意判断合法性的时候，$R$ 只用考虑 $0$ 的，已经填过的就不用考虑 $B_i$ 了！！！

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e3+10,M=(1<<8)+10,P=10;
int n,m=8,ans,inf=0x3f3f3f3f;
int w[N],r[N],f[N][M][25];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
int calc(int a,int b){return (a<=0)?0:(w[a]^w[b]);}

void solve()
{
	n=fr(),ans=inf;
	for(int i=1;i<=n;i++)
		w[i]=fr(),r[i]=fr();
	
	//i-1 做完 i 和后面 7 个人的状态为 S 上一个人是 i+k
	memset(f,0x3f,sizeof f);
	f[1][0][-1+P]=0;
	for(int i=1;i<=n;i++)
		for(int S=0;S<=(1<<m)-1;S++)
			for(int k=-8;k<=7;k++)
			{
				if(S&1) f[i+1][S>>1][k-1+P]=min(f[i+1][S>>1][k-1+P],f[i][S][k+P]); //i 做掉了
				else //i 没做 i 不做就没机会做了，必须现在做
				{
					for(int p=0,R=n;p<=7;p++)
					{
						if(!((S>>p)&1))
						{
                               R=min(R,i+p+r[i+p]); //R 只用考虑 0 的，已经填过的就不用考虑 B_i 了！！！
							if(i+p>R) break;
							f[i][S|(1<<p)][p+P]=min(f[i][S|(1<<p)][p+P],f[i][S][k+P]+calc(i+k,i+p));
						}
					}
				}
			}
	for(int k=-8;k<=0;k++)
		ans=min(ans,f[n+1][0][k+P]);
	fw(ans),nl;
}

int main()
{
	int T=fr();
	while(T--) solve();

	return 0;
}
```

相似的题：[ARC132C] Almost Sorted

给定一个长度为 $n$ 的数字序列 $A$，由 $1$ 到 $n$ 之间的整数和 $-1$ 组成。还有一个整数 $d$

现在要对这个序列进行变换，将 $A$ 中所有为 $-1$ 的 $a_i$ 替换成一个数字，使得得到的序列 $P$，满足：
- $\forall a_i \ne -1,p_i = a_i$
- $P$ 是 $1$ 到 $n$ 的一个排列
- $\forall |p_i-i| \leq m$

试问有多少种这样的排列 $P$。答案对 $998244353$ 取膜，保证输入的序列合法

- $ 1\ \le\ m\ \le\ 5 $
- $ m\ <\ n\ \le\ 500 $

这个题有一个重要的发现就是完全不用管排列这个限制！！不是排列根本不可能合法

还是设 $f_{i,S}$ 表示做完了前 $i$ 个数，$[i-m,i+m]$ 用掉的情况为 $S$，这样设计必须都全部填满，就保证是个排列了

这里既然是填了前 $i$ 个数，就刷表，因为是决策 $i+1$ 上面是填了 $i-1$，就填表，因为决策是 $i$

注意判断 $i+1+p \geq 1$ 一定要写到 $\text{if}$ 里面，不能写到 $\text{for}$ 的判断条件里面，不然后面就算合法也会被前面就 $\text{break}$ 了

```cpp
// LUOGU_RID: 127178106
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=510,M=(1<<11)+10,Q=998244353;
int n,m,k;
int a[N],f[N][M];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

signed main()
{
	n=fr(),m=fr(),k=2*m+1;
	for(int i=1;i<=n;i++) a[i]=fr();
	
	//i 做完，[i-m,i+m] 用掉的状态为 S
	f[0][1<<m]=1; //相当于只用掉了 0
	for(int i=0;i<n;i++)
		for(int S=0;S<=(1<<k)-1;S++)
		{
			if(~a[i+1])
			{
				if(a[i+1]==i+6 || !((S>>(a[i+1]-(i+1)+m+1))&1))
					(f[i+1][(S>>1)|(1<<(a[i+1]-(i+1)+m))]+=f[i][S])%=Q;
			}	
			else
			{
				for(int p=-m;p<=m && i+1+p<=n;p++)
					if(i+1+p>=1 && (p==m || !((S>>m+p+1)&1))) (f[i+1][(S>>1)|(1<<m+p)]+=f[i][S])%=Q; //注意判断 i+1+p>=1 一定要写到 if 里面，不能写到 for 的判断条件里面，不然后面就算合法也会被前面就 break 了
			}
		}
	fw(f[n][(1<<m+1)-1]);
	return 0;
}
```
