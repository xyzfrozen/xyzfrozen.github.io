我们懒得写左偏树了，不妨类似平衡树的模式，随机合并

合并两个根的时候，按照 $heap$ 左右儿子合并

那么这个复杂度级别应该是和 $fhq$ 差不多的



基础定义

```cpp
struct Heap{
	int ls,rs;
	int val,heap; //权值 堆值
}tr[N<<1];
```

从上到下按照 $val$ 满足堆的结构，根儿子顺序满足按照 $heap$ 满足堆的结构



$add$

```cpp
int add(int x){tr[++idx]={0,0,x,Rand()};return idx;}
```



$merge$ （小根堆）

```cpp
int merge(int l,int r)
{
	if(!l || !r) return l|r;
	if(tr[l].val>tr[r].val) swap(l,r);
	if(tr[l].heap<tr[r].heap)
		tr[l].rs=merge(tr[l].rs,r);
	else tr[l].ls=merge(tr[l].ls,r);
	return l;
}
```



[模板题](https://www.luogu.com.cn/problem/P3377)

加入了一个查询堆顶的操作和删除堆元素的操作

我们显然考虑并查集维护

在 `merge` 加入这一句

```cpp
p[tr[l].ls]=p[tr[l].rs]=p[l]=l;
```

同时这里还要判断权值相等按编号排序

```cpp
if(tr[l].val>tr[r].val || (tr[l].val==tr[r].val && l>r)) swap(l,r);
```



$pop$ 操作

我们首先求出堆顶 $px$ ，然后把两个儿子的堆顶指定为他们自己，然后合并两个儿子

然后一定要把 $px$ 的堆顶指定为 `merge(l,r)` ，这是因为修改堆只会修改一条路径，还有一个儿子的父亲因为路径压缩了，所以还是 $px$，这时候要把 $px$ 的堆顶（父亲）强行修改，才能修改这些父亲仍然是 $px$ 的节点

同时我们把 $px.val$ 改成 $-1$，这样就旋到低下去了，保证正确性

$O(n\log^2n)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10;
int n,m,op,x,y,idx;
mt19937 Rand(time(0)^99994919);
int p[N],vis[N];
struct Heap{
	int ls,rs;
	int val,heap;
}tr[N<<1];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int find(int x)
{
	if(p[x]!=x) p[x]=find(p[x]);
	return p[x];
}

int add(int x){tr[++idx]={0,0,x,Rand()};return idx;}
int merge(int l,int r)
{
	if(!l || !r) return l|r;
	if(tr[l].val>tr[r].val || (tr[l].val==tr[r].val && l>r)) swap(l,r);
	if(tr[l].heap<tr[r].heap)
		tr[l].rs=merge(tr[l].rs,r);
	else tr[l].ls=merge(tr[l].ls,r);
	p[tr[l].ls]=p[tr[l].rs]=p[l]=l;
	return l;
}

int main()
{
	n=fr(),m=fr();
	for(int i=1;i<=n;i++)
		p[i]=i,add(fr());
	while(m--)
	{
		op=fr();
		if(op&1)
		{
			x=fr(),y=fr();
			if(vis[x]|vis[y]) continue;
			int px=find(x),py=find(y);
			if(px^py)
				merge(px,py);
		}
		else
		{
			x=fr();
			if(vis[x]) {puts("-1");continue;}
			int px=find(x);
			fw(tr[px].val),nl;
			vis[px]=1;
			tr[px].val=-1;
			p[tr[px].ls]=tr[px].ls;
			p[tr[px].rs]=tr[px].rs;
			p[px]=merge(tr[px].ls,tr[px].rs);
		}
	}

	return 0;
}
```

[例题](https://www.luogu.com.cn/problem/P4331)

[前置文章]([浅谈单调序列绝对值和最小化 - xyzfrozen の 小站](https://xyzfrozen.github.io/undefined/浅谈单调序列绝对值和最小化/浅谈单调序列绝对值和最小化/))

前文已经证明了在两种单调区间的最值情况

我们现在需要做的就是

1. 查询中位数
2. 合并两个区间

我们考虑用随机堆维护

1. 每个堆用一个结构体存它代表的区间 $(l,r)$ ，实际元素个数 $s$，随机堆的堆顶编号 $rt$，中位数 $v$

	我们每次直接暴力弹出堆顶，然后合并两个儿子，直到 $s \leq \lceil \frac{r-l+1}2\rceil$ ，此时这个元素的权值就是中位数

2. 我们把每个点当做一个新区间 $cnt$，然后和左边的区间合并，当且仅当 $seg_{cnt-1}.v \gt seg_{cnt}.v$

	此时我们合并两个堆，拓展右端点，增加实际元素个数

考虑到合并复杂度 $O(\log n)$，每个元素最多被弹 $1$ 次，总复杂度 $O(n\log n)$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e6+10;
int n,idx,cnt,ans;
int a[N],b[N];
struct Heap{
	int ls,rs;
    int val,heap;
}tr[N];
struct node{
	int l,r,s,rt,v;
}seg[N];
mt19937 Rand(99994919);

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int add(int x) {tr[++idx]={0,0,x,rand()};return idx;}
int merge(int l,int r) //维护大根堆
{
	if(!l || !r) return l+r;
	if(tr[l].val>tr[r].val) swap(l,r);
	if(tr[l].heap>tr[r].heap) tr[r].rs=merge(tr[r].rs,l);
    else tr[r].ls=merge(tr[r].ls,l);
    return r;
}

signed main()
{
	n=fr();
	for(int i=1;i<=n;i++) a[i]=fr()-i;
	for(int i=1;i<=n;i++)
	{
		seg[++cnt]={i,i,1,add(a[i]),a[i]};
		while(cnt>1 && seg[cnt-1].v>seg[cnt].v)
		{
			cnt--;
			seg[cnt].rt=merge(seg[cnt].rt,seg[cnt+1].rt);
			seg[cnt].s+=seg[cnt+1].s;
			seg[cnt].r=seg[cnt+1].r;
			while(seg[cnt].s>(seg[cnt].r-seg[cnt].l+1+1)/2) //上取整
				seg[cnt].s--,seg[cnt].rt=merge(tr[seg[cnt].rt].ls,tr[seg[cnt].rt].rs);
			seg[cnt].v=tr[seg[cnt].rt].val;
		}
	}
	
	for(int i=1;i<=cnt;i++)
		for(int j=seg[i].l;j<=seg[i].r;j++)
			ans+=abs(a[j]-seg[i].v),b[j]=seg[i].v;
	fw(ans),nl;
	for(int i=1;i<=n;i++) fw(b[i]+i),pt;
	return 0;
}
```
