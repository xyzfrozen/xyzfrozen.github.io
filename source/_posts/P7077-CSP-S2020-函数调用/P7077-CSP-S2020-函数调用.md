类似线段树的考虑

首先把赋初值当做 $n$ 次单点修改

一个 $add$ 操作的贡献为其后所有 $mul$ 操作贡献的积 $\times add_v$

一个 $mul$ 操作的贡献同理

考虑到把每个操作当做点，这是一张 $DAG$

设

$Type=1$ $mul_i=1$

$Type=2$ $mul_i=2$

$Type=3$ $mul_i=\prod_{(i,j)} mul_j$

记 $suf_i$ 为按照 $f_i$ 序列执行，从 $i$ 开始的 $mul_i$ 后缀积

则对于操作 $f_i$，所有外部操作对它的贡献为被乘 $suf_{i+1}$，可以把每个函数执行的次数算出来

考虑到 $Type=3$ 的操作不能暴力执行，我们把贡献放到 $1,2$ 上面

考虑操作序列 $P$，当前操作三为操作 $i$

则，$P_1$ 被执行 $\prod_{j=2}^{sz_i} mul_j$，即为这个操作的内部贡献 $\prod_{j=k+1}^{sz_i} mul_k \to f_k$

然后放到 $DAG$ 上面处理

对于操作 $1$，直接给对应位置加上 $add_v \times w[\text{倍数}]$

对于操作 $2$，可直接跳过，对于操作 $3$，往下推就行了

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x]) //注意加了&
using namespace std;

const int N=3e5+10,Q=998244353;
int n,m,k,op;
int a[N],in[N],f[N],mul[N],vis[N],cnt[N];
struct node{int id,p,v;}val[N];
vector<int> as[N];

int fr(){ //double 不能快读！！！！
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
void mod(int &x,int y){if((x+=y)>=Q) x-=Q;}

void topsort()
{
	queue<int> q;
	for(int i=1;i<=m;i++)
		if(!in[i]) q.push(i);
	while(q.size())
	{
		int x=q.front();
		q.pop();
		if(val[x].id==1) mod(a[val[x].p],val[x].v*cnt[x]%Q);
		int w=cnt[x];
		go(v)
		{
			mod(cnt[v],w);
			(w*=mul[v])%=Q;
			if(!(--in[v])) q.push(v);
		}
	}
	
	for(int i=1;i<=n;i++) fw(a[i]),pt;
}

void dfs(int x)
{
	if(vis[x]) return;
	vis[x]=1;
	reverse(as[x].begin(),as[x].end());
	go(v)
	{
		dfs(v);
		(mul[x]*=mul[v])%=Q;
	}
}

signed main()
{
	n=fr();
	for(int i=1;i<=n;i++)
		val[i]={1,f[i]=i,fr()};
	m=fr()+n;
	for(int i=n+1;i<=m;i++)
	{
		op=fr();
		if(op==1) val[i]={1,fr(),fr()};
		else if(op==2) val[i]={2,0,fr()};
		else
		{
			int cnt=fr();
			while(cnt--)
			{
				int v=fr()+n;
				as[i].pb(v),in[v]++;
			}
			val[i]={3,0,0};
		}
	}
	
	k=fr()+n;
	for(int i=n+1;i<=k;i++) f[i]=fr()+n;
	for(int i=1;i<=m;i++)
		mul[i]=(val[i].id==2)?val[i].v:1;
	for(int i=1;i<=m;i++)
		if(val[i].id==3) dfs(i);
	int suf=1;
	for(int i=k;i;i--)
	{
		mod(cnt[f[i]],suf);
		(suf*=mul[f[i]])%=Q;
	}
	topsort();
	
	return 0;
}
```
