# 虚树

## Intro

[【洛谷日报#185】浅谈虚树 - SSerxhs 的博客 - 洛谷博客 (luogu.com.cn)](https://www.luogu.com.cn/blog/SSerxhs/qian-tan-xu-shu)

对于一棵树，如果我们只询问其中一些点，且 $\sum k_i$ 可以接受，我们就考虑每次建出虚树

一颗虚树包含所有询问节点+所有询问节点的 $lca$，总节点数 $\leq 2k-1$​，最劣情况就是都是虚树叶子

称所有询问节点为关键点

构建方法可以看上面那篇博客，大致就是先按 $dfn$ 排序，然后每次增量构造

$O(n\log n)$

```cpp
void ins(int x)
{
    if(!top) return void(ak[++top]=x);
    int p=lca(x,ak[top]);
    while(top>1 && d[ak[top-1]]>d[p])
        link(ak[top],ak[top-1]),top--;
    if(top && d[p]<d[ak[top]]) link(ak[top--],p);
    if(!top || ak[top]^p) ak[++top]=p;
    ak[++top]=x;
}
```



## Problem

[CF613D Kingdom and its Cities](https://www.luogu.com.cn/problem/CF613D)

建出虚树，大力 $dp$，虚树 $dp$​ 就考虑关键点和非关键点

首先特判一下父子都是关键点的无解

当前是关键点，如果儿子中由关键点都要断开

当前不是关键点，如果儿子中的关键点数量仅为 $1$，把当前点也当做关键点上传，否则如果 $\geq 2$ 直接删掉当前点

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=3e5+10;
int n,m,u,v,stm,top,ans;
int dfn[N],lg[N],pos[21][N],d[N],in[N],fa[N],f[N],ak[N];
vector<int> as[N],G[N],S;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
void link(int x,int y){G[x].pb(y),G[y].pb(x);}
void clear(){for(int &x:S) in[x]=0;S.clear();}
int get(int x,int y){return d[x]<d[y]?x:y;}
int lca(int x,int y)
{
    if(x==y) return x;
    if((x=dfn[x])>(y=dfn[y])) swap(x,y);
    int d=lg[y-x++];
    return fa[get(pos[d][x],pos[d][y-(1<<d)+1])];
}

void dfs(int x,int rt)
{
    d[x]=d[fa[x]=rt]+1;
    pos[0][dfn[x]=++stm]=x;
    go(v) if(v^rt) dfs(v,x);
}

void dp(int x,int rt)
{
    if(!f[x])
    {
        for(int &v:G[x])
            if(v^rt) dp(v,x),f[x]+=f[v];
        if(f[x]>=2) ans++,f[x]=0;
    }
    else
    {
        for(int &v:G[x])
            if(v^rt) dp(v,x),ans+=(f[v]!=0);
    }
}

void ins(int x)
{
    f[x]=1;
    if(!top) return void(ak[++top]=x);
    int p=lca(x,ak[top]);
    while(top>1 && d[ak[top-1]]>d[p])
        link(ak[top],ak[top-1]),top--;
    if(top && d[p]<d[ak[top]]) link(ak[top--],p);
    if(!top || ak[top]^p) ak[++top]=p;
    ak[++top]=x;
}

void clear(int x,int rt)
{
    in[x]=f[x]=0;
    for(int &v:G[x]) if(v^rt) clear(v,x);
    G[x].clear();
}

int solve()
{
    ans=top=0;
    int cnt=fr();
    for(int i=1,x;i<=cnt;i++)
        S.pb(x=fr()),in[x]=1;
    for(int &x:S) if(in[fa[x]]) return -1;
    
    auto tmp=[&](const int &a,const int &b){return dfn[a]<dfn[b];};
    sort(S.begin(),S.end(),tmp);
    if(!in[1]) ak[top=1]=1;
    for(int &x:S) ins(x);
    while(top>1) link(ak[top],ak[top-1]),top--;
    return dp(1,0),clear(1,0),ans;
}

int main()
{
    n=fr();
    
    for(int i=1;i<n;i++)
    {
        u=fr(),v=fr();
        as[u].pb(v),as[v].pb(u);
    }

    dfs(1,0);
    lg[0]=-1;
    for(int i=1;i<=n;i++) lg[i]=lg[i>>1]+1;
    for(int k=1;(1<<k)<=n;k++)
        for(int i=1;i+(1<<k)-1<=n;i++)
            pos[k][i]=get(pos[k-1][i],pos[k-1][i+(1<<k-1)]);
    
    m=fr();
    while(m--) fw(solve()),nl,clear();
    return 0;
}
```

```cpp
{
    d[x]=d[fa[0][x]=rt]+1;
    pos[0][dfn[x]=++stm]=x;
    for(int k=1;(1<<k)<=d[x];k++)
    {
        fa[k][x]=fa[k-1][fa[k-1][x]];
        Min[k][x]=min(Min[k-1][x],Min[k-1][fa[k-1][x]]);
    }

{
    if(d[x]<d[y]) return dtnc(y,x);
    int dh=d[x]-d[y]+1,w=inf;
    for(int k=lg[dh];~k;k--)
        if(d[fa[k][x]]>=d[y]) w=min(w,Min[k][x]),x=fa[k][x];
    return w;
}

}

//in[x] & in[v] x
//in[x] & !in[v] x
//!in[x] & in[v]
void dp(int x,int rt)
{
    for(auto &[v,w]:G[x])

        if(v==rt) continue;
        dp(v,x);

}

{

}

    auto tmp=[&](const int &a,const int &b){return dfn[a]<dfn[b];};

    ak[top=1]=1;
    for(int &x:S) ins(x);
```

[P3233 [HNOI2014\] 世界树](https://www.luogu.com.cn/problem/P3233)

看的 $\text{Rainybunny}$ 的题解

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=3e5+10;
int n,m,u,v,stm,top,inf=2e9;
int dfn[N],d[N],fa[22][N],pos[22][N],lg[N];
int ak[N],f[N],bel[N],sz[N],lost[N];
vector<int> as[N],G[N],S;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void dfs(int x,int rt)
{
    d[x]=d[fa[0][x]=rt]+(sz[x]=1);
    pos[0][dfn[x]=++stm]=x;
    for(int k=1;(1<<k)<=d[x];k++)
        fa[k][x]=fa[k-1][fa[k-1][x]];
    go(v) if(v^rt) dfs(v,x),sz[x]+=sz[v];
}

int get(int x,int y){return d[x]<d[y]?x:y;}
int lca(int x,int y)
{
    if(x==y) return x;
    if((x=dfn[x])>(y=dfn[y])) swap(x,y);
    int d=lg[y-x++];
    return fa[0][get(pos[d][x],pos[d][y-(1<<d)+1])];
}
int len(int x,int y){return d[x]+d[y]-(d[lca(x,y)]<<1);}
int son(int x,int p)
{
    int dh=d[x]-d[p];
    for(int k=lg[dh];~k;k--)
        if(d[fa[k][x]]>d[p]) x=fa[k][x];
    return x;
}

void link(int x,int y){G[x].pb(y),G[y].pb(x);}
void ins(int x)
{
    bel[x]=x;
    if(!top) return void(ak[++top]=x);
    int p=lca(x,ak[top]);
    while(top>1 && d[ak[top-1]]>d[p])
        link(ak[top],ak[top-1]),top--;
    if(top && d[ak[top]]>d[p]) link(ak[top--],p);
    if(!top || ak[top]^p) ak[++top]=p;
    ak[++top]=x;
}

void prw(int x,int rt)
{
    for(int &v:G[x])
    {
        if(v==rt) continue;
        prw(v,x);
        int dx=!bel[x]?inf:d[bel[x]]-d[x],dv=d[bel[v]]-d[x];
        if((dv<dx) || (dv==dx && bel[v]<bel[x])) bel[x]=bel[v]; 
    }
}

void turn(int x,int rt)
{
    for(int &v:G[x])
    {
        if(v==rt) continue;
        int dx=len(v,bel[x]),dv=len(v,bel[v]);
        if((dx<dv) || (dx==dv && bel[x]<bel[v])) bel[v]=bel[x];
        turn(v,x);  
    }
}

bool ok(int x,int v,int y)
{
    if(d[y]<=d[x]) return 0;
    int dx=len(bel[x],y),dv=len(bel[v],y);
    return dx>dv || (dx==dv && bel[x]>bel[v]);
}

void dp(int x,int rt)
{
    lost[x]=sz[x];
    for(int &v:G[x])
    {
        if(v==rt) continue;
        dp(v,x);
        int l=v,r=son(v,x);
        lost[x]-=sz[r];
        if(bel[v]==bel[x]) f[bel[x]]+=sz[r]-sz[v];
        else
        {
            int dh=d[v]-d[x];
            for(int k=lg[dh];~k;k--)
                if(ok(x,v,fa[k][l])) l=fa[k][l];
            f[bel[x]]+=sz[r]-sz[l];
            f[bel[v]]+=sz[l]-sz[v];
        }
    }
    f[bel[x]]+=lost[x];
}

void clear(int x,int rt)
{
    for(int &v:G[x]) if(v^rt) clear(v,x);
    f[x]=bel[x]=lost[x]=0,G[x].clear();
}

void solve()
{
    vector<int> base;
    int cnt=fr();
    while(cnt--) S.pb(fr());
    base=S;

    auto tmp=[&](const int &a,const int &b){return dfn[a]<dfn[b];};
    sort(S.begin(),S.end(),tmp);
    if(S[0]^1) ak[top=1]=1;
    for(int &x:S) ins(x);
    while(top>1) link(ak[top],ak[top-1]),top--;
    prw(1,0),turn(1,0),dp(1,0);
    for(int &x:base) fw(f[x]),pt;
    nl,clear(1,0),S.clear(),top=0;
}

int main()
{
    n=fr();
    for(int i=1;i<n;i++)
    {
        u=fr(),v=fr();
        as[u].pb(v),as[v].pb(u);
    }

    lg[0]=-1,dfs(1,0);
    for(int i=1;i<=n;i++) lg[i]=lg[i>>1]+1;
    for(int k=1;(1<<k)<=n;k++)
        for(int i=1;i+(1<<k)-1<=n;i++)
            pos[k][i]=get(pos[k-1][i],pos[k-1][i+(1<<k-1)]);

    m=fr();
    while(m--) solve();
    return 0;
}
```
