题意：能否把编号在 $[l,r]$ 通过填 $?$ 的方式把它变成同一个字符串 $S$，求 $S$ 的个数

我们考虑按每一列来做，这一列的情况有几种

1. $1,0$ 都有，不合法
2. 只有 $1,?/0,?$ 答案为 $1$
3. 只有 $?$ 答案为 $2$

我们考虑把每一行出现 $0$ 和 $1$ 的地方压成两个二进制数，然后直接放到线段树上维护即可

具体就是合并两个儿子 $or$ 一下即可

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10,M=35;
int n,m,q,op,l,r,x,ans;
int z[N],o[N];
char s[M];
struct node{
	int l,r;
	int ov,zv;
}tr[N*4];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void chf(int idx)
{
	node &t=tr[idx],&ls=tr[idx<<1],&rs=tr[idx<<1|1];
	t.ov=ls.ov|rs.ov;
	t.zv=ls.zv|rs.zv;
}

void build(int ql,int qr,int idx)
{
	tr[idx]={ql,qr};
	if(ql==qr)
	{
		tr[idx].ov=o[ql];
		tr[idx].zv=z[ql];
		return;
	}
	int mid=(ql+qr)>>1;
	build(ql,mid,idx<<1);
	build(mid+1,qr,idx<<1|1);
	chf(idx);
}

void modify(int ql,int qr,int idx)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r)
	{
		t.ov=o[ql];
		t.zv=z[ql];
		return;
	}
	
	int mid=(t.l+t.r)>>1;
	if(ql<=mid) modify(ql,qr,idx<<1);
	if(qr>mid) modify(ql,qr,idx<<1|1);
	chf(idx);
}

node query(int ql,int qr,int idx)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r)
		return t;
	int mid=(t.l+t.r)>>1;
	if(qr<=mid) return query(ql,qr,idx<<1);
	if(ql>mid) return query(ql,qr,idx<<1|1);
	node ans,ls=query(ql,qr,idx<<1),rs=query(ql,qr,idx<<1|1);
	ans.ov=ls.ov|rs.ov;
	ans.zv=ls.zv|rs.zv;
	return ans;
}

signed main()
{
	m=fr(),n=fr(),q=fr();
	for(int i=1;i<=n;i++)
	{
		scanf("%s",s);
		for(int j=0;j<m;j++)
		{
			if(s[j]=='1') o[i]|=(1<<j);
			if(s[j]=='0') z[i]|=(1<<j);
		}
	}
	
	build(1,n,1);
	while(q--)
	{
		op=fr();
		if(!op)
		{
			l=fr(),r=fr();
			node t=query(l,r,1);
			if(t.ov&t.zv) continue;
			int res=1;
			for(int j=0;j<m;j++)
			{
				if((t.ov>>j)&1 || (t.zv>>j)&1) continue;
				res<<=1ll;
			}
			ans^=res;
		}
		else
		{
			x=fr();
			scanf("%s",s);
			o[x]=z[x]=0;
			for(int j=0;j<m;j++)
			{
				if(s[j]=='1') o[x]|=(1<<j);
				if(s[j]=='0') z[x]|=(1<<j);
			}
			modify(x,x,1);
		}
	}
	fw(ans);
	return 0;
}
```
