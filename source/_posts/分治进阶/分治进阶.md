# 进阶模型 1

## Intro

我们需要处理 $[l,r]$ 的询问，并且执行了 $[1,l-1] \cup [r+1,n]$ 的所有操作，是区间维护前后缀信息求并的进一步拓展，可以用分治这种结构来降低复杂度

分治时递归，类似线段树分治，进入左侧前加入右侧操作，返回时删除，右侧同理，对应询问区间查询答案即可

注意这种方法一般都是用于 $[i,i]$ 这样的询问，复杂度可以做到 $O(n\log nV)$，$O(V)$ 为单次插入的复杂度

 

## Sol

[P5227 [AHOI2013\] 连通图](https://www.luogu.com.cn/problem/P5227)

维护并查集 $size$，并在底层查询 $[i,i]$ 信息，并且加入了 $[1,i-1] \cup [i+1,n]$ 的信息

启发式合并修改撤销即可，注意有一些边不在任何 $[1,k]$ 集合内部，我们要一开始加上，而且一条边之中加 $1$ 次就行了，我们用一个 $cnt$ 表示每条边需要被执行的次数，我们只在最后一次执行，回溯时记得加回来 $cnt$，当然你第一次执行也可以，用个 $ban$ 标记被使用都行

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define vec vector<node>
using namespace std;

const int N=2e5+10;
int n,m,k,u,v;
int p[N],sz[N],cnt[N];
vector<int> id[N];
struct node{int x,y;}ed[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int find(int x){return (p[x]^x)?find(p[x]):x;}
void del(node &v){sz[v.y]-=sz[v.x],p[v.x]=v.x;}
void merge(int x,int y,vec &now)
{
    x=find(x),y=find(y);
    if(!(x^y)) return;
    if(sz[x]>sz[y]) swap(x,y);
    sz[y]+=sz[x],p[x]=y;
    now.pb({x,y});
}

void query(int vl,int vr)
{
    if(vl==vr) return void(puts((sz[find(1)]^n)?"Disconnected":"Connected"));
    int mid=(vl+vr)>>1;vec now;
    for(int i=mid+1;i<=vr;i++)
        for(int &v:id[i])
            if(!--cnt[v]) merge(ed[v].x,ed[v].y,now);
    query(vl,mid);
    while(now.size()) del(now.back()),now.pop_back();
    for(int i=mid+1;i<=vr;i++)
        for(int &v:id[i]) cnt[v]++;

    for(int i=vl;i<=mid;i++)
        for(int &v:id[i])
            if(!--cnt[v]) merge(ed[v].x,ed[v].y,now);
    query(mid+1,vr);
    while(now.size()) del(now.back()),now.pop_back();
    now.clear();
    for(int i=vl;i<=mid;i++)
        for(int &v:id[i]) cnt[v]++;
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=m;i++)
        ed[i].x=fr(),ed[i].y=fr();
    iota(p+1,p+1+n,1);
    fill(sz+1,sz+1+n,1);

    k=fr();
    for(int i=1;i<=k;i++)
        for(int ct=fr(),x;ct;ct--)
            id[i].pb(x=fr()),cnt[x]++;
    vec now;
    for(int i=1;i<=m;i++)
        if(!cnt[i]) merge(ed[i].x,ed[i].y,now);
    query(1,k);
    return 0;
}
```

# 进阶模型 2

## Intro

[P1972 [SDOI2009\] HH的项链 ](https://www.luogu.com.cn/problem/P1972)不带修

[P1903 [国家集训队\] 数颜色 / 维护队列](https://www.luogu.com.cn/problem/P1903) 单点修

[P4690 [Ynoi2016\] 镜中的昆虫 ](https://www.luogu.com.cn/problem/P4690)区间修

参考博客：[题解 P4690 [Ynoi2016\] 镜中的昆虫 - 251Sec 的博客](https://www.luogu.com.cn/blog/251Sec/solution-p4690)

三道例题的思想都是区间前缀信息加和

分治的本质就是考察部分信息对部分信息的贡献，并且进行一些代数运算得到结果，这也是分治为什么能解决修改的本质原因

`cdq` 解决偏序问题 $O(n\log^{k-1}n)$，空间线性，这是它的优势所在，我们要做的就是转化成偏序问题



区间颜色问题，我们可以考察每个颜色最前面的那个颜色的贡献，我们观察它有什么特行，从而进行统计

我们发现是它对应颜色的上一个出现位置 $pre_i \lt l$，这就是偏序，问题转化成 $l \leq p \leq r\,\land\,pre_p \lt l$ 的 $p$ 的个数



## Sol

每个序列上的点映射为二维平面上的点 $(pos,pre) \Leftrightarrow (x,y)$

**不带修**

区间查询 $[x \leq r,y \leq l-1]-[x \leq l-1,y \leq l-1]$

`cdq` 维护二维偏序即可，$O(n\log n)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=4e6+10;
int n,m,cnt;
int lt[N],ans[N],s[N];
struct node{int x,y,ty,id;}q[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void solve(int l,int r)
{
    if(l>=r) return;
    int mid=(l+r)>>1;
    solve(l,mid),solve(mid+1,r);
    
    int i=l,res=0;
    for(int j=mid+1;j<=r;j++)
    {
        while(q[i].y<=q[j].y && i<=mid)
            res+=!q[i++].id;
        if(q[j].id) ans[q[j].id]+=q[j].ty*res;
    }

    auto tmp=[](const node &a,const node &b){return (!(a.y^b.y))?((!(a.x^b.x))?!a.id:a.x<b.x):a.y<b.y;};
    inplace_merge(q+l,q+mid+1,q+r+1,tmp);
}

int main()
{
    n=fr();
    for(int i=1;i<=n;i++)
    {
        int x=fr();
        q[++cnt]={i,lt[x]};
        lt[x]=i;
    }
    
    m=fr();
    for(int i=1;i<=m;i++)
    {
        int l=fr(),r=fr();
        q[++cnt]={l-1,l-1,-1,i};
        q[++cnt]={r,l-1,1,i};
    }

    auto tmp=[](const node &a,const node &b){return (!(a.x^b.x))?((!(a.y^b.y))?!a.id:a.y<b.y):a.x<b.x;};
    sort(q+1,q+1+cnt,tmp);
    solve(1,cnt);
    for(int i=1;i<=m;i++) fw(ans[i]),nl;
    return 0;
}
```



**单点修改**

修改就强行加入一维 $t$ 表示时间即可

我们考察一次单点修改至多影响 $3$ 个点的 $pre$ 值，注意我们只是对平面点对的做了一个信息和，每个节点的点权不再是 $1$，变成 $1$ 和 $-1$，这样就可以完成修改 $\to$ 删除点+加入点，总点数 $O(6m+n)$

维护 $pre$，和修改就对每个颜色开一个权值存位置的 `set`，$O(\log n)$ 查询修改

`cdq` 维护三维偏序即可，$O((m+n)\log ^2n)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=4e6+10,M=1e6+10;
int n,m,cnt,mx;
int lt[M],ans[M],a[M],pr[M],is_query[M],ct[M];
struct node{int x,y,ty,id;}q[N];
set<int> s[M];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
void add(int i,int x){for(;i<=mx;i+=i&(-i)) ct[i]+=x;}
int query(int i,int res=0){for(;i;i-=i&(-i)) res+=ct[i];return res;}

void solve(int l,int r)
{
    if(l>=r) return;
    int mid=(l+r)>>1;
    solve(l,mid),solve(mid+1,r);
    
    int i=l;
    for(int j=mid+1;j<=r;j++)
    {
        for(;q[i].x<=q[j].x && i<=mid;i++)
            if(!q[i].id) add(q[i].y+1,q[i].ty);
        if(q[j].id) ans[q[j].id]+=q[j].ty*query(q[j].y+1);
    }

    for(int j=l;j<=i-1;j++)
        if(!q[j].id) add(q[j].y+1,-q[j].ty);
    auto tmp=[](const node &a,const node &b){return (!(a.x^b.x))?((!(a.y^b.y))?!a.id:a.y<b.y):a.x<b.x;};
    inplace_merge(q+l,q+mid+1,q+r+1,tmp);
}

void modify(int pos,int v)
{
    q[++cnt]={pos,pr[pos],-1};
    q[++cnt]={pos,v,1};
    pr[pos]=v;
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=1e6;i++) s[i].insert(0);
    for(int i=1;i<=n;i++)
    {
        mx=max(mx,a[i]=fr());
        q[++cnt]={i,lt[a[i]],1};
        pr[i]=lt[a[i]],lt[a[i]]=i;
        s[a[i]].insert(i);
    }
    
    for(int i=1;i<=m;i++)
    {
        char op[2];
        scanf("%s",op);
        if(op[0]=='Q')
        {
            int l=fr(),r=fr();
            q[++cnt]={l-1,l-1,-1,i};
            q[++cnt]={r,l-1,1,i};
            is_query[i]=1;
        }
        else
        {
            int pos=fr(),c=fr(),pvs=a[pos];
            if(pvs==c) continue;
            mx=max(mx,c);
            auto pre=--s[pvs].lower_bound(pos);
            auto suf=s[pvs].upper_bound(pos);
            if(suf!=s[pvs].end()) modify(*suf,*pre); //pr[*suf]=*pre;
            s[pvs].erase(pos);

            a[pos]=c;
            pre=--s[c].lower_bound(pos);
            suf=s[c].upper_bound(pos);
            if(suf!=s[c].end()) modify(*suf,pos); //pr[*suf]=pos;
            modify(pos,*pre); //pr[pos]=*pre;
            s[c].insert(pos);
        }
    }

    solve(1,cnt);
    for(int i=1;i<=m;i++) 
        if(is_query[i]) fw(ans[i]),nl;
    return 0;
}
```



**区间修改**

我们想到 `ODT` 的区间推平操作，是不是也可以这样呢？

对一段极长颜色段压成一个点，我们考察到对于每个点 $[l,r,v]$，只会修改 $pre_l$

每次推平 $O(cnt)$ 个点，势能不变，点数减少 $O(cnt)$，最多裂出来 $O(1)(\text{3 个})$ 节点，势能增加 $O(3m)$，总势能 $O(n+m)$，复杂度 $O((n+m)\log n)$

维护这个需要对每个颜色开一个 $pos$ 的 `ODT`，并对全局开一个 `ODT`

三维偏序复杂度不变，$O((n+n)\log^2n)$

注意，我们仍然是多次单点修改操作，只是我们经过分析，如果我们每次修改的是段，而不是点，复杂度仍然是可行的，`ODT` 只是用来维护段的工具，并不能保证复杂度

`cdq` 的树状数组存的权值 $pre$ 是 $[1\sim n]$ ！！！！

如果要返回 `iterator `好像不能用结构体自定义函数，`insert` 的返回值会被解析成 `void` 非常神秘啊

#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])

const int N=6e5+10,M=1e5+10,V=2e5+10;
int n,m,cnt;
int lt[V],ans[M],a[M],pr[M],is_query[M],ct[V];
struct node{int l,r,v;};
bool operator<(const node &a,const node &b){return a.l<b.l;}
struct Ans{int x,y,ty,id;}q[N<<1];
set<node> s[V],cly;
unordered_map<int,int> id;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;

    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
void add(int i,int x){for(;i<=n;i+=i&(-i)) ct[i]+=x;}
int query(int i,int res=0){for(;i;i-=i&(-i)) res+=ct[i];return res;}

void solve(int l,int r)
{
    if(l>=r) return;
    int mid=(l+r)>>1;
    solve(l,mid),solve(mid+1,r);
    
    int i=l;
    for(int j=mid+1;j<=r;j++)
    {
        for(;q[i].x<=q[j].x && i<=mid;i++)
            if(!q[i].id) add(q[i].y+1,q[i].ty);
        if(q[j].id) ans[q[j].id]+=q[j].ty*query(q[j].y+1);
    }

    for(int j=l;j<=i-1;j++)
        if(!q[j].id) add(q[j].y+1,-q[j].ty);
    auto tmp=[](const Ans &a,const Ans &b){return (!(a.x^b.x))?((!(a.y^b.y))?!a.id:a.y<b.y):a.x<b.x;};
    inplace_merge(q+l,q+mid+1,q+r+1,tmp);
}

void modify(int pos,int v)
{
	if(!(v^pr[pos])) return;
    q[++cnt]={pos,pr[pos],-1};
    q[++cnt]={pos,v,1};
    pr[pos]=v;
}

#define IT set<node>::iterator
IT ins(int l,int r,int v)
{
    s[v].insert({l,r,v});
    return cly.insert({l,r,v}).first;
}
void del(int l,int r,int v)
{
    s[v].erase({l,r,v});
    cly.erase({l,r,v});
}

IT split(int p)
{
    IT now=cly.lower_bound({p,0,0});
    if(now!=cly.end() && now->l==p) return now;
    now=prev(now);
    if(now->r<p) return cly.end();
    int l=now->l,r=now->r,v=now->v;
    del(l,r,v);
    ins(l,p-1,v);
    return ins(p,r,v);
}

void assign(int l,int r,int v)
{
    IT R=split(r+1),L=split(l);
    for(IT now=L;now!=R;now++)
    {
        int sl=now->l,sr=now->r,sv=now->v;
        modify(sl,(sl^l)?sl-1:prev(s[v].lower_bound({l,0,0}))->r); //v这个颜色新增了一个点，修改 pr_l
        IT pre=--s[sv].lower_bound({sl,0,0}); //删除了 sv 的一个点，修改 pre & suf
		IT suf=s[sv].upper_bound({sr,0,0});
        if(suf!=s[sv].end()) modify(suf->l,pre->r);
        s[sv].erase({sl,sr,sv});
    }
    
    cly.erase(L,R);
    ins(l,r,v);
    IT suf=s[v].upper_bound({r,0,0});
    if(suf!=s[v].end()) modify(suf->l,r); //注意最后这里整个段合成一个了还要修改一下 v 的 suf
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=2e5;i++) s[i].insert({0,0,0});
    for(int i=1;i<=n;i++)
    {
    	a[i]=fr();
    	if(!id.count(a[i])) id[a[i]]=id.size()+1;
    	a[i]=id[a[i]];
        q[++cnt]={i,lt[a[i]],1};
        ins(i,i,a[i]);
        pr[i]=lt[a[i]],lt[a[i]]=i;
    }
    
    for(int i=1;i<=m;i++)
    {
        if(fr()&1)
        {
            int l=fr(),r=fr(),c=fr();
            if(!id.count(c)) id[c]=id.size()+1;
            c=id[c];
            assign(l,r,c);
        }
        else
        {
            int l=fr(),r=fr();
            q[++cnt]={l-1,l-1,-1,i};
            q[++cnt]={r,l-1,1,i};
            is_query[i]=1;
        }
    }
	
    solve(1,cnt);
    for(int i=1;i<=m;i++) 
        if(is_query[i]) fw(ans[i]),nl;
    return 0;
}
