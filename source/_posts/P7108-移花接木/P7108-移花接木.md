1. $a=b$

	删掉所有 $h+1$ 的儿子，$a^{h+1}$

2. $a \gt b$

​	考虑数学归纳法，设已经完成了第 $k$ 层，我们只需要移花，最后删掉第 $h$ 层所有的儿子

​	则 $k+1$ 层一共有 $b^{k}\times a$ 个儿子，我们只需要 $b^{k+1}$ 个儿子，所以要操作 $(a-b) \times b^k$ 次
$$
\begin{aligned}
ans&=\sum_{k=0}^{h-1} (a-b)\times b^k + b^h \times a\\
&=(a-b)\sum_{k=0}^{h-1} b^k+b^h \times a\\
&=(a-b) \times \frac{1-b^h}{1-b} +b^h \times a
\end{aligned}
$$

3. $a \lt b$

​	换的时候可以贪心，用第 $ℎ$ 层的树枝嫁接给之前的，最后删掉第 $h$ 层所有的儿子

​	每一层的每一个都需要补 $(b-a)$ 个树枝
$$
\begin{aligned}
ans&=\sum_{k=0}^{h-1} (b-a) \times b^k + b^h \times a -\sum_{k=0}^{h-1} (b-a) \times b^k\\
&=b^h \times a
\end{aligned}
$$

#include<bits/stdc++.h>
#define int long long

using namespace std;

const int Q=1e9+7;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int qkw(int a,int k)
{
	int ans=1,base=a;
	while(k)
	{
		if(k&1) ans=ans*base%Q;
		base=base*base%Q;
		k>>=1;
	}
	return ans;
}

int f(int a1,int q,int n)
{
	if(q==1) return n*a1%Q;
	else return a1*(1-qkw(q,n)+Q)%Q*qkw(1-q+Q,Q-2)%Q;
}

signed main()
{
	int T=fr();
	while(T--)
	{
		int a=fr(),b=fr(),h=fr();
		if(a==b) fw(qkw(a,h+1)),nl;
		else if(a>b) fw(((a-b)*f(1,b,h)%Q+qkw(b,h)*a%Q)%Q),nl;
		else fw(qkw(b,h)*a%Q),nl;
	}

	return 0;
}
