# 线性代数

# 矩阵

$$f_n=7f_{n-1}+6f_{n-2}+5n+4\times3^n$$

我们考虑搞一个 $[f_n,f_{n-1},n,3^n]$

但这样没法完成 $5n$，我们加入一个常数，至于为什么是 $1$，是为了方便 $n\to n+1$

$[f_n,f_{n-1},n,3^n,1]\to[f_{n+1},f_{n},n+1,3^{n+1},1]$

$$
\begin{bmatrix}
7&1&0&0&0\\
6&0&0&0&0\\
5&0&1&0&0\\
12&0&0&3&0\\
5&0&1&0&1\\
\end{bmatrix}
$$

$$f_n=2f_{n-1}+n^3$$

$[f_{n-1},(n-1)^3,(n-1)^2,n-1,1]\to [f_n,n^3,n^2,n,1]$

$$
\begin{bmatrix}
2&0&0&0&0\\
1&1&0&0&0\\
3&3&1&0&0\\
3&3&2&1&0\\
1&1&1&1&1\\
\end{bmatrix}
$$

## 优化 $dp$

这种 $n \leq 10^{18}$ 的一般都是要用矩阵快速幂优化到 $\log n$ 级别来做

**CF1117D Magic Gems**

一眼 $dp$

当前宝石，可变可不变

$$
\begin{cases}
f_i=f_{i-1}+f_{i-m}\;[i \geq m]\\
f_i=1\;[i \lt m]\\
f_0=1
\end{cases}
$$

然后我们发现这个方程是 $O(n)$ 的，考虑矩阵快速幂优化

我们根据矩阵最早的影响原则，可以确定加速矩阵 $A$ 为 $m \times m$ 的

我们希望一次矩阵乘后，每个都向后挪一位

![image](https://cdn.jsdelivr.net/gh/xyzfrozen/Cdn/img/202310010059942.png)

```cpp
const int N=110,Q=1e9+7;
int n,m;
struct node{
	int dot[N][N];
}a,ans,base;

node mul(node &x,node &y)
{
	for(int i=1;i<=m;i++)
	{
		for(int j=1;j<=m;j++)
		{
			base.dot[i][j]=0;
			for(int k=1;k<=m;k++)
			{
				base.dot[i][j]+=x.dot[i][k]*y.dot[k][j];
				base.dot[i][j]%=Q;
			}
		}
	}
	return base;
}

signed main()
{
	n=fr(),m=fr();
	a.dot[1][1]=a.dot[1][m]=1;
	for(int i=2,j=1;i<=m;i++,j++) a.dot[i][j]=1;
	for(int i=1;i<=m;i++) ans.dot[i][i]=1;
	
	int k=n;
	while(k)
	{
		if(k&1) ans=mul(ans,a);
		a=mul(a,a);
		k>>=1;
	}
	fw(ans.dot[1][1]);
	
	return 0;
}
```

**P4910 帕秋莉的手环**

我们不难想到 $dp$

首先断环成链，我们状态机，$0$ 表示结尾金，$1$ 表示结尾木

则 $f_{i,0}=f_{i-1,0}+f_{i-1,1}$，$f_{i,1}=f_{i-1,0}$

然后考虑环的形式

$1+...+1$，$1+....+0$

所以我们可以做两次初始化，然后 $dp$

$f_{1,0}=1$ 和 $f_{1,1}=1$

$ans=f_{n,0}+f_{n,1}$ 和 $f_{n,0}$

然后做一下矩阵加速

$$
\begin{bmatrix}
1&1\\1&0\\
\end{bmatrix}
$$


```cpp
const int Q=1e9+7;
int n;

struct node{
	int dot[3][3];
}a,ans,base,res;

node mul(node &x,node &y)
{
	for(int i=1;i<=2;i++)
		for(int j=1;j<=2;j++)
		{
			base.dot[i][j]=0;
			for(int k=1;k<=2;k++) base.dot[i][j]=(base.dot[i][j]+x.dot[i][k]*y.dot[k][j]%Q)%Q;
		}
	return base;
}

node qkw(node a,int k)
{
	node ans;
	ans.dot[1][1]=ans.dot[2][2]=1,ans.dot[1][2]=ans.dot[2][1]=0;
	node base=a;
	while(k)
	{
		if(k&1) ans=mul(ans,base);
		base=mul(base,base);
		k>>=1;
	}
	return ans;
}

void solve()
{
	int res=0;
	n=fr();
	node T=qkw(a,n-1);

	ans.dot[1][1]=1,ans.dot[1][2]=ans.dot[2][1]=ans.dot[2][2]=0;
	ans=mul(ans,T);
	res=(ans.dot[1][1]+ans.dot[1][2])%Q;

	ans.dot[1][1]=ans.dot[2][1]=ans.dot[2][2]=0,ans.dot[1][2]=1;
	ans=mul(ans,T);
	res=(res+ans.dot[1][1])%Q;
	fw(res),nl;
}

signed main()
{
	a.dot[1][1]=a.dot[1][2]=a.dot[2][1]=1;
	int T=fr();
	while(T--) solve();

	return 0;
}
```

**P3746 [六省联考 2017] 组合数问题**

我们看到组合数下面都是 $nk$ 我们考虑定义式展开，把那个多出来的 $k$ 消掉

$$C_{nk}^{xk+r}=\frac{nk!}{(nk-xk-r)!(xk+r)!}$$

发现数学上消不掉，我们考虑 $dp$

问题等价于，从 $nk$ 个数里面任取 $x[x \equiv r \pmod k]$

$f_{i,j}$ 从前 $i$ 个数里面选，选出数量 $x[x \equiv r \pmod k]$ 的方案数

$$f_{i,j}=f_{i-1,j}+f_{i-1,j-1}$$

这样类似杨辉三角的方程类似于矩阵，我们直接矩阵加速

![image](https://cdn.jsdelivr.net/gh/xyzfrozen/Cdn/img/202310010059639.png)


注意这里矩阵就是从 $0 \to k-1$ 了

$O(k^3\log nk)$

```cpp
const int N=55;
int n,p,K,r;
struct node{
	int dot[N][N];
}a,base,ans;

node mul(node &x,node y)
{
	for(int i=0;i<K;i++)
		for(int j=0;j<K;j++)
			{
				base.dot[i][j]=0;
				for(int k=0;k<K;k++) base.dot[i][j]=(base.dot[i][j]+x.dot[i][k]*y.dot[k][j]%p)%p;
			}
	return base;
}

node qkw(node a,int k)
{
	node ans;
	memset(ans.dot,0,sizeof ans.dot);
	for(int i=0;i<K;i++) ans.dot[i][i]=1;
	
	while(k)
	{
		if(k&1) ans=mul(ans,a);
		a=mul(a,a);
		k>>=1;
	}
	return ans;
}

signed main()
{
	n=fr(),p=fr(),K=fr(),r=fr();
	for(int i=0;i<=K-1;i++)
		a.dot[i][i]++,a.dot[(i-1+K)%K][i]++;
	ans.dot[0][0]=1;
	fw(mul(ans,qkw(a,n*K)).dot[0][r]);
	return 0;
}
```

## 矩阵维护线段树

**CF 1252K**

这个区间操作，我们不难想到分块/线段树

然后因为要考虑顺序，所有我们只能用线段树，考虑优化

我们发现 $A=A+B\;B=B+A$ 是一个线性变换的操作

我们可以用矩阵维护操作，如下：

$\begin{bmatrix}A\;B\end{bmatrix}$$\begin{bmatrix}
1&0\\
1&1
\end{bmatrix}
=\begin{bmatrix}A+B&B\end{bmatrix}$

设一个区间的转移矩阵为 $P$，根据结合律我们知道，查询答案

$$ans=\begin{bmatrix}A\;B\end{bmatrix}\times P_{(L,R)}$$

同样对于操作，我们可以视为矩阵转置

这里因为矩阵不大，我们直接把转置矩阵也存下就行了

$O(k^3n \log n)$ $[k=2]$

```cpp
const int N=1e5+10,Q=1e9+7;
int n,m,op,l,r,x,y;
char s[N];

struct Mat{
	int dot[3][3];
}A,B,I;

struct node{
	int l,r;
	Mat v,w;
	int zz;
}tr[N*4];

Mat mul(Mat x,Mat y)
{
	Mat ans;
	for(int i=1;i<=2;i++)
		for(int j=1;j<=2;j++)
			{
				ans.dot[i][j]=0;
				for(int k=1;k<=2;k++) ans.dot[i][j]=(ans.dot[i][j]+x.dot[i][k]*y.dot[k][j]%Q)%Q;
			}
	return ans;
}

void chf(int idx)
{
	node &t=tr[idx],&ls=tr[idx<<1],&rs=tr[idx<<1|1];
	t.v=mul(ls.v,rs.v);
	t.w=mul(ls.w,rs.w);
}

void chs(int idx)
{
	node &t=tr[idx],&ls=tr[idx<<1],&rs=tr[idx<<1|1];
	if(t.zz)
	{
		ls.zz^=1,rs.zz^=1;
		swap(ls.v,ls.w);
		swap(rs.v,rs.w);
		t.zz=0;
	}
}

void build(int ql,int qr,int idx)
{
	tr[idx]={ql,qr};
	if(ql==qr)
	{
		if(s[ql]=='A') tr[idx].v=A,tr[idx].w=B;
		else tr[idx].v=B,tr[idx].w=A;
		return;
	}
	int mid=(ql+qr)>>1;
	build(ql,mid,idx<<1);
	build(mid+1,qr,idx<<1|1);
	chf(idx);
}

void modify(int ql,int qr,int idx)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r)
	{
		swap(t.v,t.w);
		t.zz^=1;
		return;
	}
	chs(idx);
	int mid=(t.l+t.r)>>1;
	if(ql<=mid) modify(ql,qr,idx<<1);
	if(qr>mid) modify(ql,qr,idx<<1|1);
	chf(idx);
}

Mat query(int ql,int qr,int idx)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r)
		return t.v;
	chs(idx);
	int mid=(t.l+t.r)>>1;
	Mat ans=I;
	if(ql<=mid) ans=mul(ans,query(ql,qr,idx<<1));
	if(qr>mid) ans=mul(ans,query(ql,qr,idx<<1|1));
	return ans;
}

signed main()
{
	A.dot[1][1]=A.dot[2][1]=A.dot[2][2]=1,A.dot[1][2]=0;
	B.dot[1][1]=B.dot[1][2]=B.dot[2][2]=1,B.dot[2][1]=0;
	I.dot[1][1]=I.dot[2][2]=1,I.dot[1][2]=I.dot[2][1]=0;
	
	n=fr(),m=fr();
	scanf("%s",s+1);
	build(1,n,1);
	
	for(int i=1;i<=m;i++)
	{
		op=fr(),l=fr(),r=fr();
		if(op==1) modify(l,r,1);
		else
		{
			x=fr(),y=fr();
			Mat ans;
			ans.dot[1][1]=x,ans.dot[1][2]=y;
			ans=mul(ans,query(l,r,1));
			fw(ans.dot[1][1]),pt,fw(ans.dot[1][2]),nl;
		}
	}

	return 0;
}
```

**CF 718C**

看到 $fibo+1e9/1e18$ 就想矩阵！！！！

直接矩阵优化，注意不要在 $Modify$ 里面用快速幂算矩阵，直接在外面算好再用 $x$ 下传，否则复杂度退化到 $O(n\log^2 n)$

注意 $i$ 到 $2$ ，$a$ 要开到 $[3][3]$ !!!

```cpp
const int N=1e6+10,Q=1e9+7;
int n,m,op,l,r,v;
int a[N];
struct Mat{
	int a[4][4];
	void init(){memset(a,0,sizeof a);}
	Mat operator+(const Mat&p)const{
		Mat c;
		c.init();
		for(int i=1;i<=2;i++)
			for(int j=1;j<=2;j++)
				c.a[i][j]=(a[i][j]+p.a[i][j])%Q;
		return c;
	}
	Mat operator*(const Mat&p)const{
		Mat c;
		c.init();
		for(int i=1;i<=2;i++)
			for(int j=1;j<=2;j++)
			{
				c.a[i][j]+=a[i][1]*p.a[1][j];
				c.a[i][j]+=a[i][2]*p.a[2][j];
				c.a[i][j]%=Q;
			}
		return c;
	}
	bool operator!=(const Mat&p)const{
		return (a[1][1]!=p.a[1][1] || a[1][2]!=p.a[1][2] || 
		a[2][1]!=p.a[1][2] || a[2][2]!=p.a[2][2]);
	}
}I,A,P;
struct node{
	int l,r;
	Mat v,add;
}tr[N*4];

Mat qkw(Mat a,int k)
{
	Mat ans=I,base=a;
	while(k)
	{
		if(k&1) ans=ans*base;
		base=base*base;
		k>>=1;
	}
	return ans;
}

void chf(int idx)
{
	node &t=tr[idx],&ls=tr[idx<<1],&rs=tr[idx<<1|1];
	t.v=ls.v+rs.v;
}

void chs(int idx)
{
	node &t=tr[idx],&ls=tr[idx<<1],&rs=tr[idx<<1|1];
	if(t.add!=I)
	{
		ls.v=ls.v*t.add,rs.v=rs.v*t.add;
		ls.add=ls.add*t.add,rs.add=rs.add*t.add;
		t.add=I;
	}
}

void build(int ql,int qr,int idx)
{
	tr[idx]={ql,qr,I,I};
	if(ql==qr)
	{
		tr[idx].v=A*qkw(P,a[ql]-1);
		return;
	}
	int mid=(ql+qr)>>1;
	build(ql,mid,idx<<1);
	build(mid+1,qr,idx<<1|1);
	chf(idx);
}

void modify(int ql,int qr,int idx,Mat x)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r)
	{
		t.v=t.v*x,t.add=t.add*x;
		return;
	}
	
	chs(idx);
	int mid=(t.l+t.r)>>1;
	if(ql<=mid) modify(ql,qr,idx<<1,x);
	if(qr>mid) modify(ql,qr,idx<<1|1,x);
	chf(idx);
}

Mat query(int ql,int qr,int idx)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r)
		return t.v;
	chs(idx);
	int mid=(t.l+t.r)>>1;
	Mat s;s.init();
	if(ql<=mid) s=s+query(ql,qr,idx<<1);
	if(qr>mid) s=s+query(ql,qr,idx<<1|1);
	return s;
}

signed main()
{
	n=fr(),m=fr();
	for(int i=1;i<=n;i++) a[i]=fr();
	
	I.init();
	I.a[1][1]=I.a[2][2]=1;
	A.init();
	A.a[1][1]=1;
	P.init();
	P.a[1][1]=P.a[1][2]=P.a[2][1]=1;

	build(1,n,1);
	for(int i=1;i<=m;i++)
	{
		op=fr(),l=fr(),r=fr();
		if(op==1) v=fr(),modify(l,r,1,qkw(P,v));
		else fw(query(l,r,1).a[1][1]),nl;
	}

	return 0;
}
```

## 广义矩阵

一般用于图论

**P6569 [NOI Online #3 提高组] 魔法值**

这玩意长得就很线性代数，我们按照套路转换一下，改成矩阵

设 $a_{i,j}$，$1$ 表示 $i\;j$ 有边

$$f_{x,i}=a_{1,x} \times f_{1,i-1}\;xor ... xor\;a_{n,x} \times f_{n,i-1}$$

长得跟矩阵乘法的形式有点像，又不是高斯约旦

容易证明有乘法结合律（在矩阵为 $01$ 矩阵时成立），然后矩阵加速就搞定了

证明：[link](https://www.luogu.com.cn/paste/t0oqqi6f "题目")

$O(qn^3\log a_i)$

考虑优化

多测我们就把这个加速矩阵的 $2^i$ 预处理出来

只关心起点的值，我们可以只做 $[1 \times n][n \times n]$ 次矩阵乘法

$O(qn^2\log a_i+n^3\log a_i)$

```cpp
const int N=110;
int n,m,q,u,v;
int f[N];
struct node{
	int dot[N][N];
}base;
node p[34];

node mul(node &x,node &y,int n,int p,int m)
{
	for(int i=1;i<=n;i++)
		for(int j=1;j<=m;j++)
		{
			base.dot[i][j]=0;
			for(int k=1;k<=p;k++) base.dot[i][j]^=(x.dot[i][k]*y.dot[k][j]);
		}
	return base;
}

void solve()
{
	int x=fr();
	node ans;
	memset(ans.dot,0,sizeof ans.dot);
	for(int i=1;i<=n;i++) ans.dot[1][i]=f[i];
	for(int i=0;i<32;i++)
		if((x>>i)&1) ans=mul(ans,p[i],1,n,n);
	fw(ans.dot[1][1]),nl;
}

signed main()
{
	n=fr(),m=fr(),q=fr();
	for(int i=1;i<=n;i++) f[i]=fr();
	for(int i=1;i<=m;i++)
	{
		u=fr(),v=fr();
		p[0].dot[u][v]=p[0].dot[v][u]=1;
	}
	for(int i=1;i<32;i++) p[i]=mul(p[i-1],p[i-1],n,n,n);
	while(q--) solve();

	return 0;
}
```

正解

我们发现 $n \leq 100$ 可以 $floyd$

设 $f_{k,u,v}$ 表示最多用了 $k$ 次魔法，$u \to v$ 的最短距离

我们知道 $floyd$ 求解的就是 $f_{0,u,v}$ 这个矩阵

我们推广到 $k \gt 0$ 的情况

本质上就是要加入一条边，让 $u \to v$ 的距离更短

这个可以类比 https://www.luogu.com.cn/problem/P2886

$$f_{k,u,v}=min_{t=1}^n(f_{k-1,u,t}+f_{1,t,v})$$

初始化 $f_{1,u,v}=min_{(x,y)\in E}(f_{0,u,x}+f_{0,y,v}-w(x,y))$

根据广义矩阵乘法定义，我们知道 $c_{i,j}=min(a_{i,x}+b_{x,j})$ 满足分配律

我们做一遍矩阵乘法即可

$O(n^3\log k)$

注意这里的矩阵初始化，因为运算特殊，我们用一个小技巧 $ans$ 不再是 $I$ 就直接设为 $a$ 然后 $k--$

```cpp
const int N=110,M=2.5e3+10;
int n,m,K,u,v,w,inf=1e18;
int d[N][N];
struct Edge{
	int u,v,w;
}ed[M];
struct node{
	int dot[N][N];
}a,base,ans;

node mul(node &x,node &y)
{
	for(int i=1;i<=n;i++)
		for(int j=1;j<=n;j++)
		{
			base.dot[i][j]=inf;
```

**CF1070L Odd Federalization**

子图 $G'=(V',E')$

生成子图（针对原图）（$Spanning\;Subgraph$） $SS=(V,E')$

导出子图（针对子图） ($Induced\;Subgraph$) $IS=(V',E'')$

导出子图规定规定 $\forall u,v\in V'\;(u,v)\in E$，有 $(u,v)\in E''$

若对于导出子图 $\forall u\in V'\;(u,v)\in E$，有 $v\in E'$ 则称其为闭合子图

----------

之前做过一个 Edge Weight Assignment

这种 $xor$ 奇偶的题目一般就是一个很小的数字，我们盲猜不是 $1$ 就是 $2$，因为这玩意有点像最小割

显然每个点度数都是偶数，那么答案为 $1$

否则，我们设两个集合分别为 $[S,T]$，归属设为 $0$ 和 $1$，$bel_i$

我们考虑任意点 $x$，分类讨论一下，和 $x$ 相连的点且在一个集合的数量，必须是偶数

如果 $d_i$ 为偶，$xor_{(u,v)}\;bel_v=0$

如果 $d_i$ 为奇，$bel_u\;xor\;(xor_{(u,v)}\;bel_v)=1$

我们发现是高斯约旦，矩阵 $a$ 就是邻接矩阵，直接做复杂度是 $O(n^3)=8e9$，我们采用常用的 $bitset$ 优化，降到 $O(\frac{n^3}{w})=2.5e8$

注意，当为偶点时，我们不把 $a_{i,i}$ 设为 $1$，而奇点需要，这是因为没有任何一个点与他连通的情况也要算进去，同时也是满足上述式子

接下来我们证明 $ans \leq 2$

等价于方程无解，就是出现了左边 $xor$ 和为 $0$，而右边为 $1$ 的情况，且该点为奇点

而左式为 $0$ 意味着所有点在这些异或方程中都出现了偶数次，而度数为奇数的点会在自己的方程中出现一次，所以在这个导出子图中，度数为奇数的点连接了奇数个点，度数为偶数的点连接了偶数个点，这意味着这个导出子图的度数和为奇数。

对于一个无向图度数无论如何都是偶数，所以不存在无解情况，矛盾

注：在有孤立点的情况下，会出现该列全部为 $0$ 的情况，由于不存在无解情况，所以我们可以把高斯约旦 $continue$ 删掉

## 树上高斯消元

**CF1823F Random Walk**

树上随机游走

看到树上随机游走题，直接考虑高斯消元

设 $f_i$ 表示 $i$ 点的期望经过次数，$S_i$ 表示和 $i$ 点相邻的点的集合， $d_i$ 表示 $i$ 点的度数
$$
\begin{gathered}
f_i=\sum_{j \in S_i \wedge j \neq t} \frac{f_j}{d_j} \quad(i \neq s \wedge i \neq t) \\
f_i=1+\sum_{j \in S_i \wedge j \neq t} \frac{f_j}{d_j} \quad(i=s) \\
f_i=1 \quad(i=t)
\end{gathered}
$$


1 式就是比如 $j$ 点的期望经过次数为 $f_j$ ，那么有期望 $\frac{f_j}{d_j}$ 次是经过 $j$ 后再经过 $i$ 的， $j \neq t$ 是因为经过 $t$ 后就结束了

2 式和 1 式很像，就是起点刚开始有一次，所以 $+1$ 

3 式是到了 $t$ 立刻结束

这显然是 $O(n^3)$ 的高斯消元

----------

对于树上问题，我们可以用特殊的方式求解高斯消元，降到 $O(n)$

有一棵树，每一个点所代表的方程式只包含其父亲，自己，儿子

特别地，根节点的式子只和其自己，以及其儿子有关，叶子节点的式子只和其父亲有关

我们将 $f_u$ 表示成 $A_u \times f_{F} + B_u$ 的形式

----------

假设我们已经知道：

$$f_u = k \cdot f_{F} + \sum_{v \in T_u} k_{u, v} \cdot f_v$$

（其中 $T_u$ 表示 $u$ 的儿子集合）

并且已经知道 $f_v = A_v \times f_u + B_v$

那么：

$$f_u = k \cdot f_F + \sum_{v \in T_u} k_{u, v} \cdot (A_v \cdot f_u + B_v)$$

即：

$$f_u = k \cdot f_F + K' \cdot f_u + B'$$

其中 $K', B'$ 分别是什么只需要将上述式子展开即可

然后 $f_u \cdot(1 - K') = k \cdot f_F + B'$

所以 $f_u = \dfrac{k}{1 - K'} \cdot f_F + \dfrac{B'}{1 - K'}$

叶子节点没有儿子，我们从它开始递推算每个点的 $A_x\;B_x$

然后根节点没有父亲，$f_{rt}=B_{rt}$

再递推一次，从根节点开始算 $f_x$

代码实现的时候把 $k_{u,v}$ 绑定在 $vector$ 的第二个元素中

```cpp
const int N=2e5+10,Q=998244353;
int n,s,t,u,v;
int kf[N],A[N],B[N];
int f[N],inv[N];
vector<pi> as[N];

int qkw(int a,int k)
{
	int ans=1,base=a;
	while(k)
	{
		if(k&1) ans=ans*base%Q;
		base=base*base%Q;
		k>>=1;
	}
	return ans;
}

void prework(int x,int rt)
{
	go(it)
	{
		int v=it.first,&w=it.second;
		if(v==rt) continue;
		prework(v,x);
		if(v!=t && x!=t) w=qkw(as[v].size(),Q-2); //儿子到自己
	}
	if(rt && rt!=t) kf[x]=qkw(as[rt].size(),Q-2); //父亲到自己
	//特殊处理 s,t
	if(x==s) B[x]=1;
	else if(x==t) B[x]=1,kf[x]=0; //f[t]==1	
}

void dfs(int x,int rt)
{
	int a=0,b=B[x]; //K' B'
	go(it)
	{
		int v=it.first,w=it.second;
		if(v==rt) continue;
		dfs(v,x);
		a=(a+A[v]*w%Q)%Q,b=(b+B[v]*w%Q)%Q;
	}
	int div=qkw(1-a+Q,Q-2); //1/1-K'
	A[x]=kf[x]*div%Q,B[x]=b*div%Q;
}

void calc(int x,int rt)
{
	f[x]=(A[x]*f[rt]+B[x])%Q;
	go(it) if(it.first!=rt) calc(it.first,x);
}

signed main()
{
	n=fr(),s=fr(),t=fr();
	for(int i=1;i<n;i++)
	{
		u=fr(),v=fr();
		as[u].pb({v,0}),as[v].pb({u,0});
	}
	
	prework(1,0);
	dfs(1,0);
	calc(1,0);
	for(int i=1;i<=n;i++) fw(f[i]),pt;
	return 0;
}
```
