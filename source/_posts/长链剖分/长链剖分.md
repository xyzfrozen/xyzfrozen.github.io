# 长链剖分

## Intro

和重链剖分差不多

定义

长儿子为从该节点开始有最长向下路径的节点

从这个结点到长儿子的边为重边，若干条首尾衔接的重边构成长链

链顶为一条链顶为这条链深度最小的节点

链底为一条链顶为这条链深度最小的节点



一个节点到根节点的至多经过 $O(\sqrt n)$ 条链



## Problem

### 求 $k$ 级祖先

[P5903 【模板】树上 K 级祖先](https://www.luogu.com.cn/problem/P5903)

任意一个点的 $k$ 级祖先所在链的链长一定 $\geq k$

首先预处理出每个节点在哪条链上，并在链头记录向上向下分别记录链长个节点

再预处理出倍增的 $fa$ 数组，预处理总复杂度 $O(n\log n)$，空间复杂度 $O(n)$

每次查询时，设 $w=\lfloor \log_{2}^k \rfloor$，我们先跳到 $2^w$ 节点，还需要跳 $k-2^w$ 个节点，此时该点所在链长一定 $\geq 2^{w} \gt k-2^w$​ 我们可以直接查询链记录的节点表找到节点

```cpp
int query(int x,int k)
{
    if(!k) return x;
    x=fa[lg[k]][x],k-=1<<lg[k]; //k'
    k-=d[x]-d[top[x]],x=top[x]; //go to chain top
    return k>=0?up[x][k]:dn[x][-k];
}
```

```cpp
#include<bits/stdc++.h>
#define ll long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define ui unsigned int
using namespace std;

const int N=5e5+10;
int n,m,rt,lt;
ui s;ll ans;
int fa[21][N],d[N],mxd[N],son[N],top[N],lg[N];
vector<int> up[N],dn[N],as[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
inline ui get(ui x) {return x^=x<<13,x^=x>>17,x^=x<<5,s=x;}

void prw(int x)
{
    d[x]=mxd[x]=d[fa[0][x]]+1;
    for(int k=1;(1<<k)<=d[x];k++)
        fa[k][x]=fa[k-1][fa[k-1][x]];
    go(v)
    {
        prw(v);
        if(mxd[v]>mxd[x]) mxd[x]=mxd[son[x]=v];
    }    
}

void dfs(int x,int Top)
{
    top[x]=Top;
    if(x==Top)
    {
        for(int i=0,now=x;i<=mxd[x]-d[x];i++)
            up[x].pb(now),now=fa[0][now];
        for(int i=0,now=x;i<=mxd[x]-d[x];i++)
            dn[x].pb(now),now=son[now];
    }
    if(son[x]) dfs(son[x],Top);
    go(v) if(v^son[x]) dfs(v,v);
}

int query(int x,int k)
{
    if(!k) return x;
    x=fa[lg[k]][x],k-=1<<lg[k]; //k'
    k-=d[x]-d[top[x]],x=top[x]; //go to chain top
    return k>=0?up[x][k]:dn[x][-k];
}

int main()
{
    cin>>n>>m>>s;
    lg[0]=-1;
    for(int i=1;i<=n;i++)
        lg[i]=lg[i>>1]+1,as[fa[0][i]=fr()].pb(i);
    rt=as[0][0];
    prw(rt);
    dfs(rt,rt);

    for(int i=1,x,k;i<=m;i++)
    {
        x=((get(s)^lt)%n)+1;
        k=(get(s)^lt)%d[x];
        lt=query(x,k);
        ans^=1ll*lt*i;
    }
    printf("%lld",ans);
    return 0;
}
```



应用题

[U93147 旅行诗（lyric）](https://www.luogu.com.cn/problem/U93147)

$EI$ 的题

设 $f_x$ 表示 $x$ 通过 $x$ 和其子树逃脱的最小代价

那么我们查询的就是所有 $x$ 的深度 $\geq k$ 父亲（根节点深度 $0$），$f_{fa}+d(fa,x)$，可以树上前缀和+倍增最小值做

$O(n\log n+q\log n)$ 85 pts

现在把询问做到 $O(1)$，首先把这个 $d$ 改为点到根节点的差值，就变成一个 $RMQ$ 问题，我们只需要找到 $k$ 级祖先就可以 $O(1)$​​ 查询

```cpp
int anc(int x,int k)
{
    if(!k) return x;
    x=fa[lg[k]][x],k-=1<<lg[k];
    k-=d[x]-d[top[x]],x=top[x];
    return k>=0?up[x][k]:dn[x][-k];
}
int query(int x,int k)
{
    if(!k) return f[x];
    return min(mn[lg[k]][x],mn[lg[k]][anc(x,k-(1<<lg[k]))])+len[x];
}
```

```cpp
#define il inline

#define nl puts("")
#define pi pair<int,int>
#define pb push_back

using namespace std;

UI SA,SB,SC;
UI rng61() 
{
	SA^=SA<<16,SA^=SA>>5,SA^=SA<<1;
	UI t=SA;
	return SA=SB,SB=SC,SC^=t^SA,SC;
}

#define int long long
const int N=3e5+10;
int n,m,u,v,w,t1,t2,lt,ans;
int f[N],mn[21][N],len[N];
int d[N],mxd[N],fa[21][N],lg[N],son[N],top[N];
struct node{int v,w;};
vector<node> as[N];
vector<int> up[N],dn[N];

il int fr(){

    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
il void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
il int max(int a,int b){return a>b?a:b;}
il int min(int a,int b){return a<b?a:b;}

void dfs(int x)
{
	for(auto &[v,w]:as[x])
	{
        len[v]+=len[x];
		dfs(v);
		f[x]=min(f[x],f[v]+w);
	}
}

void prw(int x,int rt)
{
	mxd[x]=d[x]=d[fa[0][x]=rt]+1;
	mn[0][x]=min(f[x]-len[x],f[rt]-len[rt]);
	for(int k=1;(1<<k)<=d[x];k++)
	{
		fa[k][x]=fa[k-1][fa[k-1][x]];
		mn[k][x]=min(mn[k-1][x],mn[k-1][fa[k-1][x]]);
	}
	for(auto &[v,w]:as[x])
    {
        prw(v,x);
        if(mxd[v]>mxd[x]) mxd[x]=mxd[son[x]=v];
    }
}

void cut(int x,int Top)
{
    top[x]=Top;
    if(x==Top)
    {
        for(int i=0,now=x;i<=mxd[x]-d[x];i++)
            up[x].pb(now),now=fa[0][now];
        for(int i=0,now=x;i<=mxd[x]-d[x];i++)
            dn[x].pb(now),now=son[now];
    }

    if(son[x]) cut(son[x],Top);
    for(auto &[v,w]:as[x]) if(v^son[x]) cut(v,v);
}

int anc(int x,int k)
{
    if(!k) return x;
    x=fa[lg[k]][x],k-=1<<lg[k];
    k-=d[x]-d[top[x]],x=top[x];
    return k>=0?up[x][k]:dn[x][-k];
}
int query(int x,int k)
{
    if(!k) return f[x];
    return min(mn[lg[k]][x],mn[lg[k]][anc(x,k-(1<<lg[k]))])+len[x];
}

signed main()
{
	cin>>n>>m>>SA>>SB>>SC>>t1>>t2;
	lg[0]=d[0]=-1,f[0]=1e18;
	for(int i=1;i<=n;i++)
		f[i]=fr(),lg[i]=lg[i>>1]+1;
	for(int i=2;i<=n;i++)
	{
		u=fr(),v=i,len[i]=fr(),w=fr();
		as[u].pb({v,w});
	}
	
	dfs(1),prw(1,0),cut(1,1);
	while(m--) 
	{
		int x=(rng61()^(t1*lt))%n+1;
		int k=(rng61()^(t1*lt))%(d[x]+1)*t2;
		lt=query(x,d[x]-k);
		ans^=lt+m;
	}
	fw(ans);
	
	return 0;
}
```

### 优化 $Dp$

我们发现长链空间复杂度是 $O(n)$ 的，常常用于一些优化以深度为下标的树形 $Dp$

[CF1009F Dominant Indices](https://www.luogu.com.cn/problem/CF1009F)

只要能维护出 $d$，再每次更新时修改最大值是容易的，我们发现距离和下标绑定，考虑用长链剖分优化 $Dp$

$f_{x,j}\leftarrow f_{v,j-1}$

和启发式合并的顺序相反，我们先合并长儿子

我们发现长儿子和 $x$ 就是往后挪了一维，我们考虑用指针表示每个节点的起始地址，这样就可以 $O(1)$ 移动了

```cpp
int d[N],mxd[N],son[N],vlt[N],ans[N];
int *f[N],*cur=vlt;
```

```cpp
f[x][0]=1; //初始化
if(son[x])
{
    f[son[x]]=f[x]+1; //f是指针
    dfs(son[x],x);
    ans[x]=ans[son[x]]+1;
}
```



暴力合并轻儿子，每条链只会在链头被合并，时空复杂度均为 $O(n)$

```cpp
go(v)
{
    if(v==rt || v==son[x]) continue;
    f[v]=cur,cur+=mxd[v];
    dfs(v,x);

    for(int j=1;j<=mxd[v];j++)
    {
        f[x][j]+=f[v][j-1];
        if(f[x][j]+(j<ans[x])>f[x][ans[x]]) ans[x]=j;
    }
}
if(f[x][ans[x]]==1) ans[x]=0;
```



[P5904 [POI2014\] HOT-Hotels 加强版](https://www.luogu.com.cn/problem/P5904)

发现合法点对一定是当前点挂三个子树内的点和两个子树内的点挂一个子树外的点

但是子树外的点难以统计，我们想到不妨在哪个子树外的点和子树内的点的 $lca$ 处统计，这样就可以变为子树内部统计

三个点实在难以算，我们拆成两个点+一个点，利用子树的转移是按顺序的来进行计数 $dp$

钦定 $x$ 为点对 $(p,q,k)$ 的 $lca$，令

单点 $f_{x,j} = [d(x,y)=j][y\in T_x]$

双点 $g_{x,j} = [d(x,p)=d(y,p)=d(x,p)+j)][p=lca(x,y)]\,\cup\,[d(x,y_1)=d(x,y_2)=j][y_1,y_2 \in T_x]$



$f$ 转移显然，$g$ 转移分 $(x,y)$ 在同一子树内部和不同子树内

同一子树内 $g_{v,j+1}$，不同子树内此时原来子树并内挑一个点，当前子树挑一个点 $f_{x,j} \times f_{v,j-1}$
$$
g_{x,j} \leftarrow g_{v,j+1}+f_{x,j} \times f_{v,j-1}\\
f_{x,j} \leftarrow f_{v,j-1}\\
$$


算答案，考察三个点的分布

$g_{x,0}$ 这一定是当前点是最高点，同一子树内三个点的情况，不可能是分居三颗子树的情况，因为当 $j=0$ 时，不存在 $f_{v,j-1}$。

$g_{x,j} \times f_{v,j-1} + f_{x,j} \times g_{v,j+1}$​

注意指针移动每次 `now+=mxd[v],g[v]=now,now+=mxd[v];` 因为前后都要扩展

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e6+10;
int n,u,v,ans;
int d[N],mxd[N],son[N],vlt[N],evg[N];
int *f[N],*cur=vlt,*g[N],*now=evg;
vector<int> as[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void prw(int x,int rt)
{
    d[x]=mxd[x]=d[rt]+1;
    go(v)
    {
        if(v==rt) continue;
        prw(v,x);
        if(mxd[x]<mxd[v]) mxd[x]=mxd[son[x]=v];
    }
}

void dfs(int x,int rt)
{
    if(son[x])
    {
        f[son[x]]=f[x]+1;
        g[son[x]]=g[x]-1;
        dfs(son[x],x);
    }
    f[x][0]=1,ans+=g[x][0];

    go(v)
    {
        if(v==rt || v==son[x]) continue;
        f[v]=cur,cur+=mxd[v];
        now+=mxd[v],g[v]=now,now+=mxd[v];
        dfs(v,x);
        for(int j=1;j<=mxd[v];j++)
            ans+=f[x][j-1]*g[v][j]+g[x][j]*f[v][j-1];
        for(int j=1;j<=mxd[v];j++)
        {
            g[x][j-1]+=g[v][j];
            g[x][j]+=f[x][j]*f[v][j-1];
            f[x][j]+=f[v][j-1];
        }
    }
}

signed main()
{
    n=fr();
    for(int i=1;i<n;i++)
    {
        u=fr(),v=fr();
        as[u].pb(v),as[v].pb(u);
    }

    prw(1,0);
    for(int i=1;i<=n;i++) mxd[i]-=d[i]-1;
    f[1]=cur,cur+=mxd[1],now+=mxd[1],g[1]=now,now+=n;
    dfs(1,0);
    fw(ans);

    return 0;
}
```
