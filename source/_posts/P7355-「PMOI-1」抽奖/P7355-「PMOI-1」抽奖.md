考虑枚举用了 $k$ 张卷

相当于一共 $k$ 位，每一位有 $n$ 种取值，方案数为 $n^k$，接下来考虑换道具

每一种体验卷出现的概率是 $1-(\frac {n-1}{n})^k$ ，出现的体验卷种类数量就是 $n\times (1-(\frac{n-1}{n})^k$ ，换其中一种或者不换
$$
\begin{aligned}
ans
&=n^k \times ((n\times (1-(\frac{n-1}{n})^k))+1)\\
&=n^k \times (n-n\cdot \frac{(n-1)^k}{n^k}+1)\\
&=n^{k+1}+n^k-n\times (n-1)^k
\end{aligned}
$$


即
$$
ans=\sum_{k=1}^m (n^{k+1}+n^k-n\times (n-1)^k)+1
$$
三个等比数列求和即可

注：等比数列求和公式 
$$
S=
\begin{cases}
&\frac{a_1(1-q^n)}{1-q} &[q\not=1]\\
&na_1 &[q=1]
\end{cases}
$$

#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int Q=1e9+7;
int n,m;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int qkw(int a,int k)
{
	int ans=1,base=a;
	while(k)
	{
		if(k&1) ans=ans*base%Q;
		base=base*base%Q;
		k>>=1;
	}
	return ans;
}

int calc(int a1,int q,int n)
{
	if(!q) return a1;
	if(q==1) return n*a1%Q;
	return a1*((1ll-qkw(q,n)+Q)%Q)%Q*qkw(1ll-q+Q,Q-2)%Q;
}

void solve()
{
	n=fr(),m=fr();
	int ans=0;
	ans=calc(n*n%Q,n,m)+calc(n,n,m)-n*calc(n-1,n-1,m)%Q+1ll;
	ans=(ans%Q+Q)%Q;
	fw(ans),nl;
}

signed main()
{
	int T=fr();
	while(T--) solve();

	return 0;
}
