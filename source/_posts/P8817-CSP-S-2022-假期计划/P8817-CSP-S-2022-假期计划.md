首先把 $k \operatorname{++}$， 转换成两点距离

考虑到枚举四个点不行，但是 $O(n^2)$ 可以，我们考虑枚举两个点，然后 $O(1)$ 找出剩下两个点

首先 $O(n^2)$ 预处理每个点 $d \leq k$ 之内的点，并求出这些点中到 $1$ 距离 $\leq k$ 的权值最大的点 $f_i$ ，注意不要选到自己了

我们考虑答案为 $w_{f_i}+w_i+w_j+w_{f_j}$

但是则海洋 $f_i$ 可能与 $f_j \operatorname{or} j$ 重复，这样是不合法的，所以我们每个点要预处理前三大的点权点，因为至多为第一大第二大都被 $f_j \operatorname{or} j$ 占掉

$O(k^2n^2)$，$k=3$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=2.5e3+10;
int n,m,k,u,v,ans;
int w[N],d[N][N],f[N][3];
vector<int> as[N];
queue<int> q;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void bfs(int s)
{
	q.push(s);
	memset(d[s],0x3f,sizeof d[s]);
	d[s][s]=0;
	
	while(q.size())
	{
		auto x=q.front();
		q.pop();
		
		go(v)
		{
			if(d[s][v]>d[s][x]+1)
				d[s][v]=d[s][x]+1,q.push(v);
		}
	}
}

signed main()
{
	n=fr(),m=fr(),k=fr()+1;
	for(int i=2;i<=n;i++) w[i]=fr();
	for(int i=1;i<=m;i++)
	{
		u=fr(),v=fr();
		as[u].pb(v),as[v].pb(u);
	}
	
	for(int i=1;i<=n;i++) bfs(i);
	for(int i=2;i<=n;i++)
		for(int j=2;j<=n;j++)
		{
			if(d[i][j]>k || d[j][1]>k || i==j) continue;
			int v=w[j];
			if(v>w[f[i][0]]) f[i][2]=f[i][1],f[i][1]=f[i][0],f[i][0]=j;
			else if(v==w[f[i][0]] || v>w[f[i][1]]) f[i][2]=f[i][1],f[i][1]=j;
			else if(v==w[f[i][1]] || v>w[f[i][2]]) f[i][2]=j;
		}
	for(int i=2;i<=n;i++)
		for(int j=2;j<=n;j++)
		{
			if(d[i][j]>k || i==j) continue;
			for(int _p=0;_p<=2;_p++)
				for(int _l=0;_l<=2;_l++)
				{
					int p=f[i][_p],l=f[j][_l];
					if(p==j || p==l || l==i || !p || !l) continue;
					ans=max(ans,w[i]+w[j]+w[p]+w[l]);
				}
		}
	fw(ans);
	return 0;
}
```

------

当然可以随机化

还是预处理可达点，然后设 $f_{i,j}$ 表示到了点 $i$，当前选了 $j$ 个点的最大权值和

 $f_{x,j} \gets f_{v,j-1}+w_x [p_i=x]$

考虑正确性，设答案为 $(a,b,c,d)$

每次不出现 $p_a \gt p_b \gt p_c \gt p_d$ 和 $p_a \lt p_b \lt p_c \lt p_d$ 的概率为 $\frac {2}{4!}=\frac{11}{12}$

则正确率为 $1-(\frac{11}{12})^T$ ，$T$ 为枚举次数
