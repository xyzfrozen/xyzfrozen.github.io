莫队基本思想：

根号级别移动完成区间统计，通常用于去掉区间一维限制，难度在于如何统计信息以及 $O(1)$ 修改，查询可到 $O(\sqrt n)$ 级别，可以从这一方面根号平衡来优化

# 带修莫队

[P1903 [国家集训队\] 数颜色 / 维护队列](https://www.luogu.com.cn/problem/P1903)

修改 $\Leftrightarrow$ 加入时间这一维度，每个时间都有一个 $[1,n]$ 的数组，考察三维贡献

这里提一嘴高维莫队的基本方式，对前 $k-1$ 维进行分块，最后一维暴力统计，复杂度 $O(n^{2-\frac{1}{k}})$ 

我们先讨论基本的三维莫队形式，$O(n^{\frac {2}{3}})$

对前 $k-1$ 维分块是因为最后一维跑满了定义域，要减少移动的次数

比如二维莫队中，如果不对 $l$ 分块，$r$ 每次跑满定义域复杂度爆了

具体来说，对 $l,r$ 排序，双块内按时间排序

 设块长为 $a$，考察三个指针的移动次数，注意莫队的关键在于修改 $O(1)$

$l:O(am+2a\times \frac {n}{a})$ 块内移动 $O(m)$，块间移动 $O(2a)$，共 $O(\frac {n}{a})$ 块

$r:O(am + \frac {n}{a} \times n)$，每个 $l$ 块都可以块间移动总计 $O(n)$

$t：O(\frac {n}{a} \times \frac {n}{a} \times t)$ 每次到一对新 $(l,r)$ 块时，移动 $O(t)$

同阶，$len=n^{\frac {2}{3}}$，$O(n^{\frac {5}{3}})$

注意这个修改直接交换就行了，相当于回溯和递归

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=2e5+10,M=1e6+10;
int n,m,t,res,qc;
int a[N],cnt[M],ans[N],pos[N],col[N],c[N];
struct node{int l,r,t,id;}q[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}

    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void add(int x)
{
    if(!cnt[x]) res++;
    cnt[x]++;
}
void del(int x)
{
    if(cnt[x]==1) res--;
    cnt[x]--;
}
void mdf(int l,int r,int k)
{
    if(pos[k]>=l && pos[k]<=r)
        del(a[pos[k]]),add(col[k]);
    swap(a[pos[k]],col[k]);
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++) a[i]=fr();
    while(m--)
    {
        char op[2];
        scanf("%s",op);
        if(op[0]=='Q')
            qc++,q[qc]={fr(),fr(),t,qc};
        else t++,pos[t]=fr(),col[t]=fr();
    }

    int len=pow(n,2.0/3);
    for(int i=1;i<=n;i++) c[i]=(i-1)/len+1;
    auto tmp=[&](const node &a,const node &b){return !(c[a.l]^c[b.l])?(!(c[a.r]^c[b.r])?a.t<b.t:a.r<b.r):a.l<b.l;};
    sort(q+1,q+1+qc,tmp);
    int i=0,j=1,tim=0;
    for(int k=1;k<=qc;k++)
    {
        int l=q[k].l,r=q[k].r,t=q[k].t;
        while(i<r) add(a[++i]);
        while(j>l) add(a[--j]);
        while(i>r) del(a[i--]);
        while(j<l) del(a[j++]);
        while(tim<t) mdf(l,r,++tim);
        while(tim>t) mdf(l,r,tim--);
        ans[q[k].id]=res;  
    }
    
    for(int i=1;i<=qc;i++) fw(ans[i]),nl;
    return 0;
}
```



# 回滚莫队

[歴史の研究](https://www.luogu.com.cn/problem/AT_joisc2014_c)

当莫队的删除难以维护的时候，我们可以考察不删除，依赖指针的单调性解决问题

具体来说，分块后，我们不奇偶排序，这样让 $r$ 单增

对于 $[l,r]$ 同块的询问，暴力做 $O(\sqrt n)$

对于不同块的，我们先只计算不在 $l$ 所在块内的贡献，最后 $O(\sqrt n)$ 加入 $l$ 块的贡献

总复杂度 $O((n+m)\sqrt n)$

实现的时候保留一个只算 $r$ 的副本贡献即可，每次回退

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10,M=sqrt(N)+10;
int n,m,v,res,base;
int c[N],st[N],ed[N],a[N],cnt[N],ans[N];
struct node{int l,r,id;}q[N];
unordered_map<int,int> id,cast;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int query(int l,int r)
{
    int res=0;
    for(int i=l;i<=r;i++)
        res=max(res,(++cnt[a[i]])*cast[a[i]]);
    for(int i=l;i<=r;i++) cnt[a[i]]--;
    return res;
}

void add(int x){res=max(res,(++cnt[x])*cast[x]);}
void del(int x){cnt[x]--;}

signed main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++)
    {
        a[i]=fr();
        if(!id.count(a[i])) id[a[i]]=id.size()+1,cast[id[a[i]]]=a[i];
        a[i]=id[a[i]];
    }
    v=id.size();
    int len=sqrt(1.0*n);
    c[n]=(n-1)/len+1;
    for(int i=1;i<=c[n];i++) 
        st[i]=(i-1)*len+1,ed[i]=i*len;
    ed[c[n]]=n;
    for(int i=1;i<=n;i++)
        c[i]=(i-1)/len+1;
    
    for(int i=1;i<=m;i++) q[i]={fr(),fr(),i};
    auto tmp=[&](const node &a,const node &b){return c[a.l]^c[b.l]?a.l<b.l:a.r<b.r;};
    sort(q+1,q+1+m,tmp);
    for(int bl=1,k=1;bl<=c[n];bl++)
    {
        int i=ed[bl],j=ed[bl]+1;
        res=base=0;
        for(;k<=m && c[q[k].l]==bl;k++)
        {
            if(c[q[k].r]==bl) ans[q[k].id]=query(q[k].l,q[k].r);
            else
            {
                int l=q[k].l,r=q[k].r;
                while(i<r) add(a[++i]);
                base=res;
                while(j>l) add(a[--j]);
                ans[q[k].id]=res;
                while(j<=ed[bl]) del(a[j++]);
                res=base;
            }
        }
        while(i>ed[bl]) del(a[i--]);
    }
    
    for(int i=1;i<=m;i++) fw(ans[i]),nl;
    return 0;
}
```

# 树上莫队

欧拉序的妙用，将树上路径对应到序列上

[COT2 - Count on a tree II ](https://www.luogu.com.cn/problem/SP10707)

这里用的欧拉序是进出时加入的欧拉序，分别用 $in,out$ 表示

不妨设 $in_u \lt in_v$

1. $lca(u,v)=u$ ，$[in_u,in_v]$ 之间只出现一次的点
2. $lca(u,v)=p$，$[out_u,in_v]$ 之间只出现一次的点 $\cup$ $p$ 的贡献

出现两次代表又进又出，贡献相消

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10;
int n,m,u,v,stm,tim,res;
int dfn[N],fa[N],d[N],pos[21][N],lg[N];
int in[N],out[N],a[N],c[N],ans[N],used[N],cnt[N],cast[N];
struct node{int l,r,id,p;}q[N];
vector<int> as[N];
unordered_map<int,int> id;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
int get(int x,int y){return d[x]<d[y]?x:y;}

void dfs(int x,int rt)
{
    fa[x]=rt,d[x]=d[rt]+1;
    pos[0][dfn[x]=++stm]=x,cast[in[x]=++tim]=x;
    go(v) if(v^rt) dfs(v,x);
    cast[out[x]=++tim]=x;
}

int lca(int x,int y)
{
    if(x==y) return x;
    if((x=dfn[x])>(y=dfn[y])) swap(x,y);
    int d=lg[y-x++];
    return fa[get(pos[d][x],pos[d][y-(1<<d)+1])];
}

void add(int x){res+=(++cnt[x]==1);}
void del(int x){res-=!(--cnt[x]);}
void mdf(int p){used[p]^=1,used[p]?add(a[p]):del(a[p]);}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++)
    {
        a[i]=fr();
        if(!id.count(a[i])) id[a[i]]=id.size()+1;
        a[i]=id[a[i]];
    }
    for(int i=1;i<n;i++)
    {
        u=fr(),v=fr();
        as[u].pb(v),as[v].pb(u);
    }

    dfs(1,0);
    lg[0]=-1;
    int len=sqrt(n*2.0);
    for(int i=1;i<=(n<<1);i++)
        lg[i]=lg[i>>1]+1,c[i]=(i-1)/len+1;
    for(int k=1;k<=lg[n];k++)
        for(int i=1;i+(1<<k)-1<=n;i++)
            pos[k][i]=get(pos[k-1][i],pos[k-1][i+(1<<k-1)]);
    
    for(int i=1;i<=m;i++)
    {
        u=fr(),v=fr();
        int p=lca(u,v);
        if(in[u]>in[v]) swap(u,v);
        if(p==u) q[i]={in[u],in[v],i};
        else q[i]={out[u],in[v],i,p};
    }

    auto tmp=[&](const node &a,const node &b){return c[a.l]^c[b.l]?a.l<b.l:(c[a.l]&1?a.r<b.r:a.r>b.r);};
    sort(q+1,q+1+m,tmp);
    int i=0,j=1;
    for(int k=1;k<=m;k++)
    {
        int l=q[k].l,r=q[k].r,p=q[k].p;
        while(i<r) mdf(cast[++i]);
        while(j>l) mdf(cast[--j]);
        while(i>r) mdf(cast[i--]);
        while(j<l) mdf(cast[j++]);
        if(p) add(a[p]);
        ans[q[k].id]=res;
        if(p) del(a[p]);
    }

    for(int i=1;i<=m;i++) fw(ans[i]),nl;
    return 0;
}  
```



# 习题

## [P3245 [HNOI2016\] 大数](https://www.luogu.com.cn/problem/P3245)

子串 $\Leftrightarrow$ 子区间 $\Leftrightarrow$ 前缀/后缀作差

这里因为是左侧高维用后缀和表示，等价于求
$$
s_l=s_{r+1}
$$

Prove：
$$
\begin{gathered}
\overline{a_l a_{l+1} \ldots a_r} \equiv 0 \pmod p \\
\overline{a_l a_{l+1} \ldots a_r} \times 10^{n-r} \equiv 0 \pmod p \\
\overline{a_l a_{l+1} \ldots a_n}-\overline{a_{r+1} a_{r+2} \ldots a_n} \equiv 0 \pmod p \\
s_l=s_{r+1}
\end{gathered}
$$
区间相等数的个数，莫队即可

注意特判掉 $p\mid 10$ 的情况，这时候判定为尾数 $p\mid$ 尾数

```cpp

#define int long long

    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}

    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void work()
{
    for(int i=1;i<=n;i++)
        sum[i]=sum[i-1]+!(a[i]%p)*i,
        cnt[i]=cnt[i-1]+!(a[i]%p);
    while(m--)
    {
        l=fr(),r=fr();
        fw(sum[r]-sum[l-1]-(cnt[r]-cnt[l-1])*(l-1)),nl;
    }
    exit(0);
}

signed main()
{
    scanf("%lld%s",&p,str+1);
    n=strlen(str+1),m=fr();
    for(int i=1;i<=n;i++) a[i]=str[i]-'0';
    if(p==2 || p==5) work();

    for(int i=n,k=1;i;i--,(k*=10)%=p) mod(s[i],s[i+1]+a[i]*k%p);

    int len=sqrt(1.0*n);
    for(int i=1;i<=n;i++) c[i]=(i-1)/len+1;
    for(int i=1;i<=m;i++) q[i]={fr(),fr()+1,i};
    
    auto tmp=[&](const node &a,const node &b){return c[a.l]^c[b.l]?a.l<b.l:(c[a.l]&1?a.r>b.r:a.r<b.r);};
    sort(q+1,q+1+m,tmp);
    int i=0,j=1;
    for(int k=1;k<=m;k++)
    {
        int l=q[k].l,r=q[k].r;
        while(i<r) add(s[++i]);
        while(j>l) add(s[--j]);
        while(i>r) del(s[i--]);
        while(j<l) del(s[j++]);
        ans[q[k].id]=res;
    }

    for(int i=1;i<=m;i++) fw(ans[i]),nl;
    return 0;
}
```

## [P4137 Rmq Problem / mex](https://www.luogu.com.cn/problem/P4137)

比较简单的题

可以回滚，这里就直接用分块实现

先莫队去一维，我们利用到莫队 $O(1)$ 修改，$O(\sqrt n)$ 查询的特性

 $mex$ 直接对值域分块，$O(\sqrt n)$ 跳块判断即可

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=2e5+10,M=510;
int n,m;
int a[N];
struct node{int l,r,id;}q[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

struct Block
{
    int len,n;
    int c[N],s[M],st[M],ed[M],cnt[N];
    void build()
    {
        len=sqrt(n);
        c[n]=(n-1)/len+1;
        for(int i=1;i<=c[n];i++)
            st[i]=(i-1)*len+1,ed[i]=i*len;
        ed[c[n]]=n;
        for(int i=1;i<=n;i++) c[i]=(i-1)/len+1;
    }
    void add(int x){s[c[x]]+=!(cnt[x]++);}
    void del(int x){s[c[x]]-=!(--cnt[x]);}
    int query()
    {
        if(!cnt[0]) return 0;
        for(int k=1;k<=c[n];k++)
        {
            if(s[k]==ed[k]-st[k]+1) continue;
            for(int i=st[k];i<=ed[k];i++)
                if(!cnt[i]) return i;
        }
        return n+1;
    }
}Bl;

struct Mo
{
    int len;
    int c[N],ans[N];
    void solve()
    {
        len=sqrt(1.0*n);
        int v=0;
        for(int i=1;i<=n;i++)
            c[i]=(i-1)/len+1,v=max(v,a[i]);
        Bl.n=v,Bl.build();

        auto tmp=[&](const node &a,const node &b){return c[a.l]^c[b.l]?a.l<b.l:(c[a.l]&1?a.r>b.r:a.r<b.r);};
        sort(q+1,q+1+m,tmp);
        int i=0,j=1;
        for(int k=1;k<=m;k++)
        {
            int l=q[k].l,r=q[k].r;
            while(i<r) Bl.add(a[++i]);
            while(j>l) Bl.add(a[--j]);
            while(i>r) Bl.del(a[i--]);
            while(j<l) Bl.del(a[j++]);
            ans[q[k].id]=Bl.query();
        }

        for(int i=1;i<=m;i++) fw(ans[i]),nl;
    }
}Mo;

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++) a[i]=fr();
    for(int i=1;i<=m;i++) q[i]={fr(),fr(),i};
    Mo.solve();
    return 0;
}
```
