[dsu on tree入门 - 自为风月马前卒 - 博客园 (cnblogs.com)](https://www.cnblogs.com/zwfymqz/p/9683124.html)

利用最多 $\log n$ 条链的性质，以及一个点到根的路径上至多有 $O(\log n)$ 条轻边的特性

1. 递归轻儿子，由于轻儿子之间答案互相干扰，每次需要清空全局数据结构
2. 递归重儿子，不清空
3. 加入轻儿子和自己的贡献，并计算 $T_x$ 的答案

每个点被当做重儿子的复杂度均摊 $O(n)$，被当做轻儿子的复杂度均摊 $O(n\log n)$

总复杂度为 $O(n\log n T)$，$O(T)$ 为计算答案的复杂度

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10;
int n,u,v,stm,mxc,now;
int c[N],son[N],sz[N],cnt[N],ans[N];
vector<int> as[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void prework(int x,int rt)
{
    sz[x]=1;
    go(v)
    {
        if(v==rt) continue;
        prework(v,x);
        sz[x]+=sz[v];
        if(sz[v]>sz[son[x]]) son[x]=v;
    }
}

void clear(int x,int rt)
{
    cnt[c[x]]--;
    go(v) if(v!=rt) clear(v,x);
}
void add(int x)
{
    cnt[c[x]]++;
    if(cnt[c[x]]>mxc) mxc=cnt[c[x]],now=c[x];
    else if(cnt[c[x]]==mxc) now+=c[x];
}
void insert(int x,int rt)
{
    add(x);
    go(v) if(v!=rt) insert(v,x);
}

void dfs(int x,int rt)
{
    go(v)
        if(v!=rt && v!=son[x])
            dfs(v,x),clear(v,x),mxc=now=0;
    if(son[x]) dfs(son[x],x);
    go(v)
        if(v!=rt && v!=son[x])
            insert(v,x);
    add(x),ans[x]=now;
}

signed main()
{
    n=fr();
    for(int i=1;i<=n;i++) c[i]=fr();
    for(int i=1;i<n;i++)
    {
        u=fr(),v=fr();
        as[u].pb(v),as[v].pb(u);
    }

    prework(1,-1);
    dfs(1,-1);
    for(int i=1;i<=n;i++)
        fw(ans[i]),pt;
    return 0;
}
```

$O(n\log^2 n)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define IT set<ele>::iterator
using namespace std;

const int N=2e5+10;
int n,m,u,v,w;
int fa[20][N],d[N],lg[N],ans[N],sz[N],son[N];
vector<int> as[N];
struct query{int pos,v;};
vector<query> op[N];
struct node{
    int l,r;
    int v,id;
    node(){}
    node(int l,int r,int v,int id):
        l(l),r(r),v(v),id(id){}
}tr[N*4];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void prework(int x,int rt)
{
    d[x]=d[rt]+1;
    fa[0][x]=rt;
    sz[x]=1;
    for(int k=1;(1<<k)<=d[x];k++)
        fa[k][x]=fa[k-1][fa[k-1][x]];
    go(v)
    {
        if(!(v^rt)) continue;
        prework(v,x);
        sz[x]+=sz[v];
        if(sz[v]>sz[son[x]]) son[x]=v;
    }
}

int lca(int x,int y)
{
    if(d[x]<d[y]) return lca(y,x);
    int dh=d[x]-d[y],li=lg[dh];
    for(int k=li;~k;k--)
        if((dh>>k)&1) x=fa[k][x];
    if(x==y) return x;
    li=lg[d[x]];
    for(int k=li;~k;k--)
        if(fa[k][x]!=fa[k][y])
            x=fa[k][x],y=fa[k][y];
    return fa[0][x];
}

void build(int ql,int qr,int idx)
{
    tr[idx]=node(ql,qr,0,ql);
    if(ql==qr) return;
    int mid=(ql+qr)>>1;
    build(ql,mid,idx<<1);
    build(mid+1,qr,idx<<1|1);
}

void chf(int idx)
{
    node &t=tr[idx],&ls=tr[idx<<1],&rs=tr[idx<<1|1];
    t.v=max(ls.v,rs.v);
    t.id=t.v==ls.v?ls.id:rs.id;
}

void modify(int ql,int qr,int idx,int x)
{
    node &t=tr[idx];
    if(t.l==t.r) return void(t.v+=x);
    int mid=(t.l+t.r)>>1;
    if(ql<=mid) modify(ql,qr,idx<<1,x);
    if(qr>mid) modify(ql,qr,idx<<1|1,x);
    chf(idx);
}

void clear(int x,int rt)
{
    for(auto &it:op[x])
        modify(it.pos,it.pos,1,-it.v);
    go(v) if(v^rt) clear(v,x);
}
void add(int x)
{
    for(auto &it:op[x])
        modify(it.pos,it.pos,1,it.v);
}
void insert(int x,int rt)
{
    add(x);
    go(v) if(v^rt) insert(v,x);
}

void clear(int idx)
{
    node &t=tr[idx];
    t.v=0,t.id=t.l;
    if(t.l==t.r) return;
    if(tr[idx<<1].v) clear(idx<<1);
    if(tr[idx<<1|1].v) clear(idx<<1|1);
}

void dfs(int x,int rt)
{
    go(v)
        if(v^rt && v^son[x])
            dfs(v,x),clear(1);
    if(son[x]) dfs(son[x],x);
    go(v)
        if(v^rt && v^son[x])
            insert(v,x);
    add(x);
    ans[x]=(!tr[1].v)?0:tr[1].id;
}

int main()
{
    n=fr(),m=fr();
    lg[0]=-1;
    for(int i=1;i<=n;i++) lg[i]=lg[i>>1]+1;
    for(int i=1;i<n;i++)
    {
        u=fr(),v=fr();
        as[u].pb(v),as[v].pb(u);
    }
    prework(1,0);
    while(m--)
    {
        u=fr(),v=fr(),w=fr();
        int p=lca(u,v);
        op[u].pb({w,1}),op[v].pb({w,1});
        op[p].pb({w,-1});
        if(p^1) op[fa[0][p]].pb({w,-1});
    }

    build(1,1e5,1);
    dfs(1,-1);
    for(int i=1;i<=n;i++)
        fw(ans[i]),nl;
    return 0;
}
```
