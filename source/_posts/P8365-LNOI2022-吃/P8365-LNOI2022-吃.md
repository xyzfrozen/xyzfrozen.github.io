我们先贪心全选乘，考虑把乘集合 $P$ 元素加入加集合 $S$

首先 $a_i=1$ 的肯定加入 $S$，此时 $\forall i \in P\,a_i \gt 2$

其次我们至多加入一个元素，否则考虑这两个元素 $(i,j)\,[b_i \leq b_j]$

令 $T=\prod _{p \in P\,\land\,p\not =i,j} a_p$

则 $T \times a_i \times (\Sigma S+b_j) \gt T \times (\Sigma S+b_j+b_i)$

$(i,j)$ 的贡献从 $b_i+b_j \to a_i \times b_j$

考虑加入 $i$ 元素的贡献，$|P| \times |S| \to |P| \times \frac{|S|+b}{a}$

直接取后面那个式子的最大值

注意计算 $|S|$ 的时候不要取模，否则后面那个就错了，用 `long double` 算那个分数

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define lb long double
#define fi first
#define se second
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=5e5+10,Q=1e9+7;
int n,s=1;
vector<pi> w;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
void mod(int &x,int y){if((x+=y)>=Q) x-=Q;}

signed main()
{
	n=fr();
	w.resize(n);
	for(int i=0;i<n;i++) w[i].fi=fr();
	for(int i=0;i<n;i++) w[i].se=fr();
	sort(w.begin(),w.end());
	pi pnt={1,0};
	auto it=upper_bound(w.begin(),w.end(),pnt);
	for(auto i=w.begin();i!=it;i++) mod(s,i->se);
	if(it!=w.begin()) w.erase(w.begin(),it);
	
	lb v=s;
	int now=-1;
	for(int i=0;i<w.size();i++)
	{
		long double t=1.0*(s+w[i].se)/w[i].fi;
		if(v<t) v=t,now=i;
	}
	if(~now) mod(s,w[now].se);
	for(int i=0;i<w.size();i++)
		if(i!=now) (s*=w[i].fi)%=Q;
	fw(s);
	return 0;
}
```
