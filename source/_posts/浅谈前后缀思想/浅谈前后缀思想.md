## T2 P7300 [USACO21JAN] No Time to Paint S

给定序列 $A$ ，$Q$ 次询问 $[L,R]$ ，每个点有一种颜色，任选区间染色，较深颜色覆盖较浅颜色（字典序）

$n,q \leq 1e5$，颜色仅由小写字母构成

对于这种中间空着的，就维护前后缀，$Q$ 很多预处理

显然，深色不影响浅色，浅色影响深色

如果之前有某个颜色 $col$，并且没碰到比它更浅的颜色，现在就不用再染色了

值域又很小，可以搞个桶记下之前是否出现自己

然后每次把比自己深的颜色清空，因为那些深的颜色无法染色到自己

```cpp
const int N=1e5+10;
int n,q,l,r;
int a[N],f[N],g[N];
bool vis[30];
string s;

void prework()
{
	for(int i=1;i<=n;i++)
	{
		f[i]=f[i-1];
		for(int j=a[i]+1;j<=26;j++) vis[j]=0;
		if(!vis[a[i]]) vis[a[i]]=1,f[i]++;
	}
	
	memset(vis,0,sizeof vis);
	for(int i=n;i;i--)
	{
		g[i]=g[i+1];
		for(int j=a[i]+1;j<=26;j++) vis[j]=0;
		if(!vis[a[i]]) vis[a[i]]=1,g[i]++;
	}
}

int main()
{
	n=fr(),q=fr();
	cin>>s;
	for(int i=1;i<=n;i++) a[i]=s[i-1]-'A'+1;
	prework();
	
	while(q--)
	{
		l=fr(),r=fr();
		fw(f[l-1]+g[r+1]),nl;
	}

	return 0;
}
```

## T3 P5304 [GXOI/GZOI2019] 旅行者



$给定有向图\,G=(V,E)，点集\,S，求 \min_{x\in S,y\in S} dis(x,y)\,[x\not =y]$

$|S| \leq n \leq 1e5,m \leq 5e5,w \leq 2e9$

$Sol$

考虑枚举边 $(u,v)$，不妨设 $f_u=x$，满足 $x\in S$，$dis(x,u)$ 最小，$g_v=y$，满足 $y\in S$，$dis(v,y)$ 最小

则最终答案只可能为 $dis(f_u,u)+w+dis(v,g_v)\,[f_u \not =g_v]$

证明

反证法：假设最优路径 $P=[x,y]$ 上不存在边 $(u,v)$ 满足上述条件

那么对于路径上的任意一个节点 $u$，考虑把 $x \to f_u$ 多次迭代可以得到上述方案，对 $y$ 也进行同样的操作

求 $f_u,g_v$ 就建一个正向图和一个反向图，加上一个超级源点跑最短路即可

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10,M=5e5+10;
int n,m,k,u,v,w;
int f[N],g[N],df[N],dg[N],vis[N];
vector<pi> as[N],ed[N];
vector<int> dot;
priority_queue<pi,vector<pi>,greater<pi>> q;
struct node{int u,v,w;}Edge[M];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void build()
{
	memset(f,0,sizeof f);
	memset(g,0,sizeof g);
	
	for(auto v:dot)
		as[n+1].pb({v,0}),f[v]=v;
	q.push({0,n+1});
	memset(df,0x3f,sizeof df);
	memset(vis,0,sizeof vis);
	df[n+1]=0;
	while(q.size())
	{
		auto t=q.top();
		q.pop();
		
		int x=t.second;
		if(vis[x]) continue;
		vis[x]=1;
		
		go(it)
		{
			int v=it.first,w=it.second;
			if(df[v]>df[x]+w)
			{
				if(f[x]) f[v]=f[x];
				df[v]=df[x]+w,q.push({df[v],v});
			}
		}
	}
	
	for(auto v:dot)
		ed[n+1].pb({v,0}),g[v]=v;
	q.push({0,n+1});
	memset(dg,0x3f,sizeof dg);
	memset(vis,0,sizeof vis);
	dg[n+1]=0;
	while(q.size())
	{
		auto t=q.top();
		q.pop();
		
		int x=t.second;
		if(vis[x]) continue;
		vis[x]=1;
		
		for(auto &it:ed[x])
		{
			int v=it.first,w=it.second;
			if(dg[v]>dg[x]+w)
			{
				if(g[x]) g[v]=g[x];
				dg[v]=dg[x]+w,q.push({dg[v],v});
			}
		}
	}
}

void solve()
{
	n=fr(),m=fr(),k=fr();
	for(int i=1;i<=n+1;i++)
		as[i].clear(),ed[i].clear();
	for(int i=1;i<=m;i++)
	{
		u=fr(),v=fr(),w=fr();
		as[u].pb({v,w});
		ed[v].pb({u,w});
		Edge[i]={u,v,w};
	}
	
	dot.clear();
	for(int i=1;i<=k;i++) dot.pb(fr());
	build();
	
	int ans=1e18;
	for(int i=1;i<=m;i++)
	{
		u=Edge[i].u,v=Edge[i].v,w=Edge[i].w;
		if(f[u]^g[v] && f[u] && g[v]) ans=min(ans,df[u]+w+dg[v]);
	}
	fw(ans),nl;
}

signed main()
{
	int T=fr();
	while(T--) solve();

	return 0;
}
```



## T4 CF196E Opening Portals/CF1253F Cheap Robot

优化建图

[Opening Portals](https://www.luogu.com.cn/problem/CF196E)

如果 $n=k$ ，则问题为 $MST$

我们考虑类似虚树的做法，**只对传送门建边**

由上一题的 $(u,v)$ 最短路径原理，设每个点离它最近的点为 $pre_x$，我们可以枚举每一条边，则可以对 $(pre_u,pre_v)$ 建边 $d_u+d_v+w(u,v)$，注意这题是无向图

转化为最小生成树问题，注意我们还需要一开始加上 $d_1$ 到传送门集合的最短距离，而不是直接把 $1$ 当做传送门，因为并不能通过 $1$，直接花费 $0$ 的代价到达任意点

如果 $1$ 不是传送门，我们显然可以直接开一个最小距离传送门，剩下的直接跑 $MST$，反正要联通的

如果 $1$ 是传送门，那么这个距离为 $0$，不影响答案

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10,M=2e5+10;
int n,m,k,u,v,w,res=2e9;
int d[N],vis[N],pre[N],p[N];
vector<pi> as[N];
vector<int> dot;
struct node{
	int u,v,w;
	bool operator<(const node&Q)const{
		return w<Q.w;
	}
}Edge[M];
vector<node> ed;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int find(int x)
{
	if(x!=p[x]) p[x]=find(p[x]);
	return p[x];
}

void dij(int S)
{
	priority_queue<pi,vector<pi>,greater<pi>> q;
	q.push({0,S});
	memset(d,0x3f,sizeof d);
	memset(vis,0,sizeof vis);
	d[S]=0;
	
	while(q.size())
	{
		auto t=q.top();
		q.pop();
		
		int x=t.second;
        if(vis[x]) continue;
        vis[x]=1;
        
        go(it)
        {
            int v=it.first,w=it.second;
            if(d[v]>d[x]+w)
            {
				if(pre[x]) pre[v]=pre[x];
				d[v]=d[x]+w,q.push({d[v],v});
			}
        }
	}
}

void solve()
{
	for(auto v:dot)
		as[n+1].pb({v,0}),pre[v]=v;
	dij(n+1);
	for(int i=1;i<=m;i++)
	{
		u=Edge[i].u,v=Edge[i].v,w=Edge[i].w;
		if(pre[u]==pre[v]) continue;
		ed.pb({pre[u],pre[v],d[u]+d[v]+w});
	}
}

void kru()
{
	sort(ed.begin(),ed.end());
	dot.pb(1);
	for(auto v:dot) p[v]=v;
	int cnt=0,ans=0;
	for(auto it:ed)
	{
		u=it.u,v=it.v,w=it.w;
		int px=find(u),py=find(v);
		if(px^py)
		{
			p[px]=py;
			cnt++;
			ans+=w;
		}
		if(cnt==(int)dot.size()-1) break;
	}
	fw(ans+res);
}

signed main()
{
	n=fr(),m=fr();
	for(int i=1;i<=m;i++)
	{
		u=fr(),v=fr(),w=fr();
		as[u].pb({v,w});
		as[v].pb({u,w});
		Edge[i]={u,v,w};
	}
	
	k=fr();
	for(int i=1;i<=k;i++) dot.pb(fr());
	dij(1);
	for(auto v:dot) res=min(res,d[v]);
	solve();
	kru();
	
	return 0;
}
```

[Cheap Robot](https://www.luogu.com.cn/problem/CF1253F)

这题重要的性质就是询问的点对**一定是充电桩**

我们还是考虑最短路径限制，跑出虚图的 $MST$，我们发现两个充电桩 $(u,v)$ 的最少可达电量一定是 $P(u,v)$ 上面的最大边权，因为每到一个充电桩都可以充满电，只要 $c \geq w$ 就可以不停充电

我们发现这就是一个 $Kru$ 重构树

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=2e5+10,M=3e5+10;
int n,m,k,l,u,v,w,T;
int d[N],vis[N],pre[N],p[N],dep[N],val[N],fa[N][20],lg[N];
vector<pi> as[N];
vector<int> tr[N];
struct node{
	int u,v,w;
	bool operator<(const node&Q)const{
		return w<Q.w;
	}
}Edge[M];
vector<node> ed;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int find(int x)
{
	if(x!=p[x]) p[x]=find(p[x]);
	return p[x];
}

void dij(int S)
{
	priority_queue<pi,vector<pi>,greater<pi>> q;
	q.push({0,S});
	memset(d,0x3f,sizeof d);
	memset(vis,0,sizeof vis);
	d[S]=0;
	
	while(q.size())
	{
		auto t=q.top();
		q.pop();
		
		int x=t.second;
        if(vis[x]) continue;
        vis[x]=1;
        
        go(it)
        {
            int v=it.first,w=it.second;
            if(d[v]>d[x]+w)
            {
				if(pre[x]) pre[v]=pre[x];
				d[v]=d[x]+w,q.push({d[v],v});
			}
        }
	}
}

void solve()
{
	for(int i=1;i<=k;i++)
		as[n+1].pb({i,0}),pre[i]=i;
	dij(n+1);
	for(int i=1;i<=m;i++)
	{
		u=Edge[i].u,v=Edge[i].v,w=Edge[i].w;
		if(pre[u]==pre[v]) continue;
		ed.pb({pre[u],pre[v],d[u]+d[v]+w});
	}
}

void kru()
{
	sort(ed.begin(),ed.end());
	for(int i=1;i<=2*k-1;i++) p[i]=i;
	
	T=k;
	for(auto it:ed)
	{
		u=it.u,v=it.v,w=it.w;
		int x=find(u),y=find(v);
		if(x^y)
		{
			p[x]=p[y]=++T;
			tr[T].pb(x),tr[T].pb(y);
			val[T]=w;
		}
		if(T==2*k-1) break;
	}
}

void dfs(int x,int rt)
{
	dep[x]=dep[rt]+1;
	fa[x][0]=rt;
	for(int k=1;(1<<k)<=dep[x];k++)
		fa[x][k]=fa[fa[x][k-1]][k-1];
	for(auto &v:tr[x]) dfs(v,x);
}

int lca(int x,int y)
{
	if(dep[x]<dep[y]) return lca(y,x);
	int dh=dep[x]-dep[y],li=lg[dh];
	for(int k=li;~k;k--)
		if((dh>>k)&1) x=fa[x][k];
	if(x==y) return val[x];
	li=lg[dep[x]];
	for(int k=li;~k;k--)
		if(fa[x][k]!=fa[y][k])
			x=fa[x][k],y=fa[y][k];
	return val[fa[x][0]];
}

signed main()
{
	n=fr(),m=fr(),k=fr(),l=fr();
	for(int i=1;i<=m;i++)
	{
		u=fr(),v=fr(),w=fr();
		as[u].pb({v,w});
		as[v].pb({u,w});
		Edge[i]={u,v,w};
	}
	
	solve();
	kru();
	dfs(T,0);
	lg[0]=-1;
	for(int i=1;i<=n;i++) lg[i]=lg[i>>1]+1;
	while(l--)
	{
		u=fr(),v=fr();
		fw(lca(u,v)),nl;
	}
	
	return 0;
}
```

## T5 BZOJ 2725 [Violet 6]故乡的梦

![image](https://cdn.jsdelivr.net/gh/xyzfrozen/Cdn/img/202310022310721.png)

给定无向图 $G=(V,E)$ ，$S$，$T$，$Q$ 次询问强制选定边 $(u,v)$ ，$S$ 到 $T$ 的最短路

去掉 $(u,v)$，我们不难想到一个类似处理前缀和后缀的方法

首先无向图转有向图，我们跑出一条 $[S,T]$ 的最短路，定义为路径 $P$

先给 $P$ 上的点编号

考虑答案路径一定是 $[S,a,x,y,b,T]$，即**最短路树**上的边

![image](https://cdn.jsdelivr.net/gh/xyzfrozen/Cdn/img/202310022308026.png)

我们预处理枚举所有边 $(x,y)$，强制让最短路经过它

这条边产生的贡献，即什么时候会被算入最短路

设 $DS_i\;DT_i$ 表示 $S\;T$ 到 $i$ 的最短路

$L_x\;R_x$  表示 $x$ 到 $P$ 上离 $S/T$ 最近的点的编号（从 $[S,T]$ 重新编过号后的），这个可以用 $topsort$ + 最短路树 $O(n\log n)$ 维护 

对于 $(x,y)$，相当于删掉 $[a,b]$ 之间的边它就有可能产生答案，我们把每条边转成点，绑定在它 $id$ 较小的点上

每次枚举边相当于对 $[L_x,R_x-1]$ 的做区间 $Min$，每次询问相当于单点查询，用线段树来做

$O(n\log n+m\log n+q)$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=4e5+10;
int n,m,Q,u,v,w,S,T,len,inf=0x3f3f3f3f3f3f3f3f;
int d[4][N],p[N],pre[N],in[N],q[N];
int P[N],id[N],ans[N],L[N],R[N];
vector<pi> as[N];
vector<int> G[N];
struct Edge{int u,v,w;}E[N];
struct node{int l,r,v;}tr[N*4];

int fr(){
    int x=0,flag=1;char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9') x=x*10+(ch-'0'),ch=getchar();
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
bool ck(int x,int y){return id[x]&&id[y]&&(id[x]==id[y]+1 || id[y]==id[x]+1);}

void dij(int S,int g)
{
	priority_queue<pi,vector<pi>,greater<pi>> q;
	q.push({0,S});
	memset(d[g],0x3f,sizeof d[g]);
	memset(p,0,sizeof p);
	d[g][S]=0;
	
	while(q.size())
	{
		auto t=q.top();
		q.pop();
		
		int x=t.second;
		if(p[x]) continue;
		p[x]=1;
		
		go(it)
		{
			int v=it.first,w=it.second;
			if(d[g][v]>d[g][x]+w) d[g][v]=d[g][x]+w,pre[v]=x,q.push({d[g][v],v});
		}
	}
}

void get_path()
{
	dij(S,1);
	if(d[1][T]==inf)
	{
		Q=fr();
		while(Q--) puts("Infinity");
		exit(0);
	}
	
	for(int x=T;pre[x];x=pre[x]) P[++len]=x;
	P[++len]=S,reverse(P+1,P+1+len);
	for(int i=1;i<=len;i++) id[P[i]]=i;
	
	for(int i=1;i<=n;i++) L[i]=N;
	for(int i=1;i<=len;i++) L[P[i]]=i;
	for(int i=1;i<=m;i++)
	{
		u=E[i].u,v=E[i].v,w=E[i].w;
		if(d[1][u]+w==d[1][v]) in[v]++,G[u].pb(v);
		if(d[1][v]+w==d[1][u]) in[u]++,G[v].pb(u);
	}
	
	int hh=0,tt=-1;
	q[++tt]=S;
	while(hh<=tt)
	{
		int x=q[hh++];
		for(auto &v:G[x])
		{
			if(!id[v]) L[v]=min(L[v],L[x]);
			if(!(--in[v])) q[++tt]=v;
		}
	}
	
	dij(T,2);
	for(int i=1;i<=n;i++) G[i].clear(),in[i]=0;
	for(int i=1;i<=len;i++) R[P[i]]=i;
	for(int i=1;i<=m;i++)
	{
		u=E[i].u,v=E[i].v,w=E[i].w;
		if(d[2][u]+w==d[2][v]) in[v]++,G[u].pb(v);
		if(d[2][v]+w==d[2][u]) in[u]++,G[v].pb(u);
	}
	
	hh=0,tt=-1;
	q[++tt]=T;
	while(hh<=tt)
	{
		int x=q[hh++];
		for(auto &v:G[x])
		{
			if(!id[v]) R[v]=max(R[v],R[x]);
			if(!(--in[v])) q[++tt]=v;
		}
	}
}

void build(int ql,int qr,int idx)
{
	tr[idx]={ql,qr,inf};
	if(ql==qr) return;
	int mid=(ql+qr)>>1;
	build(ql,mid,idx<<1);
	build(mid+1,qr,idx<<1|1);
}

void modify(int ql,int qr,int idx,int x)
{
	node &t=tr[idx];
	if(ql<=t.l && qr>=t.r) {t.v=min(t.v,x);return;}
	int mid=(t.l+t.r)>>1;
	if(ql<=mid) modify(ql,qr,idx<<1,x);
	if(qr>mid) modify(ql,qr,idx<<1|1,x);
}

void chs(int idx)
{
	node &t=tr[idx];
	if(t.l==t.r) {ans[t.l]=t.v;return;}
	node &ls=tr[idx<<1],&rs=tr[idx<<1|1];
	ls.v=min(ls.v,t.v),rs.v=min(rs.v,t.v);
	chs(idx<<1),chs(idx<<1|1);
}

void solve()
{
	build(1,len-1,1);
	for(int i=1;i<=m;i++)
	{
		u=E[i].u,v=E[i].v,w=E[i].w;
		if(ck(u,v)) continue;
		if(L[u]!=N && R[v] && L[u]<R[v]) modify(L[u],R[v]-1,1,d[1][u]+w+d[2][v]);
		if(L[v]!=N && R[u] && L[v]<R[u]) modify(L[v],R[u]-1,1,d[1][v]+w+d[2][u]);
	}
	chs(1);
}

		u=fr(),v=fr(),w=fr();
		as[u].pb({v,w}),as[v].pb({u,w}),E[i]={u,v,w};
```
