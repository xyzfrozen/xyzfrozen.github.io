显然是 $kru$ 的优化建边题

优化的原则就是 $(a,b) \geq (a,c)+(c,b)$ 就没必要建 $(a,b)$

显然要把所有数去个重，重复的边权都是 $0$

我们考虑到这个边和值域有关，可以接受 $O(V\log V)$ 的复杂度，显然想到枚举倍数，因为这样做是 $O(\frac V2+\frac V3 +\dots \frac VV)$ 的

我们考虑到为什么枚举倍数最优，即怎样在枚举倍数后建边不劣

我们设倍数为 $i$，当前数为 $v$，枚举每个 $[iv,(i+1)v)$ 区间内**第一个数建边**

考虑区间内两数 $(p,q)$

首先我们讨论 $(a,b)$ 的大小可以得到 $(a,b)$ 的边等价于 $\max (a,b) \,mod \min(a,b)$

$(v,p)+(p,q)=q-p+p\,mod\,v=q\,mod\,v=(v,q)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10;
int n,W;
int a[N],p[N];
vector<int> vec;
struct node{
	int u,v,w;
	bool operator<(const node&Q)const{
		return w<Q.w;
	}
};
vector<node> Ed;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}
int find(int x)
{
	if(x!=p[x]) p[x]=find(p[x]);
	return p[x];
}

void add(int id)
{
	int v=vec[id];
	for(int i=v;i<=W;i+=v)
	{
		int q=lower_bound(vec.begin(),vec.end(),i)-vec.begin();
		if(q==id) q++;
		if(q!=(int)vec.size() && vec[q]/v==i/v)
			Ed.pb({id+1,q+1,vec[q]%v});
	}
}

void kru()
{
	sort(Ed.begin(),Ed.end());
	for(int i=1;i<=(int)vec.size();i++) p[i]=i;
	int ans=0,cnt=0;
	for(auto it:Ed)
	{
		int u=it.u,v=it.v,w=it.w;
		int px=find(u),py=find(v);
		if(px^py)
		{
			p[px]=py;
			ans+=w,cnt++;
		}
		if(cnt==(int)vec.size()-1) break;
	}
    fw(ans);
}

int main()
{
	n=fr();
	for(int i=1;i<=n;i++)
		vec.emplace_back(a[i]=fr()),W=max(W,a[i]);
	sort(vec.begin(),vec.end());
	vec.erase(unique(vec.begin(),vec.end()),vec.end());
	for(int i=0;i<(int)vec.size();i++) add(i);
	kru();

	return 0;
}
```
