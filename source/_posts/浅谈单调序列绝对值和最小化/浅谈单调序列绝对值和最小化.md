我们考虑尽可能让 $b_i=a_i$，即让 $b_i$ 序列中的每一个数都在 $a_i$ 中出现，这样就相当于把值域压缩到了 $O(n)$

即，在满足答案最小的情况下，一定存在一种方案使得 $\forall\,b_i,\exists\,a_j=b_i$

考虑数学归纳法，$n=1$ 成立

若 $b_{n-1} \leq a_i$ ，则令 $b_i=a_i$ 

否则，必然存在 $j$，$b_j\sim b_n=a_k\,[k\in [j,n]]$ ，由中位数的性质即可推得

具体来说，线性 $dp$ 肯定是要分段划分序列做

连续上升段 $a_i \leq a_{i+1} \leq a_{i+2} \,\ldots$ ，则 $b_i=a_i$

连续下降段 $a_i \geq a_{i+1} \geq a_{i+2} \,\ldots$，则 $b_i=[a_i \sim a_j](\text{中位数})$

[P2893 [USACO08FEB\] Making the Grade G - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/P2893)

考虑求不下降序列的答案，$O(n^2)$ 

设 $f_{i,j}$ 考虑前 $i$ 个数，$b_i=j\,[j=a_p][p\in [1,n]]$ 的最小答案
$$
f_{i,j} =\min_{k=1}^j f_{i-1,k}+|a_i-j|
$$
[P4597 序列 sequence - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/P4597)

考虑优化，假设当前下降了

	     y
	   z
	 -     x

   我们肯定想贪心的让 $y$ 变为 $x$ ，因为不管是升高 $x$，还是降低 $y$，答案的增加量都是一定的，肯定是让 $x$ 不上升才可以留出最多的空间给后面


       z
     -   y x
    -

但是可能下降之后变成这样了，不满足递增了

但是我们可以调整一下，我们 $y$ 不降那么多， $x$ 升一点，都变成非严格次大值即可，这样答案不变，同时最优

```
	z y x
  -
-
```

单调不升就把序列 `reverse` 一下就行了

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

int n,x,ans;
priority_queue<int> q;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

signed main()
{
	n=fr();
	for(int i=1;i<=n;i++)
	{
		q.push(x=fr());
		if(x<q.top())
		{
			ans+=q.top()-x;
			q.pop(),q.push(x);
		}
	}
	fw(ans);
	return 0;
}
```

[Sonya and Problem Wihtout a Legend - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/CF713C) 

严格递增怎么做，一个常用 $trick$

$b_i \lt b_{i+1} \lt b_{i+2} \ldots \Leftrightarrow b_i \leq b_{i+1}-1 \leq b_{i+2}-2$

我们把序列的第 $i$ 数都 $-i$ 即可

[P4331 [BalticOI 2004\] Sequence 数字序列 - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/P4331)

输出方案？

我们记录每个位置 `b_i=q.top()`，然后为了算这种下降的 `b[i]=min(b[i],b[i+1])`

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e6+10;
int n,x,ans;
int b[N];
priority_queue<int> q;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

signed main()
{
	n=fr();
	for(int i=1;i<=n;i++)
	{
		q.push(x=fr()-i);
		if(x<q.top())
		{
			ans+=q.top()-x;
			q.pop(),q.push(x);
		}
		b[i]=q.top();
	}
	fw(ans),nl;
	for(int i=n-1;i;i--) b[i]=min(b[i],b[i+1]);
	for(int i=1;i<=n;i++) fw(b[i]+i),pt;
	return 0;
}
```

对于整个区间按照值域进行二分，在每一层找到那个分界点，由于分界线左侧的那个数取 $mid$  一定比取 $mid+1$ 优秀，由题目要求的非降数列可以推知整体也符合该要求，这样也保证了非降序列

同时根据上方的证明，也能保证出现的每一个数都是原序列中的数

$O(n\log W)$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e6+10;
int n,ans;
int a[N],b[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void solve(int vl,int vr,int ql,int qr)
{
    if(ql>qr || vl>vr) return;
    int mid=(vl+vr)>>1,cnt=0,s=0;
    if(vl==vr)
    {
		for(int i=ql;i<=qr;i++) b[i]=vl;
         return;
	}
	
    //初值设为 mid+1 通过 v 来控制分界点
    for(int i=ql;i<=qr;i++) s+=abs(a[i]-mid);
    int v=s;
    for(int i=ql;i<=qr;i++)
    {
		s-=abs(a[i]-mid-1);
		s+=abs(a[i]-mid); //移动分界点
		if(s<v) v=s,cnt=i-ql+1;
	}
    
    solve(vl,mid,ql,ql+cnt-1);
	solve(mid+1,vr,ql+cnt,qr);
}

signed main()
{
	n=fr();
	for(int i=1;i<=n;i++) a[i]=fr()-i;
	solve(-2e9,2e9,1,n);
	for(int i=1;i<=n;i++) ans+=abs(a[i]-b[i]);
	fw(ans),nl;
	for(int i=1;i<=n;i++) fw(b[i]+i),pt;
	return 0;
}
```
