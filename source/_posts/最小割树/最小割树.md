# 最小割树

## Intro

图上本质不同的最小割只有 $O(n)$ 种，用树边来表达

有时候需要多次询问两点之间的最小割，可以用最小割树解决

暴力算法是 $O(\min(q,n^2)n^2m)=O(n^4m)$，最小割树是 $O(n\times n^2m+n^2+q)=O(n^3m)$ 的

## 算法流程

1. 随便选择当前原图连通块的点跑最小割，在两点的**树图**连边，边权为最小割
2. 将 $[S,T]$ 分开，分别继续递归处理
3. 当点集大小为 $0$ 时停止处理

注：每次求最小割之前需要恢复原图

原图两点之间的最小割=树图两点之间简单路径的最小边权



[模板题](https://www.luogu.com.cn/problem/P4897)

这个题数据有点问题，所有点编号 $+1$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e3+10,M=6e3+10;
int n,m,s=N-1,t=N-2,u,v,w,ans,inf=2e9;
int h[N],e[M],ne[M],c[M],idx;
int q[N],d[N],cur[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void add(int a,int b,int w)
{
	e[idx]=b,ne[idx]=h[a],c[idx]=w,h[a]=idx++;
	e[idx]=a,ne[idx]=h[b],c[idx]=0,h[b]=idx++;
}

bool bfs()
{
	int hh=0,tt=-1;
	memset(d,-1,sizeof d);
	q[++tt]=s,d[s]=0,cur[s]=h[s];
	
	while(hh<=tt)
	{
		int x=q[hh++];
		for(int i=h[x];~i;i=ne[i])
		{
			int j=e[i];
			if(!(~d[j]) && c[i])
			{
				d[j]=d[x]+1;
				cur[j]=h[j];
				if(j==t) return 1;
				q[++tt]=j;
			}
		}
	}
	return 0;
}

int find(int x,int ltf)
{
	if(x==t) return ltf;
	int rest=ltf;
	for(int i=cur[x];~i && rest;i=ne[i])
	{
		cur[x]=i;
		int j=e[i];
		if(d[j]==d[x]+1 && c[i])
		{
			int k=find(j,min(rest,c[i]));
			if(!k) d[j]=-1;
			else c[i]-=k,c[i^1]+=k,rest-=k;
		}
	}
	return ltf-rest;
}

int dinic()
{
	for(int i=0;i<idx;i+=2)
		c[i]+=c[i^1],c[i^1]=0;
	int flow,ans=0;
	while(bfs())
		while(flow=find(s,inf)) ans+=flow;
	return ans;
}

int ary[N],lq[N],rq[N],cut[N][N];
vector<pi> as[N];

void solve(int l,int r)
{
	if(l==r) return;
	s=ary[l],t=ary[r];
	int flow=dinic();
	as[s].pb({t,flow});
	as[t].pb({s,flow});
	int lf=0,rt=0;
	for(int i=l;i<=r;i++)
		~d[ary[i]]?lq[++lf]=ary[i]:rq[++rt]=ary[i];
	for(int i=1;i<=lf;i++) ary[l+i-1]=lq[i];
	for(int i=1;i<=rt;i++) ary[l+lf+i-1]=rq[i];
	solve(l,l+lf-1),solve(l+lf,r);
}

void dfs(int x,int rt,int tgt)
{
	for(auto &[v,w]:as[x])
	{
		if(v==rt) continue;
		cut[tgt][v]=min(cut[tgt][x],w);
		dfs(v,x,tgt);
	}
}

int main()
{
	memset(h,-1,sizeof h);
	n=fr()+1,m=fr();
	for(int i=1;i<=m;i++)
	{
		u=fr()+1,v=fr()+1,w=fr();
		add(u,v,w),add(v,u,w);
	}
	
	iota(ary+1,ary+1+n,1);
	solve(1,n);
	for(int i=1;i<=n;i++) cut[i][i]=inf,dfs(i,0,i);
	
	for(int q=fr();q;q--)
	{
		u=fr()+1,v=fr()+1;
		fw(cut[u][v]),nl;
	}
	return 0;
}
```
