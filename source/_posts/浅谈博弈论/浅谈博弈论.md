博弈论题，把状态压出来，然后做记忆化搜索

[P4363 [九省联考 2018\] 一双木棋 chess - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/P4363)

考虑本题性质，每一行一定连续，当前一定是一个左上三角形的局面

考虑压缩状态，我们只需要知道每一行填到哪里了即可

------

介绍一个奇妙的搜索 $\alpha - \beta$ 剪枝，要求 零和博弈 && 博弈双方足够聪明，两个人目标相反

[【算法】极大极小α-β剪枝算法 - 知乎 (zhihu.com)](https://zhuanlan.zhihu.com/p/65219446)

[理解Alpha-Beta 剪枝算法_阿尔法贝塔剪枝-CSDN博客](https://blog.csdn.net/qq_36612242/article/details/106425436)

[最清晰易懂的MinMax算法和Alpha-Beta剪枝详解-CSDN博客](https://blog.csdn.net/weixin_42165981/article/details/103263211)

$\alpha$ 表示目前所有可能解中的最大下界，$\beta$  表示目前所有可能解中的最小上界

当 $\alpha \geq \beta$ 时进行剪枝

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=12;
int n,m,inf=2e9;
int a[N][N],b[N][N],pos[N],w[N][(1<<10)+10],S[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int dfs(int now,int cur,int alpha,int beta)
{
	if(!now)
	{
		int ans=0;
		for(int i=1;i<=n;i++)
			ans+=w[i][S[i]];
		return ans;
	}
	for(int i=1;i<=n;i++)
	{
		if((pos[i-1]>pos[i] || i==1) && pos[i]!=m)
		{
			pos[i]++;
			if(!cur) S[i]|=(1<<pos[i]-1);
			if(!cur) alpha=max(alpha,dfs(now-1,cur^1,alpha,beta));
			else beta=min(beta,dfs(now-1,cur^1,alpha,beta));
			if(!cur) S[i]^=(1<<pos[i]-1);
			pos[i]--;
			if(alpha>=beta) break;
		}
	}
	return cur?beta:alpha;
}

int main()
{
	n=fr(),m=fr();
	for(int i=1;i<=n;i++)
		for(int j=1;j<=m;j++)
			a[i][j]=fr();
	for(int i=1;i<=n;i++)
		for(int j=1;j<=m;j++)
			b[i][j]=fr();
	for(int i=1;i<=n;i++)
		for(int S=0;S<=(1<<m)-1;S++)
			for(int j=1;j<=m;j++)
				w[i][S]+=((S>>j-1)&1)?(a[i][j]):(-b[i][j]);
	fw(dfs(n*m,0,-inf,inf));
	return 0;
}
```

------

轮廓线 $dp$

状态一定可以用 $m$ 个 $0$ 和 $n$ 个 $1$ 表示出来，因为前一行一定比不短于当前行

比如 $00011$ 就是 $2 \times 3$ 的矩阵的初始状态

$01001$ 就是第一行第二列 $00110$ 就是第一二行第一列

每个 $1$ 后面 $0$ 的个数就是它走到第几列，然后最左到右就是每一行，本质上是 $01 \to 10$

最初状态 `1<<(n-1)`，最终状态 `(1<<(n-1))<<m`

考虑记搜设 $f_S$ 表示当前状态到结束可以获得多少分

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=11;
int n,m,sx,sy,inf=0x3f3f3f3f;
int a[N][N],b[N][N],f[1<<22],vis[1<<22];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int dfs(int S,int cur)
{
	if(vis[S]) return f[S];
	vis[S]=1;
	f[S]=cur?inf:-inf;
	for(int i=0,hor=n,ver=0;i<n+m-1;i++) //枚举 01 的 1 在哪里
	{
		if((S>>i)&1) hor--;
		else ver++;
		if(((S>>i)&3)!=1) continue; //11 10 00
		int T=(S|(1<<i+1))^(1<<i);
		if(!cur) f[S]=max(f[S],dfs(T,cur^1)+a[hor][ver]);//先手
		else f[S]=min(f[S],dfs(T,cur^1)-b[hor][ver]);
	}
	return f[S];
}

int main()
{
	n=fr(),m=fr();
	for(int i=0;i<n;i++)
		for(int j=0;j<m;j++)
			a[i][j]=fr();
	for(int i=0;i<n;i++)
		for(int j=0;j<m;j++)
			b[i][j]=fr();
		
	vis[((1<<n)-1)<<m]=1;
	f[((1<<n)-1)<<m]=0;
	fw(dfs((1<<n)-1,0));
	return 0;
}
```
