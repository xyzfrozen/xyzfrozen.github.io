可持久化并查集=主席树+并查集

主席树是可持久化的基础，利用动态开点和复用降低复杂度

[模板题](https://www.luogu.com.cn/problem/P3402)

1. 合并集合
2. 撤销操作
3. 查询是否在同一集合

我们用主席树维护序列，对于每个叶子节点维护 $fa_x$ 

如果直接用数组维护，我们每次复用就是 $O(n)$ 的，但是主席树就是 $O(\log n)$ 的，主席树维护单点修改即可

因为需要撤销，平常时候使用的并查集优化是路径压缩，但是我们压缩一次相当于就要修改一个节点，而我们知道单点修改是要新建 $O(\log n)$ 个节点的，我们的空间复杂度变成了 $O(\log ^2 n)$

所以我们采用启发式合并即可，保证树高 $O(\log n)$

寻找 $fa$ 直接主席树查询往上跳 $O(\log^2n)$，同时再用一颗维护一下 $sz$ 即可

时间复杂度 $O(m\log^2n)$，空间复杂度 $O(m\log n)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=2e5+10;
int n,m,tim=1;
int c[N];
struct Seg_Union
{
    int idx,rt[N];
    int ls[N<<5],rs[N<<5],v[N<<5];
    Seg_Union(){idx=0;}

    int build(int vl,int vr)
    {
        int now=++idx,mid=(vl+vr)>>1;
        if(vl==vr) return v[now]=c[vl],now;
        ls[now]=build(vl,mid);
        rs[now]=build(mid+1,vr);
        return now;
    }

    int ins(int p,int vl,int vr,int pos,int val)
    {
        int q=++idx;
        ls[q]=ls[p],rs[q]=rs[p];
        if(vl==vr) return v[q]=val,q;
        int mid=(vl+vr)>>1;
        if(pos<=mid) ls[q]=ins(ls[p],vl,mid,pos,val);
        else rs[q]=ins(rs[p],mid+1,vr,pos,val);
        return q;
    }

    int query(int p,int vl,int vr,int pos)
    {
        if(vl==vr) return v[p];
        int mid=(vl+vr)>>1;
        return (pos<=mid)?query(ls[p],vl,mid,pos):query(rs[p],mid+1,vr,pos);
    }
}fa,sz;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int find(int x)
{
    while(1)
    {
        int rt=fa.query(fa.rt[tim],1,n,x);
        if(rt^x) x=rt;
        else break;
    }
    return x;
}
void Union(int x,int y,int val)
{
    fa.rt[tim]=fa.ins(fa.rt[tim],1,n,x,y);
    sz.rt[tim]=sz.ins(sz.rt[tim],1,n,y,val);
}
void merge(int x,int y)
{
    int px=find(x),py=find(y);
    if(!(px^py)) return;
    int sx=sz.query(sz.rt[tim],1,n,px),sy=sz.query(sz.rt[tim],1,n,py);
    (sx<sy)?Union(px,py,sx+sy):Union(py,px,sx+sy);
}

int main()
{
    n=fr(),m=fr();
    for(int i=1;i<=n;i++) c[i]=i;
    fa.rt[0]=fa.build(1,n);
    for(int i=1;i<=n;i++) c[i]=1;
    sz.rt[0]=sz.build(1,n);

    for(int i=1,k;i<=m;i++,tim++)
    {
        int op=fr();
        fa.rt[i]=fa.rt[i-1],sz.rt[i]=sz.rt[i-1];
        if(op==1) merge(fr(),fr());
        else if(op==2) k=fr(),fa.rt[i]=fa.rt[k],sz.rt[i]=sz.rt[k];
        else puts((find(fr())==find(fr()))?"1":"0");
    }
    return 0;
}
```
