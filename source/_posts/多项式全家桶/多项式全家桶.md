# 拉格朗日插值

## Intro

$n+1$ 个点确定一个 $n$ 次多项式 $f(x)$

把第 $i$ 个点映射到 $p_i'=(x_i,0)$ 上，不妨设
$$
f_i(x)=
\begin{cases}
(x_j,y_j) &[i=j]\\
(x_j,0) &[i \neq j] 
\end{cases}
$$
则 $f(x)=\sum_{i=1}^n f_i(x)$，$f_i(x)=a\prod_{j \neq i} (x-x_j)$

带入得 $a=\frac {y_i}{\prod_{j \neq i} (x_i-x_j)}$
$$
f_i(x)=y_i \prod_{j \neq i}\frac {(x-x_j)}{ (x_i-x_j)}
$$

$$
\begin{aligned}
&\begin{aligned}f(x)=\sum_{i=1}^ny_i\frac{\prod_{j!=i}(x-x_j)}{\prod_{j!=i}(x_i-x_j)}\end{aligned} \\
&f(x)=\sum_{i=1}^ny_i\frac{\prod_{j=1}^n(x-x_j)}{\prod_{j!=i}(x_i-x_j)*(x-x_i)}  \\\\
&\text{set}\,\omega(i)=\prod_{j!=i}(x_i-x_j) \\
&f(x)=\prod_{i=1}^n(x-x_i)\sum_{i=1}^n\frac{y_i}{(x-x_i)\omega(i)} 
\end{aligned}
$$

支持 $O(n)$ 添加

[P4781 【模板】拉格朗日插值](https://www.luogu.com.cn/problem/P4781)

```cpp
const int N=2e3+10,Q=998244353;
int n,k;
int x[N],y[N],w[N];

int qkw(int a,int k=Q-2,int ans=1)
{
	for(;k;a=(a*a)%Q,k>>=1) if(k&1) ans=ans*a%Q;
	return ans;
}

void lagrange()
{
	fill(w+1,w+1+n,1);
	for(int i=1;i<=n;i++)
		for(int j=1;j<=n;j++)
			if(i^j) (w[i]*=(x[i]-x[j]+Q)%Q)%=Q;
	int res=1,ans=0;
	for(int i=1;i<=n;i++) (res*=(k-x[i]+Q)%Q)%=Q,w[i]=qkw(w[i]);
	for(int i=1;i<=n;i++)
		mod(ans,res*y[i]%Q*qkw((k-x[i]+Q)%Q)%Q*w[i]%Q);
	fw(ans);
}

signed main()
{
	n=fr(),k=fr();
	for(int i=1;i<=n;i++) x[i]=fr(),y[i]=fr();
	lagrange();
	return 0;
}
```



## Problem

[The Sum of the k-th Powers](https://www.luogu.com.cn/problem/CF622F)

数列 $\{a_n\}$ 的是一个 $p$ 阶等差数列的充要条件是数列的通项 $a_n$ 为 $n$ 的一个 $p$ 次多项式

$\sum_{i=1}^n i^k$ 是一个 $k+1$ 次多项式

取 $1 \sim k+2$ ，带入进行拉插，不难发现分子分子都可以预处理，普通就暴力算就行 $O(k\log k)$，如果用光速幂/线性筛+线性逆元就是 $O(k)$

```cpp
const int N=1e6+10,Q=1e9+7;
int n,k;
int x[N],y[N],w[N],pre[N],suf[N];
void mod(int &x,int y){if((x+=y)>=Q) x-=Q;}

int qkw(int a,int k=Q-2,int ans=1)
{
	for(;k;a=(a*a)%Q,k>>=1) if(k&1) ans=ans*a%Q;
	return ans;
}

void lagrange()
{
	w[0]=1;
	int ans=0;
	for(int i=1;i<=k+2;i++) //i^k
	{
		y[i]=(y[i-1]+qkw(i,k))%Q;
		w[i]=w[i-1]*i%Q;
	}
	pre[0]=suf[k+3]=1;
	for(int i=1;i<=k+2;i++) pre[i]=pre[i-1]*(n-i+Q)%Q;
	for(int i=k+2;i;i--) suf[i]=suf[i+1]*(n-i+Q)%Q;
	for(int i=1;i<=k+2;i++)
	{
		int a=pre[i-1]*suf[i+1]%Q*y[i]%Q;
		int b=qkw((w[i-1]*((k+2-i)&1?-1:1)*w[k+2-i]%Q+Q)%Q);
		mod(ans,a*b%Q);
	}
	fw(ans);
}

signed main()
{
	n=fr(),k=fr();
	lagrange();
	return 0;
}
```



[P5437 【XR-2】约定](https://www.luogu.com.cn/problem/P5437)

无向图生成树个数 $n^{n-2}$，因为 $prufer$ 序列和数一一对应，长度 $n-2$，取值 $[1,n]$

边数 $(n-1)n^{n-2}$，总边数 $\frac {n(n-1)}2$，每条边等价，所以每条边的出现次数是 $2n^{n-3}$

$$
\begin{aligned}
ans&=\frac{2n^{n-3}\sum_{i=1}^{n-1}\sum_{j=i+1}^n(i+j)^k}{n^{n-2}}\\
&=\frac{2[\sum_{i=1}^n\sum_{j=1}^n(i+j)^k-\sum_{i=1}^n(2i)^k]}{2n}\\

\end{aligned}
$$

不难想到直接令 $d=i+j$，打表发现出现次数是 $d-1[d \leq n]$，$2n-d+1[d \gt n]$

$$
\begin{aligned}
ans&=\frac{\sum_{d=1}^{n} (d-1)d^k+\sum_{d=n+1}^{2n}(2n-d+1)d^k-2^k\sum_{i=1}^ni^k}{n}
\end{aligned}
$$

全部都是 $d^k$ 形式，直接大力优化 $lagrange$ 是 $O(k)$ 的

# FFT

快速傅里叶变换

[「Note」Fast Fourier Transform - hongzy 的博客](https://www.luogu.com.cn/blog/hongzy/solution-p3803)

[题解 P3803 【模板】多项式乘法（FFT） 一扶苏一 的博客 - 洛谷博客 (luogu.com.cn)](https://www.luogu.com.cn/blog/fusu2333/solution-p3803)

[Algorithm: 多项式乘法 Polynomial Multiplication: 快速傅里叶变换 FFT / 快速数论变换 NTT - BrianPeng - 博客园 (cnblogs.com)](https://www.cnblogs.com/BrianPeng/p/12251447.html)

给定系数表达式的两个多项式 $F(x),G(x)$，求乘积多项式 $H(x)=F(x)G(x)$ 的系数表达式

暴力用系数表达式乘的话是 $O(n^2)$ 的，但是如果我们知道如果知道点值表达式可以直接乘起来得到 $H$ 的点值表达式，然后再把点值表达式转成系数表达式

FFT 在 $O(n\log n)$ 实现 DFT 和 IDFT 来完成

 

## 单位根

满足 $z^n=1$ 的 $n$ 个复数根

把单位圆 $n$ 等分，$\omega_{n}^k=\cos \frac{2 \pi k}n+i \sin \frac{2 \pi k}n$

FFT 要求 $n=2^b$，补足高位系数补 $0$​

设 $n$  是一个正偶数，且 $n=2m$
$$
(\omega_n^k)^2=\omega_m^k\\
\omega_n^{m+k}=-\omega_n^k
$$

## DFT

$$
b_k=\sum_{i=0}^{n-1} a_i\times \omega_{n}^{ki}
$$

## IDFT

$$
a_k=\frac 1n \sum_{i=0}^{n-1} b_i\omega_{n}^{-ki}
$$

## 算法实现

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define cpx complex<double>
using namespace std;

const int N=(1<<22)+10;
const double Tpi=2.0*acos(-1.0);
int n,m,w,ivs;
cpx f[N],g[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
inline int max(int a,int b){return a>b?a:b;}
inline int min(int a,int b){return a<b?a:b;}

void FFT(int n,cpx *A)
{
	if(n==1) return;
	int m=n>>1;
	cpx A0[m+1],A1[m+1];
	for(int i=0;i<=n-1;i+=2)
		A0[i>>1]=A[i],A1[i>>1]=A[i+1];
	FFT(m,A0),FFT(m,A1);
	cpx wnk(1,0),dlt(cos(Tpi/n),ivs*sin(Tpi/n)); //wnk wn1
	for(int k=0;k<=m-1;k++,wnk*=dlt)
	{
		A[k]=A0[k]+wnk*A1[k];
		A[m+k]=A0[k]-wnk*A1[k];
	}
}

void FFT(cpx *f){ivs=1;FFT(w,f);} //k
void IFFT(cpx *f){ivs=-1;FFT(w,f);} //-k

int main()
{
	n=fr(),m=fr();
	for(int i=0;i<=n;i++) f[i]=fr();
	for(int j=0;j<=m;j++) g[j]=fr();
	w=1;while(w<=n+m) w<<=1;
	FFT(f),FFT(g);
	for(int i=0;i<=w;i++) f[i]*=g[i];
	IFFT(f);
	for(int i=0;i<=n+m;i++)
		fw((int)(f[i].real()/w+0.5)),pt; //一定要先/w 再上取整
	return 0;
}
```

# NTT

快速数论变换

FFT 需要用到单位根和复数，效率很低，并且有精度误差，NTT 全部采用整数来优化

## 原根

**阶**

欧拉定理 $(a,m)=1,a^{\varphi(m)} \equiv 1 \pmod m$

满足 $a^n \equiv 1 \pmod m$ 的 $n$ 称作 $a$ 模 $m$ 的阶

**原根**

$a$ 模 $m$ 的阶为 $\varphi(m)$，则 $a$ 为 $m$ 的原根

当 $p=998244353$ 是，一个原根 $g=3$

原根的次幂以 $\varphi(p)$ 为循环

```
p=7 g=3
1 3 2(9) 6 4(18) 5(12) | 1(15) 3 ....
```

发现这和 $\omega_{n}^k$ 很像，用原根的次幂来代替单位根，$n= \varphi(p)$

![image-20240223151607888](https://cdn.jsdelivr.net/gh/xyzfrozen/Cdn/img/202402231516224.png)



## 算法实现

单位根：$\omega_{n}^k=\cos \frac{2 \pi k}n+i \sin \frac{2 \pi k}n$

原根：$g^{\frac {\varphi(p)}n}$

FFT
$$
b_k=\sum_{i=0}^{n-1} a_i\times \omega_{n}^{ki}\\
a_k=\frac 1n \sum_{i=0}^{n-1} b_i\omega_{n}^{-ki}
$$
NTT
$$
b_k=\sum_{i=0}^{n-1} a_i\times (g^{\frac {\varphi(p)}n})^{ki}\\
a_k=\frac 1n \sum_{i=0}^{n-1} b_i \times (g^{\frac {\varphi(p)}n})^{-ki}
$$
预处理一下原根

# Problem

多项式如果要多次卷积，一定要清空！！！！！本质上是个多项式函数，所以可能有些地方出现意想不到的值，必须清空

[P3723 [AH2017/HNOI2017\] 礼物](https://www.luogu.com.cn/problem/P3723)
$$
\begin{gathered}
\begin{aligned}\sum_{i=1}^n(a_i-b_i+c)^2\end{aligned} \\
=\sum_{i=1}^n(a_i-b_i)^2+c^2+2c(a_i-b_i) \\
=\sum_{i=1}^na_i^2+b_i^2-2a_ib_i+c^2+2a_ic-2b_ic \\
\begin{aligned}&=\sum_{i=1}^na_i^2+\sum_{i=1}^nb_i^2-2\sum_{i=1}^na_ib_i+nc^2+2c\sum_{i=1}^n(a_i-b_i)\end{aligned} 
\end{gathered}
$$
最大化 $\sum_{i=1}^n a_ib_i$

多项式算法都是为了加速卷积

我们翻转 $b$ 数组，发现 $\sum_{i=1}^na_ib_{n-i+1}$ 是等价的，这玩意就很卷积，会对 $H(n)$ 的系数产生贡献

断环成链后，复制一份 $b$ ，那么 $H(n\sim2n)$​ 的系数就是所有要的答案 

注意 `NTT` 是 `A` ，别写成 `a` 数组了

```cpp
const int N=4e5+10,Q=998244353,G=3,IG=332748118;
int n,m,w,ivs,ans,sum;
int a[N],b[N],f[N],g[N],org[N][2];
int qkw(int a,int k=Q-2,int ans=1)
{
	for(;k;a=(a*a)%Q,k>>=1) if(k&1) ans=ans*a%Q;
	return ans;
}

void NTT(int n,int *A)
{
	if(n==1) return;
	int m=n>>1;
	int A0[m+1],A1[m+1];
	for(int i=0;i<=n-1;i+=2)
		A0[i>>1]=A[i],A1[i>>1]=A[i+1];
	NTT(m,A0),NTT(m,A1);
	int wnk=1,dlt=org[n][~ivs?0:1];
	for(int k=0;k<=m-1;k++,(wnk*=dlt)%=Q)
	{
		A[k]=(A0[k]+wnk*A1[k]%Q)%Q;
		A[m+k]=(A0[k]-wnk*A1[k]%Q+Q)%Q;
	}
}
void NTT(int *f){ivs=1;NTT(w,f);}
void INTT(int *f){ivs=-1;NTT(w,f);}

signed main()
{
	n=fr(),m=fr();
	for(int i=1;i<=n;i++) f[i]=fr();
	for(int i=1;i<=n;i++) g[i+n]=g[i]=fr();
	memcpy(a,f,sizeof a),memcpy(b,g,sizeof b);
	reverse(g+1,g+1+2*n);
	
	w=1;while(w<=3*n) w<<=1;
	for(int i=2;i<=w;i<<=1)
		org[i][0]=qkw(G,(Q-1)/i),org[i][1]=qkw(IG,(Q-1)/i);
	NTT(f),NTT(g);
	for(int i=0;i<=w;i++) (f[i]*=g[i])%=Q;
	INTT(f);
	int invn=qkw(w);
	for(int i=0;i<=3*n;i++) (f[i]*=invn)%=Q;
	
	for(int i=n;i<=(n<<1);i++) ans=min(ans,-2*f[i]);
	int sa=0,sb=0;
	for(int i=1;i<=n;i++)
		ans+=a[i]*a[i]+b[i]*b[i],sa+=a[i],sb+=b[i];
	int res=1e18;
	for(int c=-100;c<=100;c++)
		res=min(res,n*c*c+2*c*sa-2*c*sb);
	fw(ans+=res);
	return 0;
}
```



[Fuzzy Search ](https://www.luogu.com.cn/problem/CF528D)

多项式卷积做带通配符的字符串匹配问题

因为只有 $4$ 个字符，每个单独考虑，算完答案求交集即可

把当前考虑的字符向左右各扩散 $k$ 个，字符串转 $01$ 串，$s_i=[s_i=c]$，此时 $S,T$ 匹配当且仅当所有 $1$ 匹配

我们想用多项式卷积就必须转乘法，在 $i$ 匹配上当且仅当 $\sum_{j=1}^{m} S_{i+j-1}T_j=\sum T_i$

令 $A_{m-j+1}=T_j$，则 $\sum_{j=1}^{m}S_{i+j-1}A_{m-j+1}$ 会贡献到 $H(i+m)$​ 的系数上

清空 f，g！！！！！！

```cpp
{
	for(;k;a=(a*a)%Q,k>>=1) if(k&1) ans=ans*a%Q;
	return ans;
}

void NTT(int n,int *A)
{

	int m=n>>1;
	int A0[m+1],A1[m+1];
	for(int i=0;i<=n-1;i+=2)
		A0[i>>1]=A[i],A1[i>>1]=A[i+1];
	NTT(m,A0),NTT(m,A1);

	}

void calc()
{
	NTT(f),NTT(g);
	for(int i=0;i<=w;i++) (f[i]*=g[i])%=Q;
	INTT(f);
	int invn=qkw(w);
	for(int i=0;i<=n+m;i++) (f[i]*=invn)%=Q;
}

void solve(char ch)
{
	static int b[N];
	int cnt=0;
	memset(b,0,sizeof b);
	memset(f,0,sizeof f);
	memset(g,0,sizeof g);
	for(int i=1;i<=n;i++)
		if(s[i]==ch)
		{
			int l=max(i-k,1),r=min(n,i+k);
			b[l]++,b[r+1]--;
		}
	for(int i=1;i<=n;i++) b[i]+=b[i-1],f[i]=(b[i]!=0);
	for(int i=1;i<=m;i++) g[i]=(t[i]==ch),cnt+=g[i];
	calc();
	for(int i=m+1;i<=n+m;i++) ans[i]&=(f[i]==cnt);
}

{

		org[i][0]=qkw(G,(Q-1)/i),org[i][1]=qkw(IG,(Q-1)/i);
	fill(ans+1,ans+1+n+m,1);
	reverse(t+1,t+1+m);
	solve('A'),solve('C'),solve('T'),solve('G');
	int res=0;
	for(int i=m+1;i<=n+m;i++) res+=ans[i];
	fw(res);
	return 0;
}
```

[Multiset of Strings - 洛谷 | 计算机科学教育新生态 (luogu.com.cn)](https://www.luogu.com.cn/problem/CF1709F)

非常nb的题目转化

题目等价于一颗完全二叉树，从根节点向下有无穷流量，每个非叶子节点加上 $0/1$ 得到儿子节点，$x$ 向 $fa_x$ 的边为 $c_x \in [0,k]$

先简化问题，如果确定了所有的 $c_x$，那么最大流量是多少，可以在树上 $dp$

$f_{x}=\min(c_x,f_{ls}+f_{rs})$，同时观察这个式子，每一层的节点的 $dp$ 值应该是完全一样的

设 $g_{i,j}$ 表示高度为 $i$ 的节点 $x$，$f_{x}=j$ 的方案数，考察 $\min$ 的触发条件

$g_{i,j}=\sum_{x+y \geq j} g_{i+1,x}g_{i+1,y} [c_x=j]+(k-j)\sum_{x+y=j}g_{i+1,x}g_{i+1,j-x}[c_x \gt j]$

直接卷积，$O(nk\log k)$

答案为 $g_{n,x}g_{n,m-x}[\varnothing \to (0/1)]$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e6+10,Q=998244353,G=3,IG=332748118;
int n,m,k,w,ivs;
int f[N],g[N],org[N][2],dp[20][N],s[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
inline int max(int a,int b){return a>b?a:b;}
inline int min(int a,int b){return a<b?a:b;}
int qkw(int a,int k=Q-2,int ans=1)
{
	for(;k;a=a*a%Q,k>>=1) if(k&1) ans=ans*a%Q;
	return ans;
}

void NTT(int n,int *A)
{
	if(n==1) return;
	int m=n>>1;
	int A0[m+1],A1[m+1];
	for(int i=0;i<=n-1;i+=2)
		A0[i>>1]=A[i],A1[i>>1]=A[i+1];
	NTT(m,A0),NTT(m,A1);
	int wnk=1,dlt=org[n][~ivs?0:1];
	for(int k=0;k<=m-1;k++,(wnk*=dlt)%=Q)
	{
		A[k]=(A0[k]+wnk*A1[k]%Q)%Q;
		A[m+k]=(A0[k]-wnk*A1[k]%Q+Q)%Q;
	}
}
void NTT(int *f){ivs=1,NTT(w,f);}
void INTT(int *f){ivs=-1,NTT(w,f);}

void calc(int d)
{
	NTT(dp[d]);
	for(int i=0;i<=w;i++) (dp[d][i]*=dp[d][i])%=Q;
	INTT(dp[d]);
	int invn=qkw(w);
	for(int i=0;i<=w;i++) (dp[d][i]*=invn)%=Q;
}

signed main()
{
	n=fr(),k=fr(),m=fr();
	fill(dp[1],dp[1]+k+1,1);
	w=1;while(w<=(k<<1)) w<<=1;
	for(int i=2;i<=w;i<<=1)
		org[i][0]=qkw(G,(Q-1)/i),org[i][1]=qkw(IG,(Q-1)/i);
	for(int i=2;i<=n;i++)
	{
		calc(i-1);
		for(int j=(k<<1);~j;j--) s[j]=(s[j+1]+dp[i-1][j])%Q;
		for(int j=0;j<=k;j++)
			dp[i][j]=(s[j]+(k-j)*dp[i-1][j]%Q)%Q;
	}
	calc(n);
	fw(dp[n][m]);
	return 0;
}
```
