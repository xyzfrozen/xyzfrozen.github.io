# Hash

对一种图形进行 $Hash$ 就是要体现这个图的结构，主要表现在 "相邻" 结构，从而隐式的表达图结构

## 树 Hash

对于一棵树，就是通过子树表达相邻

对于有根树，可以直接从根开始，无根树可以从重心开始，如果有两个，就都存下来比较

$Hash$ 函数：$H(x)=\sum_{v \in S} H(v) \times P^{\sum sz_v}$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define fi first
#define se second
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=60,G=1331,Q=99994919;
int n,m,u,v;
int P[N],sz[N],d[N],h[N],H[N][2];
pi S[N];
vector<int> as[N];
vector<int> r;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

void get_w(int x,int rt)
{
	sz[x]=1;
	int mx=0;
	go(v)
	{
		if(v==rt) continue;
		get_w(v,x);
		sz[x]+=sz[v],mx=max(mx,sz[v]);
	}
	mx=max(mx,n-sz[x]);
	if(mx<=(n>>1)) r.pb(x);
}

void dfs(int x,int rt)
{
	h[x]=d[x]*P[1]%Q,sz[x]=1;
	go(v) {if(v!=rt) d[v]=d[x]+1,dfs(v,x);}
	int cnt=0;
	go(v) {if(v!=rt) S[++cnt]={h[v],sz[v]};}
	sort(S+1,S+1+cnt);
	for(int i=1;i<=cnt;i++)
		(h[x]+=S[i].fi*P[sz[x]])%=Q,sz[x]+=S[i].se;
}

void insert(int now)
{
	n=fr();
	for(int i=1;i<=n;i++) as[i].clear();
	for(int i=1;i<=n;i++)
	{
		int p=fr();
		if(p) as[p].pb(i),as[i].pb(p);
	}
	
	r.clear();
	get_w(1,-1);
	for(int i=0;i<r.size();i++)
	{
		d[r[i]]=1,dfs(r[i],-1);
		H[now][i]=h[r[i]];
	}
	if(H[now][0]>H[now][1]) swap(H[now][0],H[now][1]);
}

signed main()
{
	P[0]=1;
	for(int i=1;i<=55;i++) P[i]=P[i-1]*G%Q;
	m=fr();
	for(int i=1;i<=m;i++) insert(i);
	for(int i=1;i<=m;i++)
		for(int j=1;j<=i;j++)
			if(H[i][0]==H[j][0] && H[i][1]==H[j][1]) {fw(j),nl;break;}
	return 0;
}
```



## 图 Hash

除了树，还可以图 $Hash$

体现图结构就是通过相邻的点集进行传递，传递 $i$ 次就相当于距离为 $i$ 的点的结构贡献

$Hash_{x,j} = A \times Hash_{x,j-1}+B \times \sum_{(u,v) \in E} (Hash_{v,j-1}+j)$

这个 $j$ 取个 4，5 次或者直接 $O(n)$ 就差不多，每次迭代后排序 $Hash$ 值一一比对，$+j$ 就是类似字符串 $Hash$ 的 $+s_i$

每次错误概率是 $\frac 1p$ ，进行 $n$ 次相当于完整遍历图结构，$O(nm+n^2\log n)$



## Problem

**[P3167 [CQOI2014\] 通配符匹配](https://www.luogu.com.cn/problem/P3167)**

像通配的题, 我们通常设 $f_{i,j}$ 表示 $S$ 串到 $i$ 位, $T$ 串到 $j$ 位是否可行, 然后在$O(|S||T|)$ 的时间内解决

如何优化呢? 不难发现对于 $S$ 串来说, 其实很多段是可以连续转移的, 也就是那些有通配符的地方才有用

那么就可以设 $f_{i,j}$ 为 $S$ 中的第 $i$ 个通配符，$T$ 串到 $j$ 位是否可行,对于串尾的 $*/?$ 的转移单独考虑

这样的话, 每次转移会转移连续的一段, 我们就是要判断对于 $S$ 和 $T$ 对应的两段是否相同, 直接用哈希比较即可

$O(|S|+10\sum|T|)$

[题解 P3167 【[CQOI2014\]通配符匹配】 - Diu の blog - 洛谷博客 (luogu.com.cn)](https://www.luogu.com.cn/blog/diu/solution-p3167)

```cpp
#include<bits/stdc++.h>
#define ull unsigned long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10;
int n,m,q,k;
int uiq[N],f[20][N];
char s[N],t[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

struct Hash
{
    const ull P=131;
    ull s[N],p[N];
    int n;
    void init(char *str)
    {
        n=strlen(str+1),p[0]=1;
        for(int i=1;i<=n;i++)
        {
            p[i]=p[i-1]*P;
            s[i]=s[i-1]*P+str[i]-'a'+1;
        }
    }
    ull get(int l,int r){return s[r]-s[l-1]*p[r-l+1];}
}S,T;

int main()
{
    scanf("%s",s+1);
    n=strlen(s+1),s[++n]='?',s[n+1]='\0',S.init(s);
    for(int i=1;i<=n;i++)
        if(!(s[i]>='a' && s[i]<='z')) uiq[++k]=i;

    q=fr();
    while(q--)
    {
        scanf("%s",t+1);
        m=strlen(t+1),t[++m]='a',t[m+1]='\0';
        memset(f,0,sizeof f);
        f[0][0]=1,T.init(t);
        for(int i=0;i<=k;i++)
        {
            if(s[uiq[i]]=='*')
                for(int j=1;j<=m;j++)
                    f[i][j]|=f[i][j-1];
            for(int j=0;j<=m;j++)
            {
                if(!f[i][j]) continue;
                int Ls=uiq[i]+1,Rs=uiq[i+1]-1;
                int Lt=j+1,Rt=j+Rs-Ls+1;
                if(S.get(Ls,Rs)==T.get(Lt,Rt))
                    f[i+1][Rt+(s[uiq[i+1]]=='?')]=1;
            }
        }
        puts(f[k][m]?"YES":"NO");
    }
    return 0;
}
```



**[P4323 [JSOI2016\] 独特的树叶](https://www.luogu.com.cn/problem/P4323)**

首先叶子节点一定是入度为 $0$

我们直接删掉这个叶子节点，那么以他的父亲为根的树必然和 $A$ 树的某一个形态同构，可以用树 $Hash$ 来判断同构

求出 $A,B$ 两树以每个点为根的 $Hash$ 值，换根 $O(n)$，然后把 $A$ 的全部存在 `set` 里面，每个 $B$ 的每个叶子节点判断删掉它后父亲的值

这里就要设计一种方便 “撤销贡献” 的 $Hash$ 函数了，我们想到 `xor`

$Hash_x = \oplus (Hash_v \times rnd +sz_v)$

$rnd$ 选的是 99994919 主要是扩大值域，感觉这个函数不如 $Hash_x=\sum Hash_v \times prime(sz_v)$

```cpp
#include<bits/stdc++.h>
#define ull unsigned long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e5+10;
const ull rnd=99994919;
int n;
set<ull> s;

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

struct Tree
{
	int n;
    vector<int> as[N];
    int sz[N],d[N],fa[N];
    ull f[N],g[N];

    void add(int u,int v)
    {
        as[u].pb(v),as[v].pb(u);
        d[u]++,d[v]++;
    }

    void prw(int x,int rt)
    {
        sz[x]=f[x]=1,fa[x]=rt;
        go(v) if(v^rt) prw(v,x),f[x]^=f[v]*rnd+sz[v],sz[x]+=sz[v];
    }

    void dfs(int x,int rt)
    {
        go(v)
        {
            if(v==rt) continue;
            ull res=g[x]^(f[v]*rnd+sz[v]);
            g[v]=f[v],g[v]^=res*rnd+n-sz[v];
            dfs(v,x);
        }
    }
}A,B;

int main()
{
    n=fr(),A.n=n,B.n=n+1;
    for(int i=1;i<n;i++) A.add(fr(),fr());
    for(int i=1;i<=n;i++) B.add(fr(),fr());
    
    A.prw(1,0);
    A.g[1]=A.f[1],A.dfs(1,0);

    for(int i=1;i<=n+1;i++)
        if(B.d[i]!=1) {B.prw(i,0),B.g[i]=B.f[i],B.dfs(i,0);break;}

    for(int i=1;i<=n;i++) s.insert(A.g[i]);
    for(int i=1;i<=n+1;i++)
        if(B.d[i]==1 && s.count(B.g[B.fa[i]]^(B.f[i]*rnd+1ull)))
            {fw(i);return 0;}

    return 0;
}
```



# Exkmp

给定两个字符串 $a,b$，你要求出两个数组：

- $b$ 的 $z$ 函数数组 $z$，即 $b$ 与 $b$ 的每一个后缀的 LCP 长度。
- $b$ 与 $a$ 的每一个后缀的 LCP 长度数组 $p$。

$O(n)$



我们考察一个增量算法，利用 $dp$ 来计算，类似 `kmp`

首先设 $z_i$ 表示 $lcp(b[i,m],b[1,m])$，显然有 $z_1=m$

从 $i=2$ 开始，我们首先维护一个 $[l,r]$ 满足 $l \lt i\,\and\,r=l+z_l-1$ 且 $r$ 最大

```
|       |        |   |    |
1      r-l+1     l   i    r
```

注意我们现在是要在 $i$ 后面完成增加，我们利用 $z_{i-l+1}$

```
|   | p   |        |   | k   |
1 i-l+1  r-l+1     l   i     r
```

则 $[i-l+1,p]=[i,k]=z_{i-l+1}$

 并且我们可以反证得到 $s[p+1] \not =s[k+1]$ ，否则 $z_{i-l=1}$ 可以变大

但是这样有一个小问题，为什么 $k \leq r$ 呢？这是我们不知道的，但是我们知道 $z_i \geq r-i+1$

所以我们把 $z_i$ 赋值成 $\min (z_{i-l+1},r-i+1)$ 这样就可以了

但是还有一个问题，为什么 $i \leq r$ 呢，我们也不知道，同时上面赋值成 $\min$ 可能是会增大的

所以我们还要暴力让 $i$ 向后面匹配，由于匹配一定是会让 $r$ 增加，我们可以保证最多只会增加 $m$ 次

$O(m)$



```cpp
z[1]=m;
for(int i=2,l=0,r=0;i<=m;i++)
{
    if(i<=r) z[i]=min(z[i-l+1],r-i+1);
    while(i+z[i]<=m && t[i+z[i]]==t[z[i]+1]) z[i]++;
    if(r<i+z[i]-1) l=i,r=i+z[i]-1; 
}
```



设 $ext_i=lcp(a[i,n],b[1,m])$

这其实和求 $z_i$ 没有什么不同，我们把一些 $z$ 修正为 $ext$ 即可，修正那些呢？

我们要利用的是 $b$ 的前缀，所以需要给 $ext_i$ 赋值时用到 $z$

利用的是 $a[l,r]=[1,r-l+1]$ 再给 $i$ 后面**增加一段**，增加就是核心思想，推导时都可以从 $[l,r]$ 这个条件出发，再利用 $i-l+1$ 对应的 $z$

```cpp
for(int i=1,l=0,r=0;i<=n;i++)
{
    if(i<=r) ext[i]=min(z[i-l+1],r-i+1);
    while(i+ext[i]<=n && s[i+ext[i]]==t[ext[i]+1]) ext[i]++;
    if(r<i+ext[i]-1) l=i,r=i+ext[i]-1;
}
```



# Manacher

和 `Exkmp` 非常像啊

首先这个算法有一个弊端就是只能求奇数回文串，所以我们想到经典 Trick，我们可以给每个字符之间插入一个 `'#'`，再给头尾各加一个，这样所有长度为 $L$ 的回文串就会变成长度为 $2L+1$ 的回文串，所以数组要开两倍！！

还是利用之前的信息，设 $i+p_i-1$ 最大的 $i$ 为 $mid$，$r=mid+p_{mid}-1$

1. $i \leq r$ 设 $j$ 为 $i$ 关于 $mid$ 的对称点，则 $j=2\times mid-i$

	```
	    |  |  |  |  |
	    l  j mid i  r
	```

	则 $p_i=p_j$ ，当然可能 $i+p_i-1 \gt r$ 了，我们就直接赋值 $p_i=\min(p_j,r-i+1)$ 即可

2. $i \gt r$ 暴力增即可

同样 $O(n)$

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=3e7+10;
int n,ans;
int p[N];
char s[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
    if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
inline int max(int a,int b){return a>b?a:b;}
inline int min(int a,int b){return a<b?a:b;}

int main()
{
    scanf("%s",s+1);
    n=strlen(s+1)<<1;
    for(int i=n;i;i-=2) s[i]=s[i>>1],s[i-1]='#';
    s[++n]='#';
    for(int i=2,mid=0,r=0;i<=n-1;i++)
    {
        if(i<=r) p[i]=min(p[(mid<<1)-i],r-i+1);
        while(i-p[i] && i+p[i]<=n && s[i-p[i]]==s[i+p[i]]) p[i]++;
        if(i+p[i]-1>r) r=i+p[i]-1,mid=i;
        ans=max(ans,p[i]);
        //#a#a#a#a# #a#a#a#a#a#
    }
    fw(ans-1);
    return 0;
}
```
