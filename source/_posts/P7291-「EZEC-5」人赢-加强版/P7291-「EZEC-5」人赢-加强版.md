显然有 $O(n\log n)$ 的做法

我们考虑到一个性质对于 $i \lt j \,\land a_i \lt a_j$ 的一对 $(i,j)$，$i$ 一定劣于 $j$ 

可以分 $a_k$ 和 $k$ 分别在三个区间讨论证明

所以我们可以维护一个类似单调栈的东西

引理 $1$ 对于 $i \lt j \lt k \,\lor a_i \geq a_j \geq a_k$ 的一个三元组 

只用考虑 $f(i,j)$ 和 $f(j,k)$，因为 $f(i,k) \lt f(j,k)$

考虑到序列一定是上升下降的连续段拼接起来，设当前枚举 $i$

1. 对于当前的上升段，根据引理 $1$，只用考虑原来的栈顶 $j$，$ans \leftarrow f(i,j)$，同时更新单调栈

2. 如果当前下降了，这个时候不更新单调栈，因为栈上的点还是可能会变成答案，我们还是只用考虑 $f(i,j)$ 

这是因为对于连续的下降段，肯定只有下降段最右端点有用，更具体的说对于下降段的任意一个点，不是右端点无影响，是右端点肯定只可能与栈顶产生贡献

证明：对于栈上的任意一个点 $k\,[k!=j]$，$f(i,k) \lt f(j,k)$

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
using namespace std;

const int N=1e7+10;
int n,ans;
int a[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

signed main()
{
	n=fr();
	for(int i=1;i<=n;i++)
	{
		a[i]=fr();
		ans=max(ans,a[i]*i);
	}
	
	int mx=0,k=0;
	for(int i=n;i;i--)
	{
		if(a[i]>mx)
		{
			ans=max(ans,mx*(i+k));
			mx=a[i],k=i;
		}
		else ans=max(ans,a[i]*(i+k));
	}
	fw(ans);
	return 0;
}
```
