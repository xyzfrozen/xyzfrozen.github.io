# Introduction

$ODT$ 俗称珂朵莉树，适用于有区间赋值，即 `assign` 操作的题目

`ODT` 用 `set` 维护区间，存储 $[l,r,v]$ 这样的三元组信息

核心操作是 `split` 类似 `fhq` ，全部先分裂再操作

分裂一定要先 $R$，再 $L$，避免迭代器失效

定义

```cpp
#define IT set<node>::iterator
struct node{
	int l,r;
	mutable int v; //可变的，相对于 const
	bool operator<(const node&Q)const{
		return l<Q.l;
	}
};
set<node> s;
```

$Split$

```cpp
IT split(int p)
{
	IT now=s.lower_bound({p,0,0});
	if(now!=s.end() && now->l==p)
		return now;
	now=prev(now);
	if(now->r<p) return s.end();
	int l=now->l,r=now->r,v=now->v;
	s.erase(now);
	s.insert({l,p-1,v});
	return s.insert({p,r,v}).first;
}
```

$assign$

```cpp
void assign(int l,int r,int x)
{
	IT R=split(r+1),L=split(l);
	s.erase(L,R);
	s.insert({l,r,x});
}
```



# Problem

[模板题](https://www.luogu.com.cn/problem/CF896C) 

```cpp
#include<bits/stdc++.h>
#define int long long
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define IT set<node>::iterator
using namespace std;

const int N=1e5+10;
int n,m,seed,vmax,op,l,r,x,y;
int a[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

int Rnd() 
{
    int ret=seed;
    seed=(seed*7+13)%1000000007;
    return ret;
}

struct node{
	int l,r;
	mutable int v;
	bool operator<(const node&Q)const{
		return l<Q.l;
	}
};
set<node> s;

IT split(int p)
{
	IT now=s.lower_bound({p,0,0});
	if(now!=s.end() && now->l==p)
		return now;
	now=prev(now);
	if(now->r<p) return s.end();
	int l=now->l,r=now->r,v=now->v;
	s.erase(now);
	s.insert({l,p-1,v});
	return s.insert({p,r,v}).first;
}

void add(int l,int r,int x)
{
	IT R=split(r+1),L=split(l);
	for(;L!=R;L++)
		L->v+=x;
}

void assign(int l,int r,int x)
{
	IT R=split(r+1),L=split(l);
	s.erase(L,R);
	s.insert({l,r,x});
}

int kth(int l,int r,int k)
{
	IT R=split(r+1),L=split(l);
	vector<pi> vec(0);
	for(;L!=R;L++)
		vec.pb({L->v,L->r-L->l+1});
	sort(vec.begin(),vec.end());
	for(auto it:vec)
	{
		if(k>it.second) k-=it.second;
		else return it.first;
	}
	return -1;
}

int qkw(int a,int k,int Q)
{
	int ans=1,base=a;
	while(k)
	{
		if(k&1) ans=ans*base%Q;
		base=base*base%Q;
		k>>=1;
	}
	return ans;
}

int calc(int l,int r,int k,int Q)
{
	IT R=split(r+1),L=split(l);
	int res=0;
	for(;L!=R;L++)
		(res+=qkw(L->v%Q,k,Q)*(L->r-L->l+1)%Q)%=Q;
	return res;
}

signed main()
{
	n=fr(),m=fr(),seed=fr(),vmax=fr();
	for(int i=1;i<=n;i++)
		s.insert({i,i,(Rnd()%vmax)+1});
	
	while(m--)
	{
		op=(Rnd()%4)+1,l=(Rnd()%n)+1,r=(Rnd()%n)+1;
		if(l>r) swap(l,r);
        if(op==3) x=(Rnd()%(r-l+1))+1;
        else x=(Rnd()%vmax)+1;
        if(op==4) y=(Rnd()%vmax)+1;
        if(op==1) add(l,r,x);
        else if(op==2) assign(l,r,x);
        else if(op==3) fw(kth(l,r,x)),nl;
        else fw(calc(l,r,x,y)),nl;
	}

	return 0;
}
```

```cpp
#define se second

		return l<Q.l;
	}
};
set<node> s;

IT split(int p)
{
	if(p>n) return s.end();
	IT now=s.lower_bound({p,0,0});
	if(now!=s.end() && now->l==p)
		return now;
	now=prev(now);
	if(now->r<p) return s.end();
	int l=now->l,r=now->r,v=now->v;
	s.erase(now);
	s.insert({l,p-1,v});
	return s.insert({p,r,v}).first;
}

void del(int l,int r)
{
	IT R=split(r+1),L=split(l);
	s.erase(L,R);
	s.insert({l,r,0});
}

void move(int l1,int r1,int l2,int r2)
{
	IT R1=split(r1+1),L1=split(l1);
	int res=0;
	for(IT now=L1;now!=R1;now++)
		res+=(now->r-now->l+1)*now->v;
	s.erase(L1,R1);
	s.insert({l1,r1,0});
	if(!res) return;
	
	IT R2=split(r2+1),L2=split(l2);
	if(res>=r2-l2+1)
	{
		s.erase(L2,R2);
		s.insert({l2,r2,1});
		return;
	}
	
	int p=l2-1;
	for(IT now=L2;now!=R2;now++)
	{
		if(!now->v)
		{
			if(res>now->r-now->l+1) res-=now->r-now->l+1,p=now->r;
			else {p=now->l+res-1;break;}
		}
		else p=now->r;
	}
	if(p<l2) return;
	IT R=split(p+1),L=split(l2); //先 R 再 L
	s.erase(L,R);
	s.insert({l2,p,1});
}

int query(int l,int r)
{
	IT R=split(r+1),L=split(l);
	int res=0,len=0;
	for(;L!=R;L++)
		tmp[++len]={L->r-L->l+1,L->v};
	for(int i=1;i<len;i++)
		if(tmp[i+1].se==tmp[i].se)
			tmp[i+1].fi+=tmp[i].fi,tmp[i].fi=0;
	for(int i=1;i<=len;i++)
		res=max(res,tmp[i].fi*(!tmp[i].se));
	return res;
}

int main()
{
	n=fr(),m=fr();
	s.insert({1,n,1});
	
	for(int i=1;i<=m;i++)
	{
		op=fr(),l=fr(),r=fr();
		if(!op) del(l,r);
		else if(op&1)
		{
			int _l=fr(),_r=fr();
			move(l,r,_l,_r);
		}
		else fw(query(l,r)),nl;
	}
	
	return 0;
}
```

[P4979 矿洞：坍塌 ](https://www.luogu.com.cn/problem/P4979)

优化：边 `query` 边 `assign`

```cpp
#include<bits/stdc++.h>
#define pt putchar(' ')
#define nl puts("")
#define pi pair<int,int>
#define pb push_back
#define go(it) for(auto &it:as[x])
#define IT set<node>::iterator
using namespace std;

const int N=5e5+10;
int n,m,l,r;
char a[N];

int fr(){
    int x=0,flag=1;
    char ch=getchar();
    while(ch<'0' || ch>'9'){
        if(ch=='-') flag=-1;
        ch=getchar();
    }
    while(ch>='0' && ch<='9'){
        x=x*10+(ch-'0');
        ch=getchar();
    }
    return x*flag;
}
void fw(int x){
	if(x<0) putchar('-'),x=-x;
    if(x>9) fw(x/10);
    putchar(x%10+'0');
}
int max(int a,int b){return a>b?a:b;}
int min(int a,int b){return a<b?a:b;}

struct node{
	int l,r;
	mutable int v;
	bool operator<(const node&Q)const{
		return l<Q.l;
	}
};
set<node> s;

IT split(int p)
{
	IT now=s.lower_bound({p,0,0});
	if(now!=s.end() && now->l==p)
		return now;
	now=prev(now);
	if(now->r<p) return s.end();
	int l=now->l,r=now->r,v=now->v;
	s.erase(now);
	s.insert({l,p-1,v});
	return s.insert({p,r,v}).first;
}

void assign(int l,int r,int x)
{
	IT R=split(r+1),L=split(l);
	s.erase(L,R);
	s.insert({l,r,x});
}

bool query(int l,int r)
{
	IT R=split(r+1),L=split(l);
	int lt=-1,v=L->v,_l=l,_r=l;
	for(IT now=L;now!=R;now++)
	{
		if(!~lt) lt=now->v;
		else if(now->v!=lt)
		{
			if(_l!=_r) assign(_l,_r,lt);
			return 0;
		}
		_r=now->r;
	}
	
	assign(l,r,v);
	R=split(r+1),L=split(l);
	if(L!=s.begin() && R!=s.end())
	{
		L=prev(L);
		if(L->v==R->v) return 0;
	}
	return 1;
}

int main()
{
	n=fr();
	scanf("%s",a+1);
	for(int i=1;i<=n;i++)
	{
		int j=i;
		while(j<=n && a[j]==a[i]) j++;
		s.insert({i,--j,a[i]-'A'});
		i=j;
	}
	
	m=fr();
	while(m--)
	{
		char op[2];
		scanf("%s",op),l=fr(),r=fr();
		if(op[0]=='A')
		{
			char x=getchar();
			assign(l,r,x-'A');
		}
		else
		{
			if(query(l,r)) puts("Yes");
			else puts("No");
		}
	}

	return 0;
}
```
